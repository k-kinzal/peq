81    /**
82     * @return iterable<string, array{array{accepted: int, rejected: list<string>, errors: list<string>}}>
83     */
84    public static function providerWholeProjectAudit(): iterable
85    {
86        $directory = dirname(__DIR__, 2);
87        $files = (new PhpFileCollector())->collect([$directory.'/src'], [], []);
88        $index = SourceIndex::of($files, $directory);
89        $inspection = new Inspection();
90        $audit = ['accepted' => 0, 'rejected' => [], 'errors' => []];
91        foreach ($index->sources() as $source) {
92            foreach ($inspection->callables($source->statements) as $name => $callable) {
93                $text = file_get_contents($source->path);
94                assert(is_string($text));
95
96                $graph = $inspection->analyze($callable, new DependencyGraph($name, $source->path, $text));
97                ++$audit['accepted'];
98                foreach ($graph->edges as $edge) {
99                    if (!isset($graph->nodes[$edge->from], $graph->nodes[$edge->to])) {
100                        $audit['errors'][] = $name.': dangling dependency';
101                    } elseif ($edge->kind === 'reaching-definition' && $graph->nodes[$edge->from]->variable !== $graph->nodes[$edge->to]->variable) {
102                        $audit['errors'][] = $name.': definition belongs to a different variable';
103                    } elseif ($edge->from === $edge->to) {
104                        $audit['errors'][] = $name.': self dependency';
105                    }
106                }
107            }
108        }
109
110        yield 'every callable under src' => [$audit];
111    }
