265        $edges = array_merge([], ...array_map(static fn (Node $node): array => $graph->edges($node->id()), $graph->nodes()));

267        self::assertSame([], array_values(array_filter($edges, static fn (Edge $edge): bool => $graph->edge($edge->to(), $edge->from()) === null)));
268        self::assertSame([], array_values(array_filter($edges, static fn (Edge $edge): bool => $graph->node($edge->from()) === null || $graph->node($edge->to()) === null)));
269        self::assertContains($kind, NodeKind::cases());

292    {

294        $firstRandom = new RandomSource(7);

296        $firstNodes = new NodeGenerator($firstIds, $firstRandom);
297        $first = $operation(new GraphGenerator(new ClassLikeGraphGenerator($firstNodes, $firstIds, $firstRandom), new MemberGraphGenerator($firstNodes, $firstIds, $firstRandom), new LeafGraphGenerator($firstNodes), $firstIds, $firstRandom), 2);
298        $againRandom = new RandomSource(7);
299        $againIds = new NodeIdGenerator(new NameGenerator($againRandom), $againRandom);
300        $againNodes = new NodeGenerator($againIds, $againRandom);
301        $again = $operation(new GraphGenerator(new ClassLikeGraphGenerator($againNodes, $againIds, $againRandom), new MemberGraphGenerator($againNodes, $againIds, $againRandom), new LeafGraphGenerator($againNodes), $againIds, $againRandom), 2);
302
303        self::assertSame($first->root->id()->toString(), $again->root->id()->toString());
304        self::assertSame(array_map($spell, $first->graph->authoredEdges()), array_map($spell, $again->graph->authoredEdges()));
305        self::assertSame($kind, $again->root->kind());
306    }
