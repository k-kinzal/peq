41    public function testSelectedObjectTypesMatchPHPStan(string $family, string $comment, array $expected): void
42    {
43        $file = sys_get_temp_dir().'/peq-phpdoc-oracle-'.hash('sha256', $comment).'.php';
44        file_put_contents($file, '<?php namespace Doc; use Domain\Target as Item; '.$comment.' function run($value) {}');
45        $container = (new ContainerFactory())->create([$file], []);
46        $resolver = $container->getByType(FileTypeMapper::class);
47        $doc = (new DocParser())->parse($comment);
48        $resolved = $resolver->getResolvedPhpDoc($file, null, null, 'Doc\run', $comment);
49        $tags = match ($family) {
50            'param' => $resolved->getParamTags(),
51            'var' => $resolved->getVarTags(),
52            'return' => ['' => $resolved->getReturnTag()],
53        };
54        $reference = [];
55        foreach ($tags as $name => $tag) {
56            $type = $tag?->getType();
57            $parts = $type instanceof \PHPStan\Type\UnionType ? $type->getTypes() : [$type];
58            $reference[$name] = array_values(array_unique(array_merge([], ...array_map(static fn (?\PHPStan\Type\Type $part): array => $part?->getObjectClassNames() ?? [], $parts))));
59            sort($reference[$name]);
60        }
61        $names = new NameContext(new Collecting());
62        $names->startNamespace(new Name('Doc'));
63        $names->addAlias(new Name('Domain\Target'), 'Item', \PhpParser\Node\Stmt\Use_::TYPE_NORMAL);
64        $block = new DocBlock($doc, new DocScope($names));
65        $actual = [];
66        foreach ((new DocIndex())->types($block, $family) as $name => $value) {
67            $actual[$name] = array_values(array_filter(explode('|', $value->objects()), static fn (string $name): bool => $name !== ''));
68            sort($actual[$name]);
69        }
70
71        self::assertSame($expected, $reference);
72        self::assertSame($reference, $actual);
73    }
