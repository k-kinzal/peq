41    public function testDocumentationResolvesActualCalls(string $declarations, array $expected, int $version = PHP_VERSION_ID): void
42    {
43        $source = "<?php\nnamespace Doc;\nclass Target { function work() {} }\nclass Other { function work() {} }\n".$declarations;
44        $directory = sys_get_temp_dir().'/peq-phpdoc-calls-'.hash('sha256', $source);
45        is_dir($directory) || mkdir($directory, 0o777, true);
46        file_put_contents($directory.'/source.php', $source);
47        $graph = (new PhpStanAnalyzer(phpVersion: $version))->analyze($directory);
48        $calls = [];
49        foreach ($graph->forwardEdges() as $edge) {
50            if ($edge->kind() === EdgeKind::MethodCall && str_ends_with($edge->to()->toString(), '::work')) {
51                $calls[] = $edge->to()->toString();
52            }
53        }
54        sort($calls);
55        $native = new Process([PHP_BINARY, '-r', <<<'PHP'
56            require $argv[1];
57            $graph = (new App\Analyzer\NativeAnalyzer\NativeAnalyzer(phpVersion: (int) $argv[3]))->analyze($argv[2]);
58            if (class_exists(PHPStan\DependencyInjection\ContainerFactory::class, false)) {
59                throw new RuntimeException('Native analysis loaded the PHPStan engine.');
60            }
61            echo App\Analyzer\Graph\GraphSnapshot::of($graph)->toString();
62            PHP, dirname(__DIR__, 2).'/vendor/autoload.php', $directory, (string) $version]);
63        $native->mustRun();
64
65        self::assertSame($expected, $calls);
66        self::assertSame(GraphSnapshot::of($graph)->toString(), $native->getOutput());
67    }

154    /**
155     * Resolves a factory result through a type alias declared in a third source file.
156     */
157    public function testCrossFileAliasesAreAvailableBeforeTheFirstCallerIsRead(): void
158    {
159        $directory = sys_get_temp_dir().'/peq-phpdoc-cross-file-calls';
160        is_dir($directory) || mkdir($directory, 0o777, true);
161        file_put_contents($directory.'/a-caller.php', '<?php namespace Client; function run() { \Provider\Factory::make()->work(); }');
162        file_put_contents($directory.'/m-model.php', '<?php namespace Model; class Target { function work() {} } /** @phpstan-type Item Target */ class Schema {}');
163        file_put_contents($directory.'/z-factory.php', '<?php namespace Provider; /** @phpstan-import-type Item from \Model\Schema as Result */ class Factory { /** @return Result */ static function make() {} }');
164        $graph = (new PhpStanAnalyzer())->analyze($directory);
165        $calls = [];
166        foreach ($graph->forwardEdges() as $edge) {
167            if ($edge->kind() === EdgeKind::MethodCall) {
168                $calls[] = [$edge->from()->toString(), $edge->to()->toString()];
169            }
170        }
171        $native = new Process([PHP_BINARY, '-r', <<<'PHP'
172            require $argv[1];
173            $graph = (new App\Analyzer\NativeAnalyzer\NativeAnalyzer())->analyze($argv[2]);
174            echo App\Analyzer\Graph\GraphSnapshot::of($graph)->toString();
175            PHP, dirname(__DIR__, 2).'/vendor/autoload.php', $directory]);
176        $native->mustRun();
177
178        self::assertSame([['Client\run', 'Model\Target::work']], $calls);
179        self::assertSame(GraphSnapshot::of($graph)->toString(), $native->getOutput());
180    }
