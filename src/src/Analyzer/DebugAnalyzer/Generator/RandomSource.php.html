5namespace App\Analyzer\DebugAnalyzer\Generator;

7/**
8 * The seeded sequence of draws a generated graph is made of.
9 *
10 * Every choice the generators make — how many members a class declares, whether
11 * a method calls another, which word a name is spelled with — is one of three
12 * draws taken from here. The sequence is the source's own state rather than PHP's
13 * process-wide random generator: a graph drawn with a seed is only reproducible
14 * while nothing else touches the stream it is drawn from, and a shared generator
15 * is reseeded by whatever else happens to run, including the destructor of a
16 * random-data library. Owning the state is what makes "same seed, same graph" a
17 * promise the analyzer can keep whatever else the process is doing.
18 *
19 * The draws come from a 32-bit xorshift, computed in integers that never leave
20 * 64 bits, so one seed spells the same graph on every runtime peq supports.
21 *
22 * @visibility parent
23 */
24final class RandomSource
25{
26    /**
27     * The words names are spelled with.
28     *
29     * @var list<string>
30     */
31    private const array WORDS = [
32        'alias', 'amet', 'animi', 'aut', 'beatae', 'culpa', 'dolor', 'eius',
33        'enim', 'error', 'fugit', 'harum', 'illum', 'ipsum', 'iure', 'magni',
34        'minus', 'nemo', 'nihil', 'odio', 'omnis', 'porro', 'quia', 'quod',
35        'rerum', 'sequi', 'sint', 'soluta', 'tempora', 'ullam', 'velit', 'vero',
36        'ad', 'atque', 'commodi', 'dicta', 'earum', 'esse', 'facere', 'fuga',
37        'hic', 'id', 'labore', 'modi', 'natus', 'nobis', 'officia', 'quas',
38        'ratione', 'saepe', 'sed', 'similique', 'sunt', 'ut', 'vitae', 'voluptas',
39    ];
40
41    /**
42     * The constant the seed is mixed with, so that seed zero has a state to start from.
43     */
44    private const int MIX = 0x9E3779B9;
45
46    /**
47     * The mask keeping the state to 32 bits.
48     */
49    private const int MASK = 0xFFFFFFFF;
50
51    /**
52     * The current state of the sequence, never zero.
53     */
54    private int $state;
55
56    /**
57     * @param int $seed The seed the sequence starts from
58     */
59    public function __construct(int $seed)
60    {
61        $state = ($seed ^ self::MIX) & self::MASK;
62        $this->state = $state === 0 ? self::MIX : $state;
63    }
64
65    /**
66     * Draws a number between two bounds, both included.
67     *
68     * @param int $min The smallest number that may be drawn
69     * @param int $max The largest number that may be drawn, no smaller than $min
70     *
71     * @return int The drawn number
72     */
73    public function numberBetween(int $min, int $max): int
74    {
75        assert($min <= $max, 'The largest number that may be drawn must be no smaller than the smallest');
76
77        $state = $this->state;
78        $state ^= ($state << 13) & self::MASK;
79        $state ^= $state >> 17;
80        $state ^= ($state << 5) & self::MASK;
81        $this->state = $state;
82
83        return $min + $state % ($max - $min + 1);
84    }
85
86    /**
87     * Draws a boolean, true and false being equally likely.
88     *
89     * @return bool The drawn boolean
90     */
91    public function boolean(): bool
92    {
93        return $this->numberBetween(0, 1) === 1;
94    }
95
96    /**
97     * Draws a lowercase word.
98     *
99     * @return string The drawn word
100     */
101    public function word(): string
102    {
103        return self::WORDS[$this->numberBetween(0, count(self::WORDS) - 1)];
104    }
105}
