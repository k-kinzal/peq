31     */

33    public function enterNode(Node $node): null
34    {

38            return null;
39        }
40        $scope = new DocScope(clone $this->names->getNameContext(), $this->scope?->class, $this->scope?->parent, $this->scope->localTypes ?? []);
41        if ($node instanceof Stmt\ClassLike) {
42            $scope = new DocScope($scope->names, $node->namespacedName?->toString(), $node instanceof Stmt\Class_ ? $node->extends?->toString() : null);
43        }
44        $this->scope = $scope;
45        if ($comment === null) {
46            return null;
47        }
48        $doc = $this->parser->parse($comment->getText());
49        $this->scope = $scope->withTypes($doc);
50        $block = new DocBlock($doc, $this->scope);
51        $node->setAttribute('peqDocBlock', $block);
52        $names = match (true) {
53            $node instanceof Stmt\ClassLike => [$scope->class ?? ''],
54            $node instanceof Stmt\Function_ => [$node->namespacedName?->toString() ?? ''],
55            $node instanceof Stmt\ClassMethod => [$scope->class.'::'.$node->name->toString()],
56            $node instanceof Node\Param && $node->flags !== 0 && $node->var instanceof Node\Expr\Variable && is_string($node->var->name) => [$scope->class.'::$'.$node->var->name],
57            $node instanceof Stmt\Property => array_map(static fn (Node\PropertyItem $p): string => $scope->class.'::$'.$p->name->toString(), $node->props),
58            default => [],
59        };
60        foreach ($names as $name) {
61            $this->index->blocks[DocIndex::key($name)] = $block;
62        }
63
64        return null;
65    }
66
67    /**
68     * Restores the surrounding class and template declarations.
69     */
70    #[Override]
71    public function leaveNode(Node $node): null
72    {
73        $this->scope = array_pop($this->stack);
