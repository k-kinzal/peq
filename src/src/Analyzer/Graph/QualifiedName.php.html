7/**
8 * A fully qualified PHP name, split into the parts an identifier is built from.
9 *
10 * Analysis meets names as single strings — `App\Billing\Invoice`, `int`, `parent` —
11 * while a node identifier is built from a namespace and a short name. Splitting one
12 * into the other is one rule, so it lives in one type rather than being repeated
13 * wherever an identifier happens to be constructed.
14 *
15 * The same type answers whether a name is a PHP builtin, because that is a question
16 * about the name and not about the node it might become: a builtin never has a
17 * declaration to point at.
18 */
19final class QualifiedName
20{
21    /**
22     * The grammar of a single name segment, as the PHP manual writes it.
23     */
24    private const IDENTIFIER_PATTERN = '/^[a-zA-Z_\x80-\xff][a-zA-Z0-9_\x80-\xff]*$/';
25
26    /**
27     * The grammar of a namespace: identifier segments joined by single separators.
28     */
29    private const NAMESPACE_PATTERN = '/^[a-zA-Z_\x80-\xff][a-zA-Z0-9_\x80-\xff]*(\\\[a-zA-Z_\x80-\xff][a-zA-Z0-9_\x80-\xff]*)*$/';
30
31    /**
32     * Names PHP resolves itself, which therefore have no declaration in any codebase.
33     */
34    private const BUILTIN_NAMES = [
35        'self', 'static', 'parent',
36        'int', 'string', 'float', 'bool', 'array', 'iterable', 'callable',
37        'void', 'object', 'mixed', 'null', 'false', 'true', 'never',
38    ];
39
40    /**
41     * The namespace part, empty when the name is not namespaced.
42     */
43    public readonly string $namespace;
44
45    /**
46     * The last segment of the name.
47     */
48    public readonly string $shortName;
49
50    /**
51     * @example Splitting a namespaced name
52     *     (new \App\Analyzer\Graph\QualifiedName('App\\Domain\\Invoice'))->namespace // => 'App\\Domain'
53     * @example The short name is the last segment
54     *     (new \App\Analyzer\Graph\QualifiedName('App\\Domain\\Invoice'))->shortName // => 'Invoice'
55     * @example A global name has no namespace
56     *     (new \App\Analyzer\Graph\QualifiedName('Invoice'))->namespace // => ''
57     *
58     * @param string $fullName The name as analysis reported it
59     */
60    public function __construct(
61        public readonly string $fullName,
62    ) {
63        $lastSeparator = strrpos($fullName, '\\');
64        $this->namespace = $lastSeparator === false ? '' : substr($fullName, 0, $lastSeparator);
65        $this->shortName = $lastSeparator === false ? $fullName : substr($fullName, $lastSeparator + 1);
66    }
67
68    /**
69     * Reports whether a string is a name PHP would accept for a single symbol.
70     *
71     * A class, method, property, function, constant and enum case are each named by
72     * one segment, and every identifier built from them has the same precondition on
73     * it. The rule is a property of PHP names rather than of any one identifier, so
74     * it is asked here instead of being restated wherever a name is taken in.
75     *
76     * @param string $value The name to check
77     *
78     * @example A written symbol name is admissible
79     *     \App\Analyzer\Graph\QualifiedName::isIdentifier('totalAmount') // => true
80     * @example A qualified name is not one segment
81     *     \App\Analyzer\Graph\QualifiedName::isIdentifier('App\\Domain\\Invoice') // => false
82     * @example Nothing is not a name
83     *     \App\Analyzer\Graph\QualifiedName::isIdentifier('') // => false
84     *
85     * @return bool True when PHP would accept the string as an identifier
86     */
87    public static function isIdentifier(string $value): bool
88    {
89        return $value !== '' && preg_match(self::IDENTIFIER_PATTERN, $value) === 1;
90    }
91
92    /**
93     * Reports whether a string is a namespace PHP would accept.
94     *
95     * @param string $value The namespace to check, without a leading separator
96     *
97     * @example A namespace is identifier segments joined by separators
98     *     \App\Analyzer\Graph\QualifiedName::isNamespace('App\\Domain\\Billing') // => true
99     * @example The global namespace is written as no namespace at all
100     *     \App\Analyzer\Graph\QualifiedName::isNamespace('') // => false
101     *
102     * @return bool True when PHP would accept the string as a namespace
103     */
104    public static function isNamespace(string $value): bool
105    {
106        return $value !== '' && preg_match(self::NAMESPACE_PATTERN, $value) === 1;
107    }
108
109    /**
110     * Reports whether this name is one PHP resolves on its own.
111     *
112     * A builtin type, and the `self`, `static` and `parent` keywords, name nothing
113     * a codebase declares, so no edge should be drawn to them.
114     *
115     * @example A builtin type names nothing a codebase declares
116     *     (new \App\Analyzer\Graph\QualifiedName('int'))->isBuiltinType() // => true
117     * @example A declared class does
118     *     (new \App\Analyzer\Graph\QualifiedName('App\\Domain\\Invoice'))->isBuiltinType() // => false
119     *
120     * @return bool True when the name is builtin
121     */
122    public function isBuiltinType(): bool
123    {
124        return in_array(strtolower($this->fullName), self::BUILTIN_NAMES, true);
125    }
