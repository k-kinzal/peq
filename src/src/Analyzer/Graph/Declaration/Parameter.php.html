7/**
8 * One parameter of a callable declaration, as it is written.
9 *
10 * The graph already records the class-likes a signature names, because those are
11 * dependencies. A parameter is not a dependency; it is part of the shape of the thing
12 * that depends. Recording it separately is what lets a question be asked about the
13 * shape — which handlers take a request, which methods are variadic, which parameters
14 * were promoted into properties — without re-reading the source.
15 *
16 * The type is kept as it was written rather than resolved, because that is the form a
17 * reader recognises and the form the question is usually asked in.
18 */
19final readonly class Parameter
20{
21    /**
22     * @param string      $name     The parameter name, without its leading dollar sign
23     * @param null|string $type     The declared type as written, or null when none was written
24     * @param bool        $optional Whether the declaration gives it a default value
25     * @param bool        $variadic Whether it collects the remaining arguments
26     * @param bool        $byRef    Whether it is taken by reference
27     * @param bool        $promoted Whether it also declares a property of the class
28     */
29    public function __construct(
30        public string $name,
31        public ?string $type = null,
32        public bool $optional = false,
33        public bool $variadic = false,
34        public bool $byRef = false,
35        public bool $promoted = false,
36    ) {
37        assert($this->name !== '', 'A parameter must be named');
38    }
39
40    /**
41     * Writes the parameter the way its declaration writes it.
42     *
43     * The rendering is the source form minus the default value, because the default
44     * is the one part of a parameter that says nothing about how it may be called.
45     *
46     * @example A typed parameter reads as it was written
47     *     (new \App\Analyzer\Graph\Declaration\Parameter('amount', 'int'))->toString() // => 'int $amount'
48     * @example A variadic parameter keeps the spread that makes it one
49     *     (new \App\Analyzer\Graph\Declaration\Parameter('rows', 'string', variadic: true))->toString() // => 'string ...$rows'
50     * @example An untyped parameter is just its name
51     *     (new \App\Analyzer\Graph\Declaration\Parameter('value'))->toString() // => '$value'
52     *
53     * @return string The parameter as source code writes it
54     */
55    public function toString(): string
56    {
57        return ($this->type === null ? '' : $this->type.' ')
58            .($this->byRef ? '&' : '')
59            .($this->variadic ? '...' : '')
60            .'$'.$this->name;
61    }
62}
