11/**
12 * Admission rules are an allowlist. An unrecognized AST never inherits eager flow.
13 */
14final class Rules
15{
16    /**
17     * Returns the rule whose preconditions are met, or an explicit boundary reason.
18     *
19     * @return array{string, string}
20     */
21    public function classify(Node $node): array
22    {
23        if ($node instanceof Stmt) {
24            return match ($node::class) {
25                Stmt\If_::class, Stmt\Expression::class, Stmt\Return_::class,
26                Stmt\While_::class, Stmt\Do_::class, Stmt\For_::class, Stmt\Foreach_::class,
27                Stmt\Break_::class, Stmt\Continue_::class,
28                Stmt\Echo_::class, Stmt\Nop::class => ['structured-statement', ''],
29                default => ['UNSUPPORTED_STATEMENT', 'No checked transfer rule for '.$node->getType().'; its body and continuation remain unknown.'],
30            };
31        }
32        if ($node instanceof Expr\Variable) {
33            return is_string($node->name) ? ['local-read', ''] : ['DYNAMIC_VARIABLE', 'The variable name depends on a runtime value.'];
34        }
35        if ($node instanceof Expr\Assign || $node instanceof Expr\AssignOp || $node instanceof Expr\PreInc || $node instanceof Expr\PreDec || $node instanceof Expr\PostInc || $node instanceof Expr\PostDec) {
36            return $this->localWrite($node);
37        }
38        if (in_array($node::class, [Expr\Ternary::class, Expr\BinaryOp\BooleanAnd::class, Expr\BinaryOp\BooleanOr::class, Expr\BinaryOp\LogicalAnd::class, Expr\BinaryOp\LogicalOr::class, Expr\BinaryOp\Coalesce::class], true)) {
39            return ['conditional-expression', ''];
40        }
41        if ($node instanceof Expr\CallLike) {
42            return $node->isFirstClassCallable() ? ['CALLABLE_CREATION', 'A callable is created, not invoked; target binding is not solved.'] : ['OPAQUE_CALL', 'Return values, implicit local reads, reference effects and exceptional continuation are not solved.'];
43        }
44        if ($this->pure($node)) {
45            return ['pure-expression', ''];
46        }
47
48        return ['UNSUPPORTED_EXPRESSION', 'No checked evaluation rule for '.$node->getType().'; conditional evaluation and effects are unknown.'];
49    }
50
51    /**
52     * Checks address and evaluation-order preconditions for a local write.
53     *
54     * @return array{string, string}
55     */
56    public function localWrite(Expr\Assign|Expr\AssignOp|Expr\PostDec|Expr\PostInc|Expr\PreDec|Expr\PreInc $node): array
57    {
58        if ($node instanceof Expr\AssignOp && !$node instanceof Expr\AssignOp\Coalesce && !$this->pure($node->expr)) {
59            return ['EVALUATION_ORDER', 'A compound assignment with effects in its right operand requires an evaluation-order rule.'];
60        }
61
62        return $node->var instanceof Expr\Variable && is_string($node->var->name)
63            ? ['local-write', ''] : ['INDIRECT_WRITE', 'The write may involve aliases, offsets, destructuring or heap storage.'];
64    }
65
66    /**
67     * Admits only side-effect-free, explicitly named expression shapes.
68     */
69    public function pure(Node $node): bool
70    {
71        if ($node instanceof Expr\Variable) {
72            return is_string($node->name);
73        }
74        if (in_array($node::class, [Node\Scalar\Int_::class, Node\Scalar\Float_::class, Node\Scalar\String_::class], true)) {
75            return true;
76        }
77        if ($node instanceof Expr\ConstFetch) {
78            return in_array(strtolower($node->name->toString()), ['true', 'false', 'null'], true);
79        }
80        if (in_array($node::class, [Expr\BooleanNot::class, Expr\UnaryMinus::class, Expr\UnaryPlus::class, Expr\BitwiseNot::class], true)) {
81            return $this->pure($node->expr);
82        }
83        if ($node instanceof Expr\BinaryOp && in_array($node->getType(), [
84            'Expr_BinaryOp_Plus', 'Expr_BinaryOp_Minus', 'Expr_BinaryOp_Mul', 'Expr_BinaryOp_Div',
85            'Expr_BinaryOp_Mod', 'Expr_BinaryOp_Pow', 'Expr_BinaryOp_Concat', 'Expr_BinaryOp_Identical',
86            'Expr_BinaryOp_NotIdentical', 'Expr_BinaryOp_Equal', 'Expr_BinaryOp_NotEqual',
87            'Expr_BinaryOp_Smaller', 'Expr_BinaryOp_SmallerOrEqual', 'Expr_BinaryOp_Greater',
88            'Expr_BinaryOp_GreaterOrEqual', 'Expr_BinaryOp_Spaceship', 'Expr_BinaryOp_BitwiseAnd',
89            'Expr_BinaryOp_BitwiseOr', 'Expr_BinaryOp_BitwiseXor', 'Expr_BinaryOp_ShiftLeft',
90            'Expr_BinaryOp_ShiftRight', 'Expr_BinaryOp_LogicalXor',
91        ], true)) {
92            return $this->pure($node->left) && $this->pure($node->right);
93        }
94
95        return false;
96    }
