12/**
13 * Scalar proofs rule out hidden object conversions and destructors in checked rules.
14 */
15final class ScalarOrigins
16{
17    /**
18     * Checks a declared runtime parameter type, never trusting a PHPDoc annotation.
19     */
20    public static function parameter(?Node $type): bool
21    {
22        if ($type instanceof Node\NullableType) {
23            return self::parameter($type->type);
24        }
25        if ($type instanceof Node\UnionType) {
26            return $type->types !== [] && count(array_filter($type->types, self::parameter(...))) === count($type->types);
27        }
28
29        return $type instanceof Node\Identifier && in_array(strtolower($type->name), ['int', 'float', 'string', 'bool', 'true', 'false', 'null'], true);
30    }
31
32    /**
33     * Proves every reaching definition scalar before admitting a coercive operand.
34     */
35    public static function expression(Expr $node, State $state, DependencyGraph $graph): bool
36    {
37        if ($node instanceof Expr\Variable && is_string($node->name)) {
38            $definitions = array_keys($state->definitions['$'.$node->name] ?? []);
39
40            return self::inputs($definitions, $graph);
41        }
42        if ($node instanceof Expr\BooleanNot) {
43            return (new Rules())->pure($node->expr);
44        }
45        if ($node instanceof Expr\BinaryOp) {
46            return self::expression($node->left, $state, $graph) && self::expression($node->right, $state, $graph);
47        }
48        if ($node instanceof Expr\UnaryMinus || $node instanceof Expr\UnaryPlus || $node instanceof Expr\BitwiseNot) {
49            return self::expression($node->expr, $state, $graph);
50        }
51
52        return self::literal($node);
53    }
54
55    /**
56     * Distinguishes literal scalar/null values from arrays, objects and constants.
57     */
58    public static function literal(Node $node): bool
59    {
60        return $node instanceof Node\Scalar\Int_ || $node instanceof Node\Scalar\Float_ || $node instanceof Node\Scalar\String_
61            || ($node instanceof Expr\ConstFetch && in_array(strtolower($node->name->toString()), ['true', 'false', 'null'], true));
62    }
63
64    /**
65     * @param list<string> $inputs
66     */
67    public static function inputs(array $inputs, DependencyGraph $graph): bool
68    {
69        return $inputs !== [] && count(array_filter($inputs, static fn (string $id): bool => $graph->scalars[$id] ?? false)) === count($inputs);
70    }
71
72    /**
73     * Rejects potential implicit calls or heap comparisons without scalar operands.
74     */
75    public static function effects(Expr $node, State $state, DependencyGraph $graph): bool
76    {
77        if ($node instanceof Expr\BinaryOp && !in_array($node::class, [Expr\BinaryOp\BooleanAnd::class, Expr\BinaryOp\BooleanOr::class, Expr\BinaryOp\LogicalAnd::class, Expr\BinaryOp\LogicalOr::class, Expr\BinaryOp\Coalesce::class], true)) {
78            return !self::expression($node, $state, $graph);
79        }
80        if ($node instanceof Expr\UnaryMinus || $node instanceof Expr\UnaryPlus || $node instanceof Expr\BitwiseNot) {
81            return !self::expression($node->expr, $state, $graph);
82        }
83        if ($node instanceof Expr\AssignOp && !$node instanceof Expr\AssignOp\Coalesce) {
84            return !self::expression($node->var, $state, $graph) || !self::expression($node->expr, $state, $graph);
85        }
86
87        return false;
88    }
