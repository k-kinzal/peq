7use PHPStan\DependencyInjection\Container;
8use PHPStan\DependencyInjection\ContainerFactory as PhpStanContainerFactory;
9use RuntimeException;
10use Symfony\Component\Yaml\Yaml;

12/**
13 * Builds a PHPStan container that runs peq's collectors.
14 *
15 * PHPStan's container is compiled from configuration files and cannot be extended
16 * once built, so registering a collector means handing PHPStan a configuration file
17 * that declares it. That file is generated for the run and thrown away afterwards,
18 * which is why a scratch directory is involved: Nette's dependency injection derives
19 * its own cache location from the configuration path, so the configuration cannot
20 * simply live next to the analysed project.
21 *
22 * @visibility namespace
23 */
24final readonly class ContainerFactory
25{
26    /**
27     * Analysis level of the generated configuration.
28     *
29     * Collectors run at every level, and level 0 leaves out the type-checking rules
30     * peq has no use for, so nothing is spent on diagnostics that are discarded.
31     */
32    private const int COLLECTOR_ONLY_LEVEL = 0;
33
34    /**
35     * @param PhpStanAutoloader $autoloader Makes PHPStan loadable when peq runs from a PHAR
36     */
37    public function __construct(
38        private PhpStanAutoloader $autoloader = new PhpStanAutoloader(),
39    ) {}
40
41    /**
42     * Builds a container whose analysis reports what the given collectors gathered.
43     *
44     * The configuration is written into the directory PHPStan works in, under a name
45     * derived from its contents. PHPStan compiles a container class for each
46     * configuration path it is given and loads it once per process, so an analysis
47     * with the same collectors as an earlier one — in the next command, or in the
48     * next test of a suite — starts from the class that one compiled instead of
49     * compiling, loading and keeping another. The name carries the contents because
50     * PHPStan also remembers what it read from a path for the rest of the process,
51     * so a path must never be reused for a different configuration.
52     *
53     * The PHP version the sources are read as is written into that configuration
54     * twice: once as the parameter PHPStan configures its own parser and its
55     * version-dependent reflection from, and once as the argument of the re-parsed
56     * source peq reads method bodies with. Both read the same files, so a version
57     * that reached only one of them would leave the bodies of every file the other
58     * cannot parse silently missing from the graph.
59     *
60     * @param list<string>       $files      The files the analysis will cover
61     * @param list<class-string> $collectors The collectors the analysis runs
62     * @param null|int           $phpVersion The PHP version the sources are analysed as, in
63     *                                       PHP_VERSION_ID form, or null to leave the choice to PHPStan
64     *
65     * @return Container The configured PHPStan container
66     *
67     * @throws RuntimeException If the working directory cannot be determined, if PHPStan
68     *                          is not loadable, or if the generated configuration cannot be written
69     */
70    public function create(array $files, array $collectors, ?int $phpVersion = null): Container
71    {
72        $this->autoloader->ensureRegistered();
73
74        $workingDirectory = getcwd();
75        if ($workingDirectory === false) {
76            throw new RuntimeException('Unable to determine current working directory');
77        }
78
79        $services = array_map(
80            static fn (string $collector): array => ['class' => $collector, 'tags' => ['phpstan.collector']],
81            $collectors,
82        );
83        $services[] = ['class' => ReparsedSource::class, 'arguments' => ['phpVersion' => $phpVersion]];
84
85        $parameters = [
86            'customRulesetUsed' => true,
87            'level' => self::COLLECTOR_ONLY_LEVEL,
88        ];
89        if ($phpVersion !== null) {
90            $parameters['phpVersion'] = $phpVersion;
91        }
92
93        $neon = Yaml::dump([
94            'services' => $services,
95            'parameters' => $parameters,
96            'includes' => [],
97        ], 4);
98        $directory = WorkingDirectory::shared();
99        $configuration = $directory->write('phpstan-'.md5($neon).'.neon', $neon);
100
101        return (new PhpStanContainerFactory($workingDirectory))
102            ->create($directory->path, [$configuration], $files)
103        ;
104    }
105}
