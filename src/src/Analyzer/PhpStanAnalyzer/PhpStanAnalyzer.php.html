19/**
20 * Builds the dependency graph by having PHPStan analyse real sources.
21 *
22 * Resolving what a name refers to is the hard part of reading PHP, and PHPStan
23 * already does it: this analyzer runs PHPStan over the selected files with peq's own
24 * collectors registered, and assembles the nodes and edges they report. Using
25 * PHPStan's internals is deliberate — its resolved types are the reason the graph
26 * can tell a call to one class from a call to another of the same method name.
27 */
28final class PhpStanAnalyzer implements Analyzer
29{
30    /**
31     * The identifier PHPStan gives an error that stands for a failure of its own.
32     *
33     * Every other error is a diagnostic about the analysed code, which a dependency
34     * graph has no use for.
35     */
36    private const INTERNAL_ERROR = 'phpstan.internal';
37
38    /**
39     * @param list<string>       $includes         File path patterns to include in analysis
40     * @param list<string>       $excludes         File path patterns to exclude from analysis
41     * @param ContainerFactory   $containerFactory Builds the PHPStan container that runs the collectors
42     * @param PhpFileCollector   $fileCollector    Selects which files the analysis covers
43     * @param GraphBuilder       $graphBuilder     Assembles the graph from what the collectors reported
44     * @param list<class-string> $collectors       The collectors the graph is assembled from: declarations and method bodies unless narrowed
45     */
46    public function __construct(
47        private readonly array $includes = [],
48        private readonly array $excludes = [],
49        private readonly ContainerFactory $containerFactory = new ContainerFactory(),
50        private readonly PhpFileCollector $fileCollector = new PhpFileCollector(),
51        private readonly GraphBuilder $graphBuilder = new GraphBuilder(),
52        private readonly array $collectors = [DependencyCollector::class, InClassMethodCollector::class],
53    ) {}
54
55    /**
56     * Analyses the sources under the given path and builds their dependency graph.
57     *
58     * A path that holds no analysable file is not a failure: it produces an empty
59     * graph, which is what the caller would otherwise have to construct itself.
60     *
61     * @param string $path The file or directory to analyse
62     *
63     * @return Graph The dependency graph of the analysed sources
64     *
65     * @throws AnalysisFailedException If PHPStan cannot be configured to run the collectors,
66     *                                 or if it could not finish analysing one of the files
67     */
68    public function analyze(string $path): Graph
69    {
70        $realPath = realpath($path);
71        $files = $this->fileCollector->collect(
72            [$realPath !== false ? $realPath : $path],
73            $this->includes,
74            $this->excludes,
75        );
76
77        if ($files === []) {
78            return new Graph();
79        }
80
81        $report = $this->collect($this->containerFactory->create($files, $this->collectors), $files);
82
83        return $this->graphBuilder->build($report->symbols());
84    }
85
86    /**
87     * Runs the analysis and reads back what peq's collectors reported.
88     *
89     * Running PHPStan in process is what gives peq the resolved types a dependency
90     * graph needs, and this is the only place that does it: the container, the
91     * analyser and the shape of its result are PHPStan's own vocabulary, and every
92     * other part of peq works on the symbols this hands back.
93     *
94     * PHPStan parses a file in full only when its parser has been told the file is
95     * one of those analysed; any other file is read with its function bodies
96     * stripped, which is enough for reflection but leaves nothing for a collector
97     * to see inside a function. PHPStan's own commands register the analysed files
98     * with the parser before analysing, and so does this.
99     *
100     * A file PHPStan could not finish is reported as a failure rather than left out:
101     * a graph that quietly lacks a file would claim that nothing depends on what
102     * that file declares, which is the one answer an impact analysis must never get
103     * wrong. PHPStan records such a failure as an internal error next to the
104     * diagnostics peq discards, so those are read before the collected data is.
105     *
106     * @param Container    $container The container the analysis runs in
107     * @param list<string> $files     The files to analyse
108     *
109     * @return CollectorReport What the collectors reported
110     *
111     * @throws AnalysisFailedException If the container was built without a parser or an
112     *                                 analyser, or if PHPStan could not finish analysing one of the files
113     */
114    public function collect(Container $container, array $files): CollectorReport
115    {
116        try {
117            $parser = $container->getService('pathRoutingParser');
118            $analyser = $container->getByType(PhpStanAnalyser::class);
119        } catch (MissingServiceException $missing) {
120            throw new AnalysisFailedException(
121                'PHPStan was built without its parser or its analyser, so no source can be analysed.',
122                0,
123                $missing,
124            );
125        }
126        if (!is_object($parser) || $parser::class !== PathRoutingParser::class) {
127            throw new AnalysisFailedException('PHPStan has no parser to tell which files are analysed, so no source can be analysed.');
128        }
129        $parser->setAnalysedFiles($files);
130
131        $result = $analyser->analyse($files, null, null, false, $files);
132        $failures = array_values(array_filter(
133            $result->getErrors(),
134            static fn (Error $error): bool => $error->getIdentifier() === self::INTERNAL_ERROR,
135        ));
136        if ($failures !== []) {
137            throw new AnalysisFailedException(implode(PHP_EOL, array_map(
138                static fn (Error $failure): string => sprintf('PHPStan could not finish analysing %s: %s', $failure->getFile(), $failure->getMessage()),
139                $failures,
140            )));
141        }
142
143        return CollectorReport::of($result->getCollectedData(), $this->collectors);
144    }
145}
