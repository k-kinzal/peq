23/**
24 * Builds the dependency graph by having PHPStan analyse real sources.
25 *
26 * Resolving what a name refers to is the hard part of reading PHP, and PHPStan
27 * already does it: this analyzer runs PHPStan over the selected files with peq's own
28 * collectors registered, and assembles the nodes and edges they report. Using
29 * PHPStan's internals is deliberate — its resolved types are the reason the graph
30 * can tell a call to one class from a call to another of the same method name.
31 */
32final readonly class PhpStanAnalyzer implements Analyzer
33{
34    /**
35     * The identifier PHPStan gives an error that stands for a failure of its own.
36     *
37     * Every other error is a diagnostic about the analysed code, which a dependency
38     * graph has no use for — including the syntax error of a file no PHP version
39     * would accept, which PHPStan reads past and so does peq.
40     */
41    private const string INTERNAL_ERROR = 'phpstan.internal';
42
43    /**
44     * @param list<string>       $includes         File path patterns to include in analysis
45     * @param list<string>       $excludes         File path patterns to exclude from analysis
46     * @param null|int           $phpVersion       The PHP version the sources are read as, in PHP_VERSION_ID
47     *                                             form, or null to leave the version to the analysis engine
48     * @param ContainerFactory   $containerFactory Builds the PHPStan container that runs the collectors
49     * @param PhpFileCollector   $fileCollector    Selects which files the analysis covers
50     * @param GraphBuilder       $graphBuilder     Assembles the graph from what the collectors reported
51     * @param list<class-string> $collectors       The collectors the graph is assembled from: declarations and method bodies unless narrowed
52     * @param null|PhaseCache    $cache            Reuses completed phases, or null for uncached analysis
53     */

56        private array $excludes = [],

58        private ContainerFactory $containerFactory = new ContainerFactory(),
59        private PhpFileCollector $fileCollector = new PhpFileCollector(),
60        private GraphBuilder $graphBuilder = new GraphBuilder(),
61        private array $collectors = [DependencyCollector::class, InClassMethodCollector::class],
62        private ?PhaseCache $cache = null,
63    ) {}

65    /**
66     * Analyses the sources under the given path and builds their dependency graph.
67     *
68     * A path that holds no analysable file is not a failure: it produces an empty
69     * graph, which is what the caller would otherwise have to construct itself.
70     *
71     * @param string $path The file or directory to analyse
72     *
73     * @return Graph The dependency graph of the analysed sources
74     *
75     * @throws AnalysisFailedException If PHPStan cannot be configured to run the collectors,
76     *                                 or if it could not finish analysing one of the files
77     */
78    #[Override]
79    public function analyze(string $path): Graph
80    {
81        $realPath = realpath($path);
82        $files = $this->fileCollector->collect(
83            [$realPath !== false ? $realPath : $path],
84            $this->includes,
85            $this->excludes,
86        );
87
88        if ($files === []) {
89            return new Graph();
90        }
91
92        $build = function () use ($files): Graph {
93            $report = $this->collect($this->containerFactory->create($files, $this->collectors, $this->phpVersion), $files);
94
95            return $this->graphBuilder->build($report->symbols());
96        };
97        if ($this->cache === null) {
98            return CallEnrichment::of($build(), $this->phpVersion);
99        }
100        $slot = serialize([$realPath !== false ? $realPath : $path, getcwd(), $this->includes, $this->excludes, $this->phpVersion, $this->collectors]);
101        $fingerprint = AnalysisInputs::fingerprint($files, [$this->phpVersion, $this->collectors]);
102
103        return $this->cache->remember('phpstan-enriched', $slot, $fingerprint, Graph::class, fn (): Graph => CallEnrichment::of(
104            $this->cache->remember('phpstan-graph', $slot, $fingerprint, Graph::class, $build),
105            $this->phpVersion,
106            $this->cache,
107        ));
108    }

110    /**
111     * Runs the analysis and reads back what peq's collectors reported.
112     *
113     * Running PHPStan in process is what gives peq the resolved types a dependency
114     * graph needs, and this is the only place that does it: the container, the
115     * analyser and the shape of its result are PHPStan's own vocabulary, and every
116     * other part of peq works on the symbols this hands back.
117     *
118     * PHPStan parses a file in full only when its parser has been told the file is
119     * one of those analysed; any other file is read with its function bodies
120     * stripped, which is enough for reflection but leaves nothing for a collector
121     * to see inside a function. PHPStan's own commands register the analysed files
122     * with the parser before analysing, and so does this.
123     *
124     * A file PHPStan could not finish is reported as a failure rather than left out:
125     * a graph that quietly lacks a file would claim that nothing depends on what
126     * that file declares, which is the one answer an impact analysis must never get
127     * wrong. PHPStan records such a failure as an internal error next to the
128     * diagnostics peq discards, so those are read before the collected data is. The
129     * message names the PHP version the sources were read as, because a file the
130     * analysis could not get through is most often one written for another version.
131     *
132     * @param Container    $container The container the analysis runs in
133     * @param list<string> $files     The files to analyse
134     *
135     * @return CollectorReport What the collectors reported
136     *
137     * @throws AnalysisFailedException If the container was built without a parser or an
138     *                                 analyser, or if PHPStan could not finish analysing one of the files
139     */
140    public function collect(Container $container, array $files): CollectorReport
141    {
142        try {
143            $parser = $container->getService('pathRoutingParser');
144            $analyser = $container->getByType(PhpStanAnalyser::class);
145        } catch (MissingServiceException $missing) {
146            throw new AnalysisFailedException(
147                'PHPStan was built without its parser or its analyser, so no source can be analysed.',
148                0,
149                $missing,
150            );
151        }
152        if (!is_object($parser) || $parser::class !== PathRoutingParser::class) {
153            throw new AnalysisFailedException('PHPStan has no parser to tell which files are analysed, so no source can be analysed.');
154        }
155        $parser->setAnalysedFiles($files);
156
157        $result = $analyser->analyse($files, null, null, false, $files);
158        $failures = array_values(array_filter(
159            $result->getErrors(),
160            static fn (Error $error): bool => $error->getIdentifier() === self::INTERNAL_ERROR,
161        ));
162        if ($failures !== []) {
163            throw new AnalysisFailedException(implode(PHP_EOL, array_map(
164                fn (Error $failure): string => sprintf(
165                    'PHPStan could not finish analysing %s as %s: %s',
166                    $failure->getFile(),
167                    $this->analysedVersion(),
168                    $failure->getMessage(),
169                ),
170                $failures,
171            )));
172        }
173
174        return CollectorReport::of($result->getCollectedData(), $this->collectors);
175    }

177    /**
178     * Names the PHP version the sources are read as.
179     *
180     * A file that could not be read is nearly always a file written for a different
181     * version than the one it was read as, so the message about it says which version
182     * that was — including when nobody chose it.
183     *
184     * @return string The version as a project would write it, or a name for the unstated one
185     */
186    public function analysedVersion(): string
187    {
188        if ($this->phpVersion === null) {
189            return 'the PHP version peq runs on';
190        }
191
192        return sprintf('PHP %d.%d', intdiv($this->phpVersion, 10000), intdiv($this->phpVersion, 100) % 100);
193    }
