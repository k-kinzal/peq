7/**
8 * Orchestrates configuration loading from multiple sources.
9 *
10 * Readers are consulted in the order they are given and each one overlays what the
11 * previous ones reported, which is what makes the documented cascade — defaults,
12 * then the configuration file, then the environment, then the command line — hold.
13 * The overlay reaches individual settings rather than whole groups, so passing
14 * `--debug-seed` on the command line does not discard a `debug.depth` written in
15 * the configuration file.
16 *
17 * @phpstan-import-type ConfigField from ConfigReader
18 * @phpstan-import-type ConfigFields from ConfigReader
19 * @phpstan-import-type ConfigGroup from ConfigReader
20 */
21final readonly class ConfigLoader
22{
23    /**
24     * @param list<ConfigReader> $readers The readers to consult, from lowest to highest priority
25     */
26    public function __construct(
27        private array $readers,
28    ) {}
29
30    /**
31     * Loads and merges configuration from all registered readers.
32     *
33     * @return Config A configuration whose every field was read as the type it needs to be
34     *
35     * @throws ConfigException If a reader fails, or if the merged data cannot be read as a configuration
36     */
37    public function load(): Config
38    {
39        $merged = [];
40        foreach ($this->readers as $reader) {
41            $merged = self::overlay($merged, $reader->read());
42        }
43
44        return Config::fromArray($merged);
45    }
46
47    /**
48     * Overlays what one source reported on top of what earlier sources reported.
49     *
50     * A setting reported by the later source wins. Two groups of named settings are
51     * combined setting by setting, because a source that mentions one setting of a
52     * group is not saying anything about the rest of it. A list, on the other hand,
53     * is one value: a source that reports `excludes` replaces the whole list rather
54     * than adding to it, so a narrower configuration can always be narrower.
55     *
56     * @param ConfigFields $base    What earlier sources reported
57     * @param ConfigFields $overlay What this source reported
58     *
59     * @return ConfigFields The combined data
60     */
61    public static function overlay(array $base, array $overlay): array
62    {
63        foreach ($overlay as $key => $value) {
64            $existing = $base[$key] ?? null;
65            if (self::isNamedGroup($existing) && self::isNamedGroup($value)) {
66                $base[$key] = self::overlayGroup($existing, $value);
67
68                continue;
69            }
70            $base[$key] = $value;
71        }
72
73        return $base;
74    }
75
76    /**
77     * Overlays one group of named settings on top of another.
78     *
79     * A group holds single settings and lists of them, never further groups, so a
80     * setting reported by the later source replaces the one below it and the rest of
81     * the group is kept.
82     *
83     * @param ConfigGroup $base    What earlier sources reported for the group
84     * @param ConfigGroup $overlay What this source reported for it
85     *
86     * @return ConfigGroup The combined group
87     */
88    public static function overlayGroup(array $base, array $overlay): array
89    {
90        return array_merge($base, $overlay);
91    }
92
93    /**
94     * Reports whether a value is a group of named settings rather than a single value.
95     *
96     * @param null|ConfigField $value The value to classify
97     *
98     * @return bool True when the value is a non-empty array with only string keys
99     *
100     * @phpstan-assert-if-true ConfigGroup $value
101     */
102    public static function isNamedGroup(array|bool|float|int|string|null $value): bool
103    {
104        if (!is_array($value) || $value === []) {
105            return false;
106        }
107
108        foreach (array_keys($value) as $key) {
109            if (!is_string($key)) {
110                return false;
111            }
112        }
113
114        return true;
115    }
