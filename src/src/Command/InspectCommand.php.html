28/**
29 * Command to inspect the dependency tree of a PHP function, method, or class.
30 *
31 * The command is the input and output boundary. It declares the command line, stacks
32 * the configuration sources, runs the inspection and hands the result to the reporter
33 * the configuration asks for. It decides nothing about analysis or presentation: which
34 * analyzer runs is the use case's decision, and which reporter renders is the reporter
35 * factory's, so this class holds no branch on either.

80    /**
81     * Runs the inspection and writes its report.
82     *
83     * @param InputInterface  $input  The parsed command line
84     * @param OutputInterface $output Where the report and any error is written
85     *
86     * @return int Command::SUCCESS, Command::INVALID for unusable arguments, or
87     *             Command::FAILURE when the configuration or the symbol is rejected
88     */
89    #[Override]
90    protected function execute(InputInterface $input, OutputInterface $output): int
91    {
92        $memoryLimit = $input->getOption('memory-limit');
93        if (is_string($memoryLimit)) {
94            ini_set('memory_limit', $memoryLimit);
95        }
96
97        $target = $input->getArgument('target');
98        if (!is_string($target)) {
99            $output->writeln('<error>The target must be written as a single symbol name.</error>');
100
101            return Command::INVALID;
102        }
103
104        $configPath = $input->getOption('config');
105        if (!is_string($configPath)) {
106            $output->writeln('<error>The --config option must be a path to a file.</error>');
107
108            return Command::INVALID;
109        }
110
111        try {
112            $config = (new ConfigLoader([
113                new DefaultConfigReader(),
114                new EnvConfigReader(),
115                new YamlConfigLoader($configPath),
116                new InputConfigReader($input),
117            ]))->load();
118
119            $result = $this->action->execute(new InspectActionInput(
120                config: $config,
121                target: $target,
122            ));
123        } catch (ConfigException|SymbolNotFoundException $rejected) {
124            $output->writeln(sprintf('<error>%s</error>', $rejected->getMessage()));
125
126            return Command::FAILURE;
127        }
128
129        $this->reporters->create($config)->report($result->graph, $result->symbol->id(), $output);
130
131        return Command::SUCCESS;
132    }
