14/**
15 * Turns the graph peq analysed into the graph a query can be written against.
16 *
17 * The two are the same graph seen from different sides. peq's own is built for
18 * walking away from a symbol and back towards it; a query's is built for matching
19 * patterns, which means labels to select by, properties to compare, and adjacency in
20 * both directions.
21 *
22 * Doing the turning once, up front, is deliberate. A query may match a pattern
23 * against every symbol in a codebase, and working out a symbol's labels each time it
24 * is considered would make the cost of a pattern depend on how often it fails.
25 *
26 * @visibility App
27 */
28final class GraphProjection
29{
30    /**
31     * Returns the analysed graph as a query sees it.
32     *
33     * @param Graph $graph The graph analysis produced
34     *
35     * @example Every analysed symbol becomes a symbol a pattern can match
36     *     $graph = new \App\Analyzer\Graph\Graph();
37     *     $graph->addNode(new \App\Analyzer\Graph\Node\ClassNode(\App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Invoice'), true));
38     *     count(\App\Gql\Element\GraphProjection::of($graph)->nodes()) // => 1
39     * @example A relation is reachable from both of its ends
40     *     $graph = new \App\Analyzer\Graph\Graph();
41     *     $meta = new \App\Analyzer\Graph\FileMeta('/project/src/Invoice.php', 12, 1);
42     *     $caller = new \App\Analyzer\Graph\Node\MethodNode(\App\Analyzer\Graph\NodeId\MethodNodeId::of('App\\Invoice', 'total'), true);
43     *     $called = new \App\Analyzer\Graph\Node\MethodNode(\App\Analyzer\Graph\NodeId\MethodNodeId::of('App\\Money', 'add'), true);
44     *     $graph->addEdge(new \App\Analyzer\Graph\Edge\Usage\MethodCallEdge($caller, $called, $meta));
45     *     count(\App\Gql\Element\GraphProjection::of($graph)->arriving('App\\Money::add')) // => 1
46     *
47     * @return ElementGraph The graph a query is written against
48     */
49    public static function of(Graph $graph): ElementGraph
50    {
51        $nodes = [];
52        foreach ($graph->nodes() as $node) {
53            $nodes[$node->id()->toString()] = self::node($node);
54        }
55
56        $leaving = [];
57        $arriving = [];
58        foreach ($graph->forwardEdges() as $edge) {
59            $datum = self::edge($edge);
60            $leaving[$datum->origin][] = $datum;
61            $arriving[$datum->target][] = $datum;
62        }
63
64        return new ElementGraph($nodes, $leaving, $arriving);
65    }
66
67    /**
68     * Returns one symbol as a query sees it.
69     *
70     * @param Node $node The analysed symbol
71     *
72     * @example A symbol carries the labels a pattern selects it by
73     *     $named = \App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Invoice');
74     *     \App\Gql\Element\GraphProjection::node(new \App\Analyzer\Graph\Node\ClassNode($named, true))->labels // => ['Class', 'ClassLike', 'Resolved']
75     *
76     * @return NodeDatum The symbol
77     */
78    public static function node(Node $node): NodeDatum
79    {
80        return new NodeDatum(
81            $node->id()->toString(),
82            NodeLabels::of($node),
83            NodeProperties::of($node),
84        );
85    }
86
87    /**
88     * Returns one relation as a query sees it.
89     *
90     * Relations retain their source occurrence and resolution evidence in their
91     * identity. Two calls written on the same line remain distinct edges, so TRAIL
92     * forbids revisiting an occurrence rather than a pair of symbols.
93     *
94     * @param Edge $edge The analysed relation
95     *
96     * @example A relation is identified by what it joins and how
97     *     $meta = new \App\Analyzer\Graph\FileMeta('/project/src/Invoice.php', 12, 1);
98     *     $caller = new \App\Analyzer\Graph\Node\MethodNode(\App\Analyzer\Graph\NodeId\MethodNodeId::of('App\\Invoice', 'total'), true);
99     *     $called = new \App\Analyzer\Graph\Node\MethodNode(\App\Analyzer\Graph\NodeId\MethodNodeId::of('App\\Money', 'add'), true);
100     *     $written = new \App\Analyzer\Graph\Edge\Usage\MethodCallEdge($caller, $called, $meta);
101     *     \App\Gql\Element\GraphProjection::edge($written)->origin // => 'App\\Invoice::total'
102     *
103     * @return EdgeDatum The relation
104     */
105    public static function edge(Edge $edge): EdgeDatum
106    {
107        $origin = $edge->from()->toString();
108        $target = $edge->to()->toString();
109
110        return new EdgeDatum(
111            EdgeIdentity::of($edge),
112            EdgeLabels::of($edge->kind()),
113            EdgeProperties::of($edge),
114            $origin,
115            $target,
116        );
117    }
118}
