9/**
10 * What a pattern requires of an element beyond its labels.
11 *
12 * GQL offers two ways of writing the requirement and this holds both, because they
13 * are asked at the same moment. `{ name: 'Invoice' }` requires properties to equal
14 * values; `WHERE p.line > 100` requires a predicate to hold. The first is shorthand
15 * the second could express, and it exists because most requirements are equalities
16 * and writing them out would bury the shape of the pattern in punctuation.
17 *
18 * Both are evaluated while matching rather than after it, which is what makes a
19 * selective pattern cheap: an element that fails here is never followed.
20 */
21final readonly class ElementFilter
22{
23    /**
24     * @param array<string, Expression> $properties The properties that must equal given values
25     * @param null|Expression           $predicate  What must hold of the element, or null when nothing beyond the properties must
26     */
27    public function __construct(
28        public array $properties = [],
29        public ?Expression $predicate = null,
30    ) {}
31
32    /**
33     * Reports whether the filter requires anything at all.
34     *
35     * @example A pattern that writes no requirement requires nothing
36     *     (new \App\Gql\Syntax\Pattern\ElementFilter())->empty() // => true
37     *
38     * @return bool True when neither properties nor a predicate were written
39     */
40    public function empty(): bool
41    {
42        return $this->properties === [] && $this->predicate === null;
43    }
44}
