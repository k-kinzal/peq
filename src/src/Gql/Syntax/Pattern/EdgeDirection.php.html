7/**
8 * Which way a pattern crosses an edge.
9 *
10 * Direction is the whole point of a dependency graph. `(a)-[:methodCall]->(b)` asks
11 * what `a` calls; `(a)<-[:methodCall]-(b)` asks what calls `a`; and the undirected
12 * form asks which methods `a` is involved in a call with, either way round. Those are
13 * three different questions about the same edges, and a query language that could
14 * only ask one of them would answer half of what impact analysis needs.
15 *
16 * A pattern written undirected matches an edge twice when both its ends match the
17 * same node pattern, which is GQL's rule and is worth knowing: a self-call shows up
18 * once in each direction.
19 *
20 * GQL writes seven edge patterns, and three of them — `~[ ]~`, `<~[ ]~` and `~[ ]~>` —
21 * are about undirected edges. peq reads source code, and a relation in source code
22 * always points one way, so the graph has no undirected edge: `~[ ]~` matches nothing,
23 * and the two mixed forms match exactly what their directed half does. The remaining
24 * two, `-[ ]-` and `<-[ ]->`, differ only on undirected edges, so both are `Either`.
25 */
26enum EdgeDirection
27{
28    /** The pattern crosses the edge the way the edge points */
29    case Along;
30
31    /** The pattern crosses the edge against the way it points */
32    case Against;
33
34    /** The pattern crosses the edge whichever way it points */
35    case Either;
36
37    /** The pattern crosses only an edge that points neither way */
38    case Undirected;
39
40    /**
41     * Writes the direction the way a pattern writes it, around a given label.
42     *
43     * @param string $inside What is written between the brackets
44     *
45     * @example A pattern that follows an edge is written pointing forward
46     *     \App\Gql\Syntax\Pattern\EdgeDirection::Along->spelling(':calls') // => '-[:calls]->'
47     * @example One that reads an edge backwards is written pointing back
48     *     \App\Gql\Syntax\Pattern\EdgeDirection::Against->spelling(':calls') // => '<-[:calls]-'
49     * @example One that does not care is written without a point
50     *     \App\Gql\Syntax\Pattern\EdgeDirection::Either->spelling('') // => '-[]-'
51     *
52     * @return string The edge pattern, as a query writes it
53     */
54    public function spelling(string $inside): string
55    {
56        return match ($this) {
57            self::Along => '-['.$inside.']->',
58            self::Against => '<-['.$inside.']-',
59            self::Either => '-['.$inside.']-',
60            self::Undirected => '~['.$inside.']~',
61        };
62    }
63}
