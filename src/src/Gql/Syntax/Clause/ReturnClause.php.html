10/**
11 * What the reader is shown, and how the rows behind it are summarised.
12 *
13 * `RETURN` carries more than the other clauses because GQL puts more in it: the
14 * columns, whether duplicates are dropped, what the rows are grouped by, and the
15 * ordering and paging that apply to the result rather than to the rows going into it.
16 * That last point is the reason they are here rather than as separate clauses after
17 * it — an `ORDER BY` written after a `RETURN` orders what the reader sees, while one
18 * written before it orders what the projection is computed from.
19 *
20 * Returning everything — `RETURN *` — is written as no columns at all, because that
21 * is what it means: every name in scope, decided when the query runs rather than when
22 * it is written.
23 */
24final readonly class ReturnClause implements Clause
25{
26    /**
27     * @param list<Projection> $columns  What the reader is shown, or nothing at all to show every name in scope
28     * @param bool             $distinct Whether rows that repeat are shown once
29     * @param list<Expression> $groupBy  What the rows are grouped by, or nothing when they are not grouped
30     * @param list<SortKey>    $orderBy  How the result is ordered, most significant first
31     * @param null|PageClause  $page     Which stretch of the result is shown, or null for all of it
32     */
33    public function __construct(
34        public array $columns = [],
35        public bool $distinct = false,
36        public array $groupBy = [],
37        public array $orderBy = [],
38        public ?PageClause $page = null,
39    ) {}
40
41    /**
42     * Reports whether the query asks for every name in scope.
43     *
44     * @example A projection that names no column asks for all of them
45     *     (new \App\Gql\Syntax\Clause\ReturnClause())->everything() // => true
46     *
47     * @return bool True when no column was named
48     */
49    public function everything(): bool
50    {
51        return $this->columns === [];
52    }
53}
