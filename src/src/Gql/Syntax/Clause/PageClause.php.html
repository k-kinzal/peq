9/**
10 * Taking a stretch of the rows and dropping the rest.
11 *
12 * `OFFSET` and `LIMIT` are one clause here because they are one operation — skip
13 * some, then keep some — and because GQL lets them be written together. Either may
14 * be absent: written alone, `LIMIT` skips nothing and `OFFSET` keeps everything after
15 * what it skipped.
16 *
17 * On a graph query this is a safety valve as much as a convenience. A pattern with an
18 * unbounded quantifier over a large codebase can match a great many paths, and a
19 * limit is how a reader asks for an answer rather than for all of them.
20 */
21final readonly class PageClause implements Clause
22{
23    /**
24     * @param int      $offset How many rows to skip
25     * @param null|int $limit  How many to keep after that, or null for all of them
26     */
27    public function __construct(
28        public int $offset = 0,
29        public ?int $limit = null,
30    ) {
31        assert($this->offset >= 0, 'A page cannot skip a negative number of rows');
32        assert($this->limit === null || $this->limit >= 0, 'A page cannot keep a negative number of rows');
33    }
34}
