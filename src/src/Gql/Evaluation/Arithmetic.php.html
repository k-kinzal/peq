19/**
20 * Arithmetic, under GQL's rules about what mixing kinds of number produces.
21 *
22 * Three rules decide everything here, and all three are the standard's. An expression
23 * on two exact numbers produces an exact one — `0.1 + 0.2` is `0.3` — which
24 * `ExactArithmetic` works out. An expression that meets an approximate number produces
25 * an approximate one, so `1 + 1.5e0` is a float. And an expression that meets the
26 * absence of a value produces the absence of one, so a sum over a property some
27 * symbols do not carry is absent rather than wrong.
28 *
29 * Division by zero is reported rather than turned into infinity the way PHP's own
30 * division would, and an exact result too large to hold is reported rather than turned
31 * into a float.
32 *
33 * @visibility App\Gql
34 */
35final class Arithmetic
36{
37    /**
38     * Applies an arithmetic operator to two values.
39     *
40     * @param BinaryOperator $operator What to apply
41     * @param Datum          $left     The value on its left
42     * @param Datum          $right    The value on its right
43     *
44     * @example Two exact numbers make an exact one
45     *     \App\Gql\Evaluation\Arithmetic::apply(\App\Gql\Syntax\Expression\BinaryOperator::Add, new \App\Gql\Datum\DecimalDatum(1, 1), new \App\Gql\Datum\DecimalDatum(2, 1))->toText() // => '0.3'
46     * @example An approximate number makes the result approximate
47     *     \App\Gql\Evaluation\Arithmetic::apply(\App\Gql\Syntax\Expression\BinaryOperator::Add, new \App\Gql\Datum\IntegerDatum(1), new \App\Gql\Datum\FloatDatum(1.5))->kind() // => \App\Gql\Datum\DatumKind::Float
48     * @example An absent value makes the result absent
49     *     \App\Gql\Evaluation\Arithmetic::apply(\App\Gql\Syntax\Expression\BinaryOperator::Add, new \App\Gql\Datum\IntegerDatum(1), new \App\Gql\Datum\NullDatum())->kind() // => \App\Gql\Datum\DatumKind::Null
50     *
51     * @return Datum What the operator produced
52     *
53     * @throws GqlException             If a value is not a number, the divisor is zero, or an exact result is out of range
54     * @throws InvalidArgumentException If the operator is not an arithmetic one
55     */
56    public static function apply(BinaryOperator $operator, Datum $left, Datum $right): Datum
57    {
58        if (!in_array($operator, [BinaryOperator::Add, BinaryOperator::Subtract, BinaryOperator::Multiply, BinaryOperator::Divide], true)) {
59            throw new InvalidArgumentException(sprintf('%s is not an arithmetic operator', $operator->spelling()));
60        }
61        if ($left->kind() === DatumKind::Null || $right->kind() === DatumKind::Null) {
62            return new NullDatum();
63        }
64        $first = NumberArgument::exact($left);
65        $second = NumberArgument::exact($right);
66        if ($first !== null && $second !== null) {
67            if ($operator === BinaryOperator::Add) {
68                return ExactArithmetic::add($first, $second);
69            }
70            if ($operator === BinaryOperator::Subtract) {
71                return ExactArithmetic::subtract($first, $second);
72            }
73
74            return $operator === BinaryOperator::Multiply
75                ? ExactArithmetic::multiply($first, $second)
76                : ExactArithmetic::divide($first, $second);
77        }
78
79        return self::approximate($operator, (float) NumberArgument::of($left), (float) NumberArgument::of($right));
80    }

82    /**
83     * Applies an arithmetic operator to two approximate numbers.
84     *
85     * @param BinaryOperator $operator What to apply
86     * @param float          $left     The number on its left
87     * @param float          $right    The number on its right
88     *
89     * @example Approximate division keeps its fraction
90     *     \App\Gql\Evaluation\Arithmetic::approximate(\App\Gql\Syntax\Expression\BinaryOperator::Divide, 7.0, 2.0)->toText() // => '3.5'
91     * @example It still cannot divide by zero
92     *     \App\Gql\Evaluation\Arithmetic::approximate(\App\Gql\Syntax\Expression\BinaryOperator::Divide, 1.0, 0.0) // throws \App\Gql\GqlException: division by zero
93     *
94     * @return FloatDatum The result
95     *
96     * @throws GqlException             If the divisor is zero
97     * @throws InvalidArgumentException If the operator is not an arithmetic one
98     */
99    public static function approximate(BinaryOperator $operator, float $left, float $right): FloatDatum
100    {
101        if (!in_array($operator, [BinaryOperator::Add, BinaryOperator::Subtract, BinaryOperator::Multiply, BinaryOperator::Divide], true)) {
102            throw new InvalidArgumentException(sprintf('%s is not an arithmetic operator', $operator->spelling()));
103        }
104        if ($operator === BinaryOperator::Divide && $right === 0.0) {
105            throw GqlException::because(StatusCode::DivisionByZero, 'a number cannot be divided by zero');
106        }
107
108        $result = $operator === BinaryOperator::Divide ? $left / $right : $left * $right;
109        if ($operator === BinaryOperator::Add) {
110            $result = $left + $right;
111        }
112        if ($operator === BinaryOperator::Subtract) {
113            $result = $left - $right;
114        }
115
116        return new FloatDatum($result);
117    }

119    /**
120     * Returns a number with its sign reversed.
121     *
122     * @param Datum $value The number
123     *
124     * @example An exact number stays exact
125     *     \App\Gql\Evaluation\Arithmetic::negate(new \App\Gql\Datum\DecimalDatum(15, 1))->toText() // => '-1.5'
126     * @example The absence of a number has no sign to reverse
127     *     \App\Gql\Evaluation\Arithmetic::negate(new \App\Gql\Datum\NullDatum())->kind() // => \App\Gql\Datum\DatumKind::Null
128     *
129     * @return Datum The number, negated
130     *
131     * @throws GqlException If the value is not a number, or its negation is out of range
132     */
133    public static function negate(Datum $value): Datum
134    {
135        if ($value->kind() === DatumKind::Null) {
136            return new NullDatum();
137        }
138        $exact = NumberArgument::exact($value);
139        if ($exact === null) {
140            return new FloatDatum(-(float) NumberArgument::of($value));
141        }
142
143        return ExactArithmetic::subtract(new IntegerDatum(0), $exact);
144    }
