13/**
14 * Which rule an operator written before one value follows.
15 *
16 * The null tests and the truth tests — `IS [NOT] NULL`, `IS [NOT] TRUE`, `IS [NOT]
17 * FALSE`, `IS [NOT] UNKNOWN` — are the only expressions in GQL that can never come
18 * out undecided, and that is what they are for: they are the way out of three-valued
19 * logic, the questions about an absent value that have a definite answer.
20 *
21 * @visibility App\Gql
22 */
23final class UnaryOperation
24{
25    /**
26     * Applies an operator to one value.
27     *
28     * @param UnaryOperator $operator What to apply
29     * @param Datum         $operand  The value it is applied to
30     *
31     * @example A negation follows three-valued logic
32     *     \App\Gql\Evaluation\UnaryOperation::apply(\App\Gql\Syntax\Expression\UnaryOperator::Not, new \App\Gql\Datum\NullDatum())->kind() // => \App\Gql\Datum\DatumKind::Null
33     * @example A null test always has an answer
34     *     \App\Gql\Evaluation\UnaryOperation::apply(\App\Gql\Syntax\Expression\UnaryOperator::IsNull, new \App\Gql\Datum\NullDatum())->toText() // => 'TRUE'
35     * @example A sign is arithmetic
36     *     \App\Gql\Evaluation\UnaryOperation::apply(\App\Gql\Syntax\Expression\UnaryOperator::Negate, new \App\Gql\Datum\IntegerDatum(3))->toText() // => '-3'
37     *
38     * @return Datum What the operator produced
39     *
40     * @throws GqlException If the value is not what the operator accepts
41     */
42    public static function apply(UnaryOperator $operator, Datum $operand): Datum
43    {
44        return match ($operator) {
45            UnaryOperator::Not => Logic::datum(Logic::negate(Logic::truth($operand))),
46            UnaryOperator::Negate => Arithmetic::negate($operand),
47            UnaryOperator::Identity => self::identity($operand),
48            UnaryOperator::IsNull => Logic::datum($operand->kind() === DatumKind::Null),
49            UnaryOperator::IsNotNull => Logic::datum($operand->kind() !== DatumKind::Null),
50            UnaryOperator::IsTrue => Logic::datum(Logic::truth($operand) === true),
51            UnaryOperator::IsNotTrue => Logic::datum(Logic::truth($operand) !== true),
52            UnaryOperator::IsFalse => Logic::datum(Logic::truth($operand) === false),
53            UnaryOperator::IsNotFalse => Logic::datum(Logic::truth($operand) !== false),
54            UnaryOperator::IsUnknown => Logic::datum(Logic::truth($operand) === null),
55            UnaryOperator::IsNotUnknown => Logic::datum(Logic::truth($operand) !== null),
56        };
57    }
58
59    /**
60     * Returns a number as it stands, having checked that it is one.
61     *
62     * A plus sign written before a value changes nothing about it, but it still says
63     * the value is meant to be a number, and a query that writes it over something
64     * else has made a mistake worth reporting.
65     *
66     * @param Datum $operand The value
67     *
68     * @example A number written with a sign is the number
69     *     \App\Gql\Evaluation\UnaryOperation::identity(new \App\Gql\Datum\IntegerDatum(3))->toText() // => '3'
70     * @example Something that is not a number is reported
71     *     \App\Gql\Evaluation\UnaryOperation::identity(new \App\Gql\Datum\StringDatum('3')) // throws \App\Gql\GqlException: invalid value type
72     *
73     * @return Datum The value
74     *
75     * @throws GqlException If the value is not a number
76     */
77    public static function identity(Datum $operand): Datum
78    {
79        if ($operand->kind() === DatumKind::Null) {
80            return $operand;
81        }
82        NumberArgument::of($operand);
83
84        return $operand;
85    }
