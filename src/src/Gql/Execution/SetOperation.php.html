15/**
16 * Combining what two runs of clauses answered.
17 *
18 * Set operations are how one query asks a question with two shapes: "everything that
19 * reaches this class, whether by calling it or by extending it" is two patterns and
20 * one answer. The difference and the intersection are the ones a question about
21 * layering reaches for — what the controllers reach that the domain does not.
22 *
23 * Both sides must have the same number of columns, because a row of one has to be a
24 * row of the other for any of this to mean anything. The headings are taken from the
25 * left, which is the side a reader wrote first.
26 *
27 * @visibility App\Gql
28 */
29final class SetOperation
30{
31    /**
32     * Returns the two results combined.
33     *
34     * @param SetOperator $operator How to combine them
35     * @param ResultTable $left     What the run on the left answered
36     * @param ResultTable $right    What the run on the right answered
37     *
38     * @example Keeping everything keeps the rows of both
39     *     $row = new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]);
40     *     $column = [new \App\Gql\Result\ResultColumn('n', 'INT64')];
41     *     $table = new \App\Gql\Result\ResultTable($column, [$row]);
42     *     count(\App\Gql\Execution\SetOperation::combine(\App\Gql\Syntax\SetOperator::UnionAll, $table, $table)->rows) // => 2
43     * @example Dropping what repeats keeps one of them
44     *     $row = new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]);
45     *     $column = [new \App\Gql\Result\ResultColumn('n', 'INT64')];
46     *     $table = new \App\Gql\Result\ResultTable($column, [$row]);
47     *     count(\App\Gql\Execution\SetOperation::combine(\App\Gql\Syntax\SetOperator::Union, $table, $table)->rows) // => 1
48     *
49     * @return ResultTable The two combined
50     *
51     * @throws GqlException If the two sides do not have the same number of columns
52     */
53    public static function combine(SetOperator $operator, ResultTable $left, ResultTable $right): ResultTable
54    {
55        $rows = match ($operator) {
56            SetOperator::Otherwise => null,
57            SetOperator::UnionAll => [...$left->rows, ...$right->rows],
58            SetOperator::Union => self::once([...$left->rows, ...$right->rows]),
59            SetOperator::Intersect, SetOperator::Except => self::once(self::sharing($left->rows, $right->rows, $operator === SetOperator::Intersect)),
60        };
61        if ($rows === null) {
62            return $left->rows === [] ? $right : $left;
63        }
64        self::requireSameShape($left, $right);
65
66        return new ResultTable(self::retyped($left->columns, $rows), $rows);
67    }

69    /**
70     * Returns the columns of a combined result, typed by what they hold once combined.
71     *
72     * A column's type describes its values, and a union holds the values of both
73     * sides: `RETURN 1 AS n UNION ALL RETURN 'a' AS n` holds a number and a string, so
74     * its column holds any value rather than the whole numbers the left side held.
75     *
76     * @param list<ResultColumn> $columns The columns, named as the left side names them
77     * @param list<ResultRow>    $rows    The rows of the combined result
78     *
79     * @example A column holding the values of two kinds holds any value
80     *     $columns = [new \App\Gql\Result\ResultColumn('n', 'INT64')];
81     *     $rows = [new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]), new \App\Gql\Result\ResultRow([new \App\Gql\Datum\StringDatum('a')])];
82     *     \App\Gql\Execution\SetOperation::retyped($columns, $rows)[0]->type // => 'ANY'
83     *
84     * @return list<ResultColumn> The columns
85     */
86    public static function retyped(array $columns, array $rows): array
87    {
88        $retyped = [];
89        foreach ($columns as $index => $column) {
90            $retyped[] = new ResultColumn($column->heading, ResultTable::typeOf($rows, $index));
91        }
92
93        return $retyped;
94    }

96    /**
97     * Reports two results that cannot be combined as the mistake they are.
98     *
99     * @param ResultTable $left  What the run on the left answered
100     * @param ResultTable $right What the run on the right answered
101     *
102     * @example Two results of the same shape pass without comment
103     *     $table = new \App\Gql\Result\ResultTable([new \App\Gql\Result\ResultColumn('n', 'INT64')], []);
104     *     \App\Gql\Execution\SetOperation::requireSameShape($table, $table) // => null
105     * @example Two of different shapes are reported rather than lined up
106     *     $left = new \App\Gql\Result\ResultTable([new \App\Gql\Result\ResultColumn('n', 'INT64')], []);
107     *     \App\Gql\Execution\SetOperation::requireSameShape($left, \App\Gql\Result\ResultTable::nothing()) // throws \App\Gql\GqlException: syntax error
108     *
109     * @throws GqlException If the two sides do not have the same number of columns
110     */
111    public static function requireSameShape(ResultTable $left, ResultTable $right): void
112    {
113        if (count($left->columns) === count($right->columns)) {
114            return;
115        }
116
117        throw GqlException::because(
118            StatusCode::SyntaxError,
119            sprintf(
120                'two results can only be combined when they have the same columns, and these have %d and %d',
121                count($left->columns),
122                count($right->columns),
123            ),
124        );
125    }

127    /**
128     * Returns the rows with those that repeat shown once.
129     *
130     * @param list<ResultRow> $rows The rows
131     *
132     * @example Rows that hold the same values are shown once
133     *     $rows = [new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]), new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)])];
134     *     count(\App\Gql\Execution\SetOperation::once($rows)) // => 1
135     *
136     * @return list<ResultRow> The rows, without repetition
137     */
138    public static function once(array $rows): array
139    {
140        $seen = [];
141        $kept = [];
142        foreach ($rows as $row) {
143            $key = DatumIdentity::keyOfAll($row->values);
144            if (isset($seen[$key])) {
145                continue;
146            }
147            $seen[$key] = true;
148            $kept[] = $row;
149        }
150
151        return $kept;
152    }

154    /**
155     * Returns the rows of one side that the other side does or does not also have.
156     *
157     * @param list<ResultRow> $left   The rows on the left
158     * @param list<ResultRow> $right  The rows on the right
159     * @param bool            $shared Whether to keep the rows both sides have, rather than the rest
160     *
161     * @example The rows both sides have
162     *     $row = new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]);
163     *     count(\App\Gql\Execution\SetOperation::sharing([$row], [$row], true)) // => 1
164     * @example The rows only the left side has
165     *     $row = new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)]);
166     *     \App\Gql\Execution\SetOperation::sharing([$row], [$row], false) // => []
167     *
168     * @return list<ResultRow> The rows kept
169     */
170    public static function sharing(array $left, array $right, bool $shared): array
171    {
172        $known = [];
173        foreach ($right as $row) {
174            $known[DatumIdentity::keyOfAll($row->values)] = true;
175        }
176
177        $kept = [];
178        foreach ($left as $row) {
179            if (isset($known[DatumIdentity::keyOfAll($row->values)]) === $shared) {
180                $kept[] = $row;
181            }
182        }
183
184        return $kept;
185    }
