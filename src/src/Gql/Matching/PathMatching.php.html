21/**
22 * Finding every way one path pattern matches the graph.
23 *
24 * The search walks the pattern from left to right, carrying every attempt that is
25 * still alive. A symbol pattern at the start is the only one that has to look at the
26 * whole graph; after that, every step follows relations from where the last one
27 * arrived, which is why a pattern that pins down its first symbol is so much cheaper
28 * than one that does not.
29 *
30 * A name already bound is what makes the search cheap in the other direction too:
31 * a pattern whose symbol is bound by an earlier clause starts from that symbol rather
32 * than from everything, so building an answer up in steps costs no more than writing
33 * it as one pattern.
34 *
35 * @visibility App\Gql
36 */
37final readonly class PathMatching
38{

40     * Crossing relations, once or repeatedly.
41     */
42    private EdgeMatching $edges;

55    /**
56     * Returns every way a path pattern matches, as rows of what it bound.
57     *
58     * @param PathPattern $path The pattern
59     * @param BindingRow  $row  What is already bound when the pattern is reached
60     *
61     * @example A pattern that matches nothing binds nothing
62     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
63     *     $pattern = new \App\Gql\Syntax\Pattern\PathPattern([new \App\Gql\Syntax\Pattern\NodePattern('p')]);
64     *     $matching->match($pattern, \App\Gql\Binding\BindingRow::unit()) // => []
65     *
66     * @return list<BindingRow> One row per way the pattern matched
67     *
68     * @throws GqlException If a requirement written in the pattern cannot be worked out
69     */
70    public function match(PathPattern $path, BindingRow $row): array
71    {
72        $rows = [];
73        foreach ($this->matchTerms($path->terms, 0, MatchState::before($row), $path->mode) as $state) {
74            $rows[] = $path->variable === null || $state->path === null
75                ? $state->row
76                : $state->row->with($path->variable, $state->path);
77        }
78
79        return $rows;
80    }

82    /**
83     * Returns every way the rest of a pattern matches from where an attempt stands.
84     *
85     * @param list<PathTerm> $terms The pieces of the pattern
86     * @param int            $index Which piece to match next
87     * @param MatchState     $state How far the attempt has got
88     * @param PathMode       $mode  What the path may visit more than once
89     *
90     * @example A pattern with nothing left to match has matched
91     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
92     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
93     *     count($matching->matchTerms([], 0, $state, \App\Gql\Syntax\Pattern\PathMode::Trail)) // => 1
94     *
95     * @return list<MatchState> The attempts that matched the rest of it
96     *
97     * @throws GqlException If a requirement written in the pattern cannot be worked out
98     */
99    public function matchTerms(array $terms, int $index, MatchState $state, PathMode $mode): array
100    {
101        if ($index >= count($terms)) {
102            return [$state];
103        }
104
105        $term = $terms[$index];
106        $reached = $term instanceof EdgePattern
107            ? $this->edges->matches($term, $state, $mode)
108            : $this->matchStanding($term, $state, $mode);
109
110        $matched = [];
111        foreach ($reached as $next) {
112            array_push($matched, ...$this->matchTerms($terms, $index + 1, $next, $mode));
113        }
114
115        return $matched;
116    }

119     * Returns every way the piece of pattern standing at a symbol matches.
120     *
121     * @param PathTerm   $term  The piece of pattern
122     * @param MatchState $state How far the attempt has got
123     * @param PathMode   $mode  What the path may visit more than once
124     *
125     * @example A piece of pattern that is neither a symbol nor a group matches nothing
126     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
127     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
128     *     $matching->matchStanding(new \App\Gql\Syntax\Pattern\NodePattern(), $state, \App\Gql\Syntax\Pattern\PathMode::Trail) // => []
129     *
130     * @return list<MatchState> The attempts that matched it
131     *
132     * @throws GqlException If a requirement written in the pattern cannot be worked out
133     */
134    public function matchStanding(PathTerm $term, MatchState $state, PathMode $mode): array
135    {
136        if ($term instanceof GroupPattern) {
137            return $this->matchGroup($term, $state, $mode);
138        }
139        if ($term instanceof NodePattern) {
140            return $this->matchNode($term, $state, $mode);
141        }
142
143        return [];
144    }

146    /**
147     * Returns every way a symbol pattern matches from where an attempt stands.
148     *
149     * An attempt that has not started yet considers every symbol the pattern could
150     * be about; one that has arrived somewhere considers only where it arrived, since
151     * that is the symbol the relation before it led to.
152     *
153     * @param NodePattern $pattern What the pattern requires of the symbol
154     * @param MatchState  $state   How far the attempt has got
155     * @param PathMode    $mode    What the path may visit more than once
156     *
157     * @example A pattern over a graph with no symbols matches nothing
158     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
159     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
160     *     $matching->matchNode(new \App\Gql\Syntax\Pattern\NodePattern('p'), $state, \App\Gql\Syntax\Pattern\PathMode::Trail) // => []
161     *
162     * @return list<MatchState> The attempts that matched it
163     *
164     * @throws GqlException If a requirement written in the pattern cannot be worked out
165     */
166    public function matchNode(NodePattern $pattern, MatchState $state, PathMode $mode): array
167    {
168        $matched = [];
169        foreach ($this->candidates($pattern, $state) as $node) {
170            if (!LabelMatching::satisfies($pattern->labels, $node->labels)
171                || !ElementMatching::agrees($pattern->variable, $node, $state->row)
172                || !ElementMatching::satisfies($node, $pattern->variable, $pattern->filter, $state->row, $this->evaluation)
173            ) {
174                continue;
175            }
176            if ($state->current === null && !PathModeRule::allowsNode($mode, $state, $node)) {
177                continue;
178            }
179
180            $arrived = $state->current === null ? $state->startingAt($node) : $state;
181            $matched[] = $pattern->variable === null ? $arrived : $arrived->bind($pattern->variable, $node);
182        }
183
184        return $matched;
185    }

187    /**
188     * Returns the symbols a pattern could be about from where an attempt stands.
189     *
190     * @param NodePattern $pattern What the pattern requires of the symbol
191     * @param MatchState  $state   How far the attempt has got
192     *
193     * @example An attempt that has arrived somewhere is only about that symbol
194     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
195     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
196     *     count($matching->candidates(new \App\Gql\Syntax\Pattern\NodePattern('p'), $state)) // => 1
197     *
198     * @return list<NodeDatum> The symbols to consider
199     */
200    public function candidates(NodePattern $pattern, MatchState $state): array
201    {
202        if ($state->current !== null) {
203            return [$state->current];
204        }
205        if ($pattern->variable !== null && $state->row->has($pattern->variable)) {
206            $bound = $state->row->value($pattern->variable);
207
208            return $bound instanceof NodeDatum ? [$bound] : [];
209        }
210
211        return $this->graph->nodes();
212    }

214    /**
215     * Returns every way a parenthesised stretch of pattern matches, repeated.
216     *
217     * Each repetition carries on from where the last one arrived, which is what makes
218     * a repeated group a chain of the shape it describes rather than a set of
219     * unrelated matches of it.
220     *
221     * A name a repeated group binds is a group variable: each repetition binds it
222     * afresh, and once the group is matched it is bound to the list of what every
223     * repetition bound, in the order they were matched. `((a)-[]->(b)){2}` binds `a`
224     * and `b` to two symbols each, not to one symbol that has to recur. A group that
225     * is only parenthesised, with no quantifier, repeats nothing and binds as the
226     * pattern inside it does.
227     *
228     * @param GroupPattern $group The stretch of pattern
229     * @param MatchState   $state How far the attempt has got
230     * @param PathMode     $mode  What the path may visit more than once
231     *
232     * @example A group over a graph with no symbols matches only by not repeating
233     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
234     *     $group = new \App\Gql\Syntax\Pattern\GroupPattern([new \App\Gql\Syntax\Pattern\NodePattern()], new \App\Gql\Syntax\Pattern\Quantifier(0, 2));
235     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
236     *     count($matching->matchGroup($group, $state, \App\Gql\Syntax\Pattern\PathMode::Trail)) // => 1
237     *
238     * @return list<MatchState> The attempts that matched it
239     *
240     * @throws GqlException If a requirement written in the pattern cannot be worked out
241     */
242    public function matchGroup(GroupPattern $group, MatchState $state, PathMode $mode): array
243    {
244        $quantifier = $group->quantifier;
245        if ($quantifier === null) {
246            return $this->matchTerms($group->terms, 0, $state, $mode);
247        }
248
249        $variables = array_values(array_unique(PatternVariables::inTerms($group->terms)));
250        $start = $state;
251        foreach ($variables as $name) {
252            $start = $start->bind($name, new ListDatum([]));
253        }
254        $reached = $quantifier->allows(0) ? [$start] : [];
255        $frontier = [$start];
256        $ceiling = $quantifier->most ?? PHP_INT_MAX;
257
258        for ($times = 1; $times <= $ceiling && $frontier !== []; ++$times) {
259            $next = [];
260            foreach ($frontier as $standing) {
261                array_push($next, ...$this->repeat($group, $variables, $standing, $mode));
262            }
263            $frontier = $next;
264            if ($quantifier->allows($times)) {
265                array_push($reached, ...$frontier);
266            }
267        }
268
269        return $reached;
270    }

272    /**
273     * Returns every way one more repetition of a group matches, with its names added to their lists.
274     *
275     * @param GroupPattern $group     The group
276     * @param list<string> $variables The names it binds, each bound to the list of what earlier repetitions bound
277     * @param MatchState   $standing  How far the attempt has got
278     * @param PathMode     $mode      What the path may visit more than once
279     *
280     * @example A repetition adds what it bound to the list the name is bound to
281     *     $node = new \App\Gql\Datum\NodeDatum('a');
282     *     $matching = new \App\Gql\Matching\PathMatching(new \App\Gql\Element\ElementGraph(['a' => $node], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
283     *     $group = new \App\Gql\Syntax\Pattern\GroupPattern([new \App\Gql\Syntax\Pattern\NodePattern('n')], \App\Gql\Syntax\Pattern\Quantifier::exactly(1));
284     *     $standing = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit()->with('n', new \App\Gql\Datum\ListDatum([])));
285     *     $matching->repeat($group, ['n'], $standing, \App\Gql\Syntax\Pattern\PathMode::Walk)[0]->row->value('n')->toText() // => '[a]'
286     *
287     * @return list<MatchState> The attempts that matched it once more
288     *
289     * @throws GqlException If a requirement written in the pattern cannot be worked out
290     */
291    public function repeat(GroupPattern $group, array $variables, MatchState $standing, PathMode $mode): array
292    {
293        $fresh = $standing->withRow(new BindingRow(array_diff_key($standing->row->values(), array_flip($variables))));
294        $repeated = [];
295        foreach ($this->matchTerms($group->terms, 0, $fresh, $mode) as $after) {
296            $row = $after->row;
297            foreach ($variables as $name) {
298                $row = $row->with($name, self::collected($standing->row->value($name), $after->row->value($name)));
299            }
300            $repeated[] = $after->withRow($row);
301        }
302
303        return $repeated;
304    }

306    /**
307     * Returns a group variable's list with what one more repetition bound it to added.
308     *
309     * A repeated relation inside a repeated group binds a list of its own on every
310     * repetition, and that list is added element by element, so that a group variable
311     * is always one list of elements in the order the path crossed them.
312     *
313     * @param Datum $collected What earlier repetitions bound, as a list
314     * @param Datum $bound     What this repetition bound
315     *
316     * @example A repetition adds one element to the list
317     *     \App\Gql\Matching\PathMatching::collected(new \App\Gql\Datum\ListDatum([new \App\Gql\Datum\NodeDatum('a')]), new \App\Gql\Datum\NodeDatum('b'))->toText() // => '[a, b]'
318     * @example A list a repetition bound is added element by element
319     *     \App\Gql\Matching\PathMatching::collected(new \App\Gql\Datum\ListDatum([]), new \App\Gql\Datum\ListDatum([new \App\Gql\Datum\NodeDatum('a')]))->toText() // => '[a]'
320     *
321     * @return ListDatum The list
322     */
323    public static function collected(Datum $collected, Datum $bound): ListDatum
324    {
325        $items = $collected instanceof ListDatum ? $collected->items : [];
326
327        return new ListDatum([...$items, ...($bound instanceof ListDatum ? $bound->items : [$bound])]);
328    }
