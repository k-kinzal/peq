7/**
8 * What makes two values the same value, for grouping and for removing duplicates.
9 *
10 * This is deliberately not comparison. `DISTINCT` and `GROUP BY` have to put two
11 * nulls in the same group, while `NULL = NULL` is unknown; a query that grouped by a
12 * property some rows do not carry would otherwise produce one group per missing
13 * value, which is the opposite of what grouping is for.
14 *
15 * Sameness is decided by writing a value out in a form that is unique to it, and
16 * comparing those. That makes a group key a string, which is what a table of groups
17 * wants to be keyed by anyway.
18 *
19 * @visibility App\Gql
20 */
21final class DatumIdentity
22{
23    /**
24     * Writes a value out in a form that only equal values share.
25     *
26     * The kind is part of the form, so that a number and the string spelling it do
27     * not collapse into one group. Whole and approximate numbers do share a form when
28     * they stand for the same number, because grouping by a count should not split
29     * into two groups on how the count happened to be computed.
30     *
31     * @param Datum $value The value to identify
32     *
33     * @example Two ways of writing the same number identify the same value
34     *     \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\IntegerDatum(2)) === \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\FloatDatum(2.0)) // => true
35     * @example A number and the string spelling it do not
36     *     \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\IntegerDatum(2)) === \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\StringDatum('2')) // => false
37     * @example Two absent values identify the same value, which grouping needs
38     *     \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\NullDatum()) === \App\Gql\Datum\DatumIdentity::key(new \App\Gql\Datum\NullDatum()) // => true
39     *
40     * @return string A form unique to values equal to this one
41     */
42    public static function key(Datum $value): string
43    {
44        if ($value->kind()->numeric()) {
45            return 'number:'.self::number($value);
46        }
47        if ($value instanceof ListDatum) {
48            return 'list:['.implode(',', array_map(self::key(...), $value->items)).']';
49        }
50        if ($value instanceof PathDatum) {
51            return 'path:['.implode(',', array_map(self::key(...), $value->elements)).']';
52        }
53        if ($value instanceof NodeDatum || $value instanceof EdgeDatum) {
54            return 'element:'.$value->kind()->typeName().':'.$value->id;
55        }
56
57        return strtolower($value->kind()->typeName()).':'.$value->toText();
58    }

60    /**
61     * Writes a number in a form that does not depend on how it was computed.
62     *
63     * @param Datum $value The value, which must be of a numeric kind
64     *
65     * @example A number that came out approximate identifies the whole number it equals
66     *     \App\Gql\Datum\DatumIdentity::number(new \App\Gql\Datum\FloatDatum(2.0)) // => '2'
67     * @example One with a fractional part keeps it
68     *     \App\Gql\Datum\DatumIdentity::number(new \App\Gql\Datum\FloatDatum(2.5)) // => '2.5'
69     * @example A result that is not a number groups with the other results that are not
70     *     \App\Gql\Datum\DatumIdentity::number(new \App\Gql\Datum\FloatDatum(NAN)) // => 'NAN'
71     *
72     * @return string The number, in a form equal numbers share
73     */
74    public static function number(Datum $value): string
75    {
76        if ($value instanceof DecimalDatum) {
77            return self::plain($value->toText());
78        }
79        $number = DatumOrder::numberOf($value);
80        if (is_int($number)) {
81            return (string) $number;
82        }
83        if (is_nan($number)) {
84            return 'NAN';
85        }
86        if (is_infinite($number)) {
87            return $number > 0 ? 'INF' : '-INF';
88        }
89
90        return self::plain(var_export($number, true));
91    }

93    /**
94     * Writes a number without an exponent and without trailing zeros.
95     *
96     * A float is keyed by the shortest decimal that reads back as the same float —
97     * what PHP writes for it — and an exact number by its digits, so a float and an
98     * exact number that `=` finds equal are grouped together, and `0.1 + 0.2` as a
99     * float is not grouped with `0.3`, which `=` finds different.
100     *
101     * @param string $written The number, as `1.5E-7` or `2.50` is written
102     *
103     * @example An exponent is written out
104     *     \App\Gql\Datum\DatumIdentity::plain('1.5E-7') // => '0.00000015'
105     * @example Trailing zeros and a bare point are left out
106     *     \App\Gql\Datum\DatumIdentity::plain('3.0') // => '3'
107     * @example Negative zero is zero
108     *     \App\Gql\Datum\DatumIdentity::plain('-0.0') // => '0'
109     *
110     * @return string The number, written plainly
111     */
112    public static function plain(string $written): string
113    {
114        $sign = str_starts_with($written, '-') ? '-' : '';
115        $parts = explode('E', strtoupper(ltrim($written, '-+')));
116        $mantissa = $parts[0];
117        $exponent = $parts[1] ?? '0';
118        [$whole, $fraction] = explode('.', $mantissa.'.');
119        $digits = $whole.$fraction;
120        $point = strlen($whole) + (int) $exponent;
121        if ($point <= 0) {
122            $digits = str_repeat('0', 1 - $point).$digits;
123            $point = 1;
124        }
125        $digits = str_pad($digits, $point, '0');
126        $wholePart = ltrim(substr($digits, 0, $point), '0');
127        $fractionPart = rtrim(substr($digits, $point), '0');
128        $plain = ($wholePart === '' ? '0' : $wholePart).($fractionPart === '' ? '' : '.'.$fractionPart);
129
130        return $plain === '0' ? '0' : $sign.$plain;
131    }

133    /**
134     * Writes several values out as one form, for grouping by more than one thing.
135     *
136     * @param list<Datum> $values The values that together decide the group
137     *
138     * @example Grouping by nothing puts every row in one group
139     *     \App\Gql\Datum\DatumIdentity::keyOfAll([]) // => ''
140     *
141     * @return string A form unique to rows that group together
142     */
143    public static function keyOfAll(array $values): string
144    {
145        return implode("\0", array_map(self::key(...), $values));
146    }
