11/**
12 * An exact number with digits after the point.
13 *
14 * GQL keeps two kinds of number apart, as SQL does. An exact number is exactly what was
15 * written: `1.5` is one and a half, and `0.1 + 0.2` is `0.3`. An approximate number is
16 * the nearest one a binary fraction can hold, which is what `1.5e0`, `1.5f` and `1.5d`
17 * ask for. A literal with a point and no exponent is exact, so it is not a float.
18 *
19 * The value is held as a whole number and a scale — `1.5` is 15 with a scale of 1 — so
20 * that nothing is ever rounded to fit a binary fraction. The whole number is a PHP
21 * integer, which bounds the digits a value may have at eighteen; a value that would
22 * need more is out of range rather than silently approximated.
23 */
24final readonly class DecimalDatum implements Datum
25{

29     */

31        public int $unscaled,
32        public int $scale,
33    ) {
34        assert($this->scale >= 0, 'An exact number has no fewer than no digits after its point');
35    }

37    /**
38     * Reads an exact number as a query writes it.
39     *
40     * A number with no digits after the point, once any exponent is applied, is a whole
41     * number: `15M` and `1.5e1M` are both fifteen, and both are integers.
42     *
43     * @param string $written The literal, with or without its exponent and its `M`
44     *
45     * @example A point makes a number exact rather than approximate
46     *     \App\Gql\Datum\DecimalDatum::written('1.5')->toText() // => '1.5'
47     * @example An exponent moves the point
48     *     \App\Gql\Datum\DecimalDatum::written('1.5e-1M')->toText() // => '0.15'
49     * @example A number with nothing after the point is a whole number
50     *     \App\Gql\Datum\DecimalDatum::written('1.5e1M') instanceof \App\Gql\Datum\IntegerDatum // => true
51     *
52     * @return DecimalDatum|IntegerDatum The number
53     *
54     * @throws GqlException If it has more digits than an exact number can hold
55     */
56    public static function written(string $written): IntegerDatum|self
57    {
58        $plain = rtrim($written, 'mM');
59        $exponent = 0;
60        $marked = stripos($plain, 'e');
61        if ($marked !== false) {
62            $exponent = (int) substr($plain, $marked + 1);
63            $plain = substr($plain, 0, $marked);
64        }
65        $point = strpos($plain, '.');
66        $digits = str_replace('.', '', $plain);
67        $scale = ($point === false ? 0 : strlen($plain) - $point - 1) - $exponent;
68        if ($scale < 0) {
69            $digits .= str_repeat('0', -$scale);
70            $scale = 0;
71        }
72
73        return self::of(self::whole($digits), $scale);
74    }

76    /**
77     * Returns an exact number, as an integer when it has nothing after its point.
78     *
79     * @param int $unscaled The digits, as one whole number
80     * @param int $scale    How many of them come after the point
81     *
82     * @example A scale of nothing is an integer
83     *     \App\Gql\Datum\DecimalDatum::of(15, 0) instanceof \App\Gql\Datum\IntegerDatum // => true
84     *
85     * @return DecimalDatum|IntegerDatum The number
86     */
87    public static function of(int $unscaled, int $scale): IntegerDatum|self
88    {
89        return $scale === 0 ? new IntegerDatum($unscaled) : new self($unscaled, $scale);
90    }

92    /**
93     * Reads a run of digits as a whole number, refusing one too long to hold.
94     *
95     * @param string $digits The digits, with no sign
96     *
97     * @example Digits read as the number they spell
98     *     \App\Gql\Datum\DecimalDatum::whole('0042') // => 42
99     * @example One too long for an exact number is out of range
100     *     \App\Gql\Datum\DecimalDatum::whole('99999999999999999999') // throws \App\Gql\GqlException: numeric value out of range
101     *
102     * @return int The number
103     *
104     * @throws GqlException If it is too long to hold
105     */
106    public static function whole(string $digits): int
107    {
108        $trimmed = ltrim($digits, '0');
109        $limit = (string) PHP_INT_MAX;
110        if (strlen($trimmed) > strlen($limit) || (strlen($trimmed) === strlen($limit) && strcmp($trimmed, $limit) > 0)) {
111            throw GqlException::because(
112                StatusCode::NumericValueOutOfRange,
113                sprintf('%s has more digits than an exact number can hold', $digits),
114            );
115        }
116
117        return (int) $trimmed;
118    }

123     * @example A decimal is one
124     *     (new \App\Gql\Datum\DecimalDatum(15, 1))->kind() // => \App\Gql\Datum\DatumKind::Decimal

132    /**
133     * Writes the number with exactly as many digits after its point as its scale.
134     *
135     * @example A number is written as exactly what it is
136     *     (new \App\Gql\Datum\DecimalDatum(-5, 2))->toText() // => '-0.05'
137     */
138    #[Override]
139    public function toText(): string
140    {
141        $digits = str_pad(ltrim((string) $this->unscaled, '-'), $this->scale + 1, '0', STR_PAD_LEFT);
142        $point = strlen($digits) - $this->scale;
143
144        return ($this->unscaled < 0 ? '-' : '').substr($digits, 0, $point).'.'.substr($digits, $point);
145    }

147    /**
148     * Returns the nearest approximate number, for arithmetic that is approximate anyway.
149     *
150     * @example A decimal read as a float is its nearest binary fraction
151     *     (new \App\Gql\Datum\DecimalDatum(15, 1))->toFloat() // => 1.5
152     *
153     * @return float The number
154     */
155    public function toFloat(): float
156    {
157        return $this->unscaled / 10 ** $this->scale;
158    }
