11/**
12 * Which functions a query can call, and what calling one does.
13 *
14 * Names are matched without regard to case, as GQL matches every name a grammar
15 * reserves, so `COUNT` and `count` and `Coalesce` are the functions a reader expects
16 * them to be. A name that belongs to no function is reported as an unsupported
17 * feature rather than as a missing value, because a query calling a function that
18 * does not exist is a query that will never mean what it says.
19 *
20 * `trim` is the one name that does two things, and which one it does depends on how
21 * many arguments it is given: one string is trimmed of its whitespace, and a list
22 * with a length is cut to that length. That is GQL's own arrangement.
23 *
24 * @visibility App\Gql
25 */
26final class FunctionCatalog
27{
28    /**
29     * Calls a function.
30     *
31     * @param string      $name      The function name, as the query wrote it
32     * @param list<Datum> $arguments What it is applied to, in order
33     *
34     * @example A function is called by the name a query asks for it by
35     *     \App\Gql\Invocation\FunctionCatalog::call('upper', [new \App\Gql\Datum\StringDatum('a')])->toText() // => 'A'
36     * @example A name that belongs to nothing says so
37     *     \App\Gql\Invocation\FunctionCatalog::call('sqrt', []) // throws \App\Gql\GqlException: there is no function called
38     * @example So does a call given the wrong number of arguments
39     *     \App\Gql\Invocation\FunctionCatalog::call('upper', []) // throws \App\Gql\GqlException: syntax error
40     *
41     * @return Datum What the function produced
42     *
43     * @throws GqlException If no function goes by that name, or it cannot take these arguments
44     */
45    public static function call(string $name, array $arguments): Datum
46    {
47        $given = count($arguments);
48        $called = strtolower($name);
49
50        return match ($called) {
51            'char_length', 'upper', 'lower', 'trim' => self::text($called, $arguments),
52            'size' => ListFunctions::size(self::argument($name, $arguments, 0, 1)),
53            'elements' => GraphFunctions::elements(self::argument($name, $arguments, 0, 1)),
54            'path_length' => GraphFunctions::pathLength(self::argument($name, $arguments, 0, 1)),
55            'left' => TextFunctions::left(self::argument($name, $arguments, 0, 2), self::argument($name, $arguments, 1, 2)),
56            'right' => TextFunctions::right(self::argument($name, $arguments, 0, 2), self::argument($name, $arguments, 1, 2)),
57            'nullif' => GeneralFunctions::nullif(self::argument($name, $arguments, 0, 2), self::argument($name, $arguments, 1, 2)),
58            'coalesce' => self::atLeastOne($name, $arguments),
59            'zoned_datetime' => self::moment($name, $arguments, $given),
60            default => throw GqlException::because(
61                StatusCode::UnknownFeature,
62                sprintf('there is no function called "%s"', $name),
63            ),
64        };
65    }

67    /**
68     * Calls one of the functions that take a string, or the list form of trim.
69     *
70     * @param string      $called    The function name, in lower case
71     * @param list<Datum> $arguments What it is applied to, in order
72     *
73     * @example Trimming a string takes its whitespace off
74     *     \App\Gql\Invocation\FunctionCatalog::text('trim', [new \App\Gql\Datum\StringDatum(' a ')])->toText() // => 'a'
75     * @example Trimming a list with a length cuts it to that length
76     *     $rows = new \App\Gql\Datum\ListDatum([new \App\Gql\Datum\IntegerDatum(1), new \App\Gql\Datum\IntegerDatum(2)]);
77     *     \App\Gql\Invocation\FunctionCatalog::text('trim', [$rows, new \App\Gql\Datum\IntegerDatum(1)])->toText() // => '[1]'
78     *
79     * @return Datum What the function produced
80     *
81     * @throws GqlException If the call cannot take these arguments
82     */
83    public static function text(string $called, array $arguments): Datum
84    {
85        if ($called === 'trim' && count($arguments) === 2) {
86            return ListFunctions::trimTo($arguments[0], $arguments[1]);
87        }
88
89        $value = self::argument($called, $arguments, 0, 1);
90
91        return match ($called) {
92            'char_length' => TextFunctions::charLength($value),
93            'upper' => TextFunctions::upper($value),
94            'lower' => TextFunctions::lower($value),
95            default => TextFunctions::trim($value),
96        };
97    }

99    /**
100     * Returns one argument of a call, having checked how many it was given.
101     *
102     * @param string      $name      The function name, as the query wrote it
103     * @param list<Datum> $arguments What it was applied to
104     * @param int         $index     Which argument to return, counting from zero
105     * @param int         $expected  How many it must have been given
106     *
107     * @example An argument is returned when the count is right
108     *     \App\Gql\Invocation\FunctionCatalog::argument('upper', [new \App\Gql\Datum\StringDatum('a')], 0, 1)->toText() // => 'a'
109     * @example A call given the wrong number of arguments says so
110     *     \App\Gql\Invocation\FunctionCatalog::argument('upper', [], 0, 1) // throws \App\Gql\GqlException: syntax error
111     *
112     * @return Datum The argument
113     *
114     * @throws GqlException If the call was given the wrong number of arguments
115     */
116    public static function argument(string $name, array $arguments, int $index, int $expected): Datum
117    {
118        if (count($arguments) !== $expected) {
119            throw GqlException::because(
120                StatusCode::SyntaxError,
121                sprintf('%s takes %d argument(s), and was given %d', $name, $expected, count($arguments)),
122            );
123        }
124
125        return $arguments[$index];
126    }

128    /**
129     * Calls a function that takes any number of arguments, but at least one.
130     *
131     * @param string      $name      The function name, as the query wrote it
132     * @param list<Datum> $arguments What it was applied to
133     *
134     * @example The first value that is there is the answer
135     *     $values = [new \App\Gql\Datum\NullDatum(), new \App\Gql\Datum\StringDatum('b')];
136     *     \App\Gql\Invocation\FunctionCatalog::atLeastOne('coalesce', $values)->toText() // => 'b'
137     * @example A call given nothing to choose between says so
138     *     \App\Gql\Invocation\FunctionCatalog::atLeastOne('coalesce', []) // throws \App\Gql\GqlException: syntax error
139     *
140     * @return Datum What the function produced
141     *
142     * @throws GqlException If the call was given nothing
143     */
144    public static function atLeastOne(string $name, array $arguments): Datum
145    {
146        if ($arguments === []) {
147            throw GqlException::because(
148                StatusCode::SyntaxError,
149                sprintf('%s takes at least one argument, and was given none', $name),
150            );
151        }
152
153        return GeneralFunctions::coalesce($arguments);
154    }

156    /**
157     * Calls the function that reads a moment in time.
158     *
159     * @param string      $name      The function name, as the query wrote it
160     * @param list<Datum> $arguments What it was applied to
161     * @param int         $given     How many arguments it was given
162     *
163     * @example Asked for nothing, it answers with now
164     *     \App\Gql\Invocation\FunctionCatalog::moment('zoned_datetime', [], 0)->kind() // => \App\Gql\Datum\DatumKind::DateTime
165     * @example Asked for too much, it says so
166     *     $twice = [new \App\Gql\Datum\StringDatum('a'), new \App\Gql\Datum\StringDatum('b')];
167     *     \App\Gql\Invocation\FunctionCatalog::moment('zoned_datetime', $twice, 2) // throws \App\Gql\GqlException: syntax error
168     *
169     * @return Datum The moment
170     *
171     * @throws GqlException If the call was given more than one argument, or one that is not a moment
172     */
173    public static function moment(string $name, array $arguments, int $given): Datum
174    {
175        if ($given > 1) {
176            throw GqlException::because(
177                StatusCode::SyntaxError,
178                sprintf('%s takes at most one argument, and was given %d', $name, $given),
179            );
180        }
181
182        return GeneralFunctions::zonedDatetime($arguments);
183    }
