9 *
10 * A relation is drawn as an arrow from one symbol to another, named by what it is.
11 * Both ends are identities rather than symbols, so the drawing can be built up in any
12 * order and can leave out an arrow whose far end it never reached.
13 */
14final readonly class DiagramEdge
15{
16    /**
17     * @param string $origin What identifies the symbol the arrow leaves
18     * @param string $target What identifies the symbol it arrives at
19     * @param string $label  What the relation is
20     */
21    public function __construct(
22        public string $origin,
23        public string $target,
24        public string $label = '',
25    ) {}
26
27    /**
28     * Returns what tells this relation apart from every other in the drawing.
29     *
30     * Two symbols may be related more than once — a method that both calls another
31     * and takes its type as a parameter — and a drawing should show both arrows. What
32     * it should not show is the same arrow twice, which happens whenever a walk
33     * reaches the same pair from two directions.
34     *
35     * @example A relation is told apart by its two ends and what it is
36     *     (new \App\Reporter\Diagram\DiagramEdge('a', 'b', 'calls'))->signature() // => 'a|calls|b'
37     *
38     * @return string What identifies the relation in the drawing
39     */
40    public function signature(): string
41    {
42        return $this->origin.'|'.$this->label.'|'.$this->target;
43    }
