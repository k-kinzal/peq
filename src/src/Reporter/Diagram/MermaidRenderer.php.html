10/**
11 * Writes a drawing as a Mermaid flowchart, for a page that renders one.
12 *
13 * GitHub, GitLab and most documentation sites draw a `mermaid` code block as a
14 * picture, which makes this the format for putting a piece of the graph in a pull
15 * request or a design note. Symbols are numbered in the order they were drawn, as
16 * `n1`, `n2` and so on, because a symbol's name is not something Mermaid accepts as an
17 * identifier; the name is the label instead. Every relation is an arrow labelled with
18 * what it is, and the flowchart runs left to right, as the terminal drawing does.
19 *
20 * @visibility App\Reporter
21 */
22final class MermaidRenderer implements DiagramRenderer
23{
24    /**
25     * Writes the flowchart, or nothing when the drawing holds nothing.
26     *
27     * @param Diagram         $diagram The symbols and relations to write
28     * @param OutputInterface $output  Where the flowchart is written
29     */
30    #[Override]
31    public function render(Diagram $diagram, OutputInterface $output): void
32    {
33        foreach ($this->lines($diagram) as $line) {
34            $output->writeln($line, OutputInterface::OUTPUT_RAW);
35        }
36    }
37
38    /**
39     * Returns the flowchart, one string to a line.
40     *
41     * @param Diagram $diagram The symbols and relations to write
42     *
43     * @example A relation is an arrow between numbered symbols, labelled with what it is
44     *     $diagram = new \App\Reporter\Diagram\Diagram();
45     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('App\\A'));
46     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('App\\B'));
47     *     $diagram->relate(new \App\Reporter\Diagram\DiagramEdge('App\\A', 'App\\B', 'calls'));
48     *     (new \App\Reporter\Diagram\MermaidRenderer())->lines($diagram) // => ['flowchart LR', '    n1["App\\A"]', '    n2["App\\B"]', '    n1 -->|"calls"| n2']
49     * @example A drawing that holds nothing is no flowchart
50     *     (new \App\Reporter\Diagram\MermaidRenderer())->lines(new \App\Reporter\Diagram\Diagram()) // => []
51     *
52     * @return list<string> The lines
53     */
54    public function lines(Diagram $diagram): array
55    {
56        if ($diagram->empty()) {
57            return [];
58        }
59
60        $lines = ['flowchart LR'];
61        foreach ($diagram->nodes() as $node) {
62            $lines[] = sprintf('    n%d["%s"]', $diagram->numberOf($node->id) ?? 0, self::escape($node->id));
63        }
64        foreach ($diagram->nodes() as $node) {
65            foreach ($diagram->leaving($node->id) as $edge) {
66                $lines[] = $edge->label === ''
67                    ? sprintf('    n%d --> n%d', $diagram->numberOf($edge->origin) ?? 0, $diagram->numberOf($edge->target) ?? 0)
68                    : sprintf(
69                        '    n%d -->|"%s"| n%d',
70                        $diagram->numberOf($edge->origin) ?? 0,
71                        self::escape($edge->label),
72                        $diagram->numberOf($edge->target) ?? 0,
73                    );
74            }
75        }
76
77        return $lines;
78    }
79
80    /**
81     * Writes text so that Mermaid reads it as text, whatever it holds.
82     *
83     * Mermaid ends a quoted label at a double quote and reads angle brackets as HTML,
84     * so those are written as the entity codes Mermaid defines, and so is the `#` that
85     * begins one.
86     *
87     * @param string $text The text
88     *
89     * @example A double quote is written as an entity
90     *     \App\Reporter\Diagram\MermaidRenderer::escape('say "hi"') // => 'say #quot;hi#quot;'
91     * @example A name needs nothing done to it
92     *     \App\Reporter\Diagram\MermaidRenderer::escape('App\\Http\\Controller::show') // => 'App\\Http\\Controller::show'
93     *
94     * @return string The text as Mermaid is to read it
95     */
96    public static function escape(string $text): string
97    {
98        return strtr($text, ['#' => '#35;', '"' => '#quot;', '<' => '#lt;', '>' => '#gt;']);
99    }
100}
