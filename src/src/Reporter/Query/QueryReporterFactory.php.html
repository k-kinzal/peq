14/**
15 * Chooses and assembles the reporter a configuration asks a query for.
16 *
17 * The formats are the same closed set the inspection uses, because they answer the
18 * same question about the reader rather than about the command: who is going to read
19 * this, and in what. A format added without a reporter here is a static analysis
20 * error rather than a surprise at the moment someone asks for it.
21 *
22 * What differs is what each format means for a table rather than for a walk. A table
23 * is written as a table; a drawing is of the piece of graph the answer holds; and a
24 * tree is of the paths the query chose to bind, which is nothing unless it bound some.
25 */
26final class QueryReporterFactory
27{
28    /**
29     * Builds the reporter for a configuration.
30     *
31     * The two formats that draw the answer are given the graph it came from, so that
32     * they can show the relations between the symbols a query found rather than only
33     * the ones it happened to bind. The other three have no use for it.
34     *
35     * @param Config            $config The application configuration
36     * @param null|ElementGraph $graph  The graph the answer came from, or null when there is none
37     *
38     * @example A configuration that asks for a table is given one
39     *     $config = \App\Config\Config::fromArray(['basePath' => '.', 'direction' => 'uses', 'type' => 'native', 'output' => 'table']);
40     *     (new \App\Reporter\Query\QueryReporterFactory())->create($config) instanceof \App\Reporter\Query\TableWriter // => true
41     * @example One that asks for a drawing is given that instead
42     *     $config = \App\Config\Config::fromArray(['basePath' => '.', 'direction' => 'uses', 'type' => 'native', 'output' => 'graph']);
43     *     (new \App\Reporter\Query\QueryReporterFactory())->create($config) instanceof \App\Reporter\Query\DiagramWriter // => true
44     *
45     * @return QueryReporter A reporter writing the answer in the format the configuration asks for
46     */
47    public function create(Config $config, ?ElementGraph $graph = null): QueryReporter
48    {
49        return match ($config->output) {
50            OutputFormat::Table => new TableWriter(),
51            OutputFormat::Json => new JsonWriter(),
52            OutputFormat::Graph => new DiagramWriter($graph, new TerminalRenderer((new Terminal())->getWidth())),
53            OutputFormat::Mermaid => new DiagramWriter($graph, new MermaidRenderer()),
54            OutputFormat::Dot => new DotWriter($graph),
55            OutputFormat::Tree => new TreeWriter(),
56        };
57    }
58}
