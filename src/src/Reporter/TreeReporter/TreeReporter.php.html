14/**

16 *
17 * This reporter visualizes the dependency graph the way the Unix `tree` command
18 * visualizes a directory: one line per node, with box-drawing characters showing
19 * which branches continue below. It owns the traversal it was given and the line
20 * format, and delegates the bookkeeping a tree needs — how deep the walk is, which
21 * node is the last of its siblings, which nodes were already expanded — to a
22 * cursor created for the duration of one report.
23 */
24final class TreeReporter implements Reporter
25{
26    /**
27     * @param TreeReporterOptions $options      How much of the tree to print
28     * @param Traversal           $traversal    The strategy deciding which relations the tree follows
29     * @param LineRenderer        $lineRenderer The format of a single line
30     */
31    public function __construct(
32        private readonly TreeReporterOptions $options,
33        private readonly Traversal $traversal,
34        private readonly LineRenderer $lineRenderer = new LineRenderer(),
35    ) {}

37    /**
38     * Writes the dependency tree rooted at the given symbol.
39     *
40     * @param Graph           $graph  The dependency graph to report on
41     * @param NodeId<Node>    $symbol The symbol the tree is rooted at
42     * @param OutputInterface $output Where the tree is written
43     */
44    public function report(Graph $graph, NodeId $symbol, OutputInterface $output): void
45    {
