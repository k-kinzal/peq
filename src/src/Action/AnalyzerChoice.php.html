14/**
15 * Turning the configured analyzer kind into the analyzer itself.
16 *
17 * Two use cases now build a graph — inspecting a symbol and querying the graph — and
18 * both have to make exactly the same decision from exactly the same settings. Making
19 * it twice would be two places to forget a new analyzer in, and the second of them
20 * would be found by a reader wondering why one command supports something the other
21 * does not.
22 *
23 * Each arm is handed only what its analyzer needs — the files to read, and the PHP
24 * version to read them as — so both commands read the same sources the same way. An
25 * analyzer is told its parameters rather than given the application's configuration to
26 * read, which is what keeps the analyzers independent of how peq happens to be
27 * configured.
28 */
29final class AnalyzerChoice
30{
31    /**
32     * Returns the analyzer a configuration asks for.
33     *
34     * @param Config $config The merged application configuration
35     *
36     * @example The kind a configuration names is the analyzer it gets
37     *     $config = \App\Config\Config::fromArray(['basePath' => '.', 'direction' => 'uses', 'type' => 'native']);
38     *     \App\Action\AnalyzerChoice::forConfig($config) instanceof \App\Analyzer\NativeAnalyzer\NativeAnalyzer // => true
39     * @example So is the one that builds a graph without reading any sources
40     *     $config = \App\Config\Config::fromArray(['basePath' => '.', 'direction' => 'uses', 'type' => 'debug']);
41     *     \App\Action\AnalyzerChoice::forConfig($config) instanceof \App\Analyzer\DebugAnalyzer\DebugAnalyzer // => true
42     *
43     * @return Analyzer The analyzer
44     */
45    public static function forConfig(Config $config): Analyzer
46    {
47        return match ($config->analyzer) {
48            AnalyzerKind::PhpStan => new PhpStanAnalyzer(
49                includes: $config->includes,
50                excludes: $config->excludes,
51                phpVersion: $config->phpVersion?->id,
52            ),
53            AnalyzerKind::Native => new NativeAnalyzer(
54                includes: $config->includes,
55                excludes: $config->excludes,
56                phpVersion: $config->phpVersion?->id,
57            ),
58            AnalyzerKind::Debug => new DebugAnalyzer(
59                seed: $config->debug->seed,
60                depth: $config->debug->depth,
61            ),
62        };
63    }
64}
