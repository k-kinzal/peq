9/**
10 * Raised when the symbol asked about is not in the analyzed graph.
11 *
12 * The name may be misspelled, may live outside the analyzed path, or may sit in a
13 * file the include and exclude patterns filtered out. None of those are programmer
14 * errors, so this is a checked exception: a caller has to decide what to tell the
15 * user rather than letting it escape.
16 */
17final class SymbolNotFoundException extends Exception
18{
19    /**
20     * Builds the exception for a symbol name that resolved to nothing.
21     *
22     * The PHP version the sources were read as is named when it is known, because a
23     * file written for another version is one the analysis cannot read at all: its
24     * symbols are absent from the graph rather than merely unreachable in it, and
25     * nothing else peq prints would say so.
26     *
27     * @param string      $target     The symbol name that was asked about
28     * @param null|string $phpVersion The PHP version the sources were read as, if one was settled
29     *
30     * @return self An exception naming the symbol and why it may be absent
31     */
32    public static function forTarget(string $target, ?string $phpVersion = null): self
33    {
34        if ($phpVersion === null) {
35            return new self(sprintf(
36                'Symbol "%s" is not in the dependency graph. Check the spelling, the analyzed path, and the include and exclude patterns.',
37                $target,
38            ));
39        }
40
41        return new self(sprintf(
42            'Symbol "%s" is not in the dependency graph, which was read as PHP %s. Check the spelling, the analyzed path, the include and exclude patterns, and the PHP version the sources are written for.',
43            $target,
44            $phpVersion,
45        ));
46    }
47}
