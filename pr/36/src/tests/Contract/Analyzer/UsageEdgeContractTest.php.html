563    private function assertEdgeExists(
564        Graph $graph,
565        string $fromSuffix,
566        string $toSuffix,
567        EdgeKind $kind,
568        string $msg,
569    ): void {
570        foreach ($graph->nodes() as $node) {
571            if (!str_ends_with($node->id()->toString(), $fromSuffix)) {
572                continue;
573            }
574            foreach ($graph->edges($node->id()) as $edge) {
575                if ($edge->kind() === $kind && str_ends_with($edge->to()->toString(), $toSuffix)) {
576                    $this->addToAssertionCount(1);
577
578                    return;
579                }
580            }
581        }
582        self::fail($msg."\nNodes: ".implode(', ', array_map(fn ($n) => $n->id()->toString(), $graph->nodes())));
583    }

597                if ($edge->kind() === $kind && str_ends_with($edge->to()->toString(), $toSuffix)) {
598                    self::fail($msg);
599                }
600            }
601        }
