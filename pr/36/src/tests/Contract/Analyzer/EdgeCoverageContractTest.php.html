240    public function testEveryMappedEdgeKindIsCoveredOrGapped(): void
241    {
242        $allMappedKinds = [];
243        foreach (self::NODE_TO_EDGE_KINDS as $edgeKinds) {
244            foreach ($edgeKinds as $kind) {
245                $allMappedKinds[$kind->value] = true;
246            }
247        }
248
249        /** @var array<string, string> $knownGaps */
250        $knownGaps = self::KNOWN_GAPS;
251        foreach (array_keys($allMappedKinds) as $kindValue) {
252            $isTested = isset(self::TESTED_EDGE_KINDS[$kindValue]);
253            $isGap = isset($knownGaps[$kindValue]);
254            $isStructural = in_array($kindValue, self::STRUCTURAL_EDGE_KINDS, true);
255
256            self::assertTrue(
257                $isTested || $isGap || $isStructural,
258                sprintf(
259                    'EdgeKind "%s" is mapped from a DEPENDENCY_PRODUCING node but is '
260                    .'not in TESTED_EDGE_KINDS, KNOWN_GAPS, or STRUCTURAL_EDGE_KINDS',
261                    $kindValue,
262                ),
263            );
