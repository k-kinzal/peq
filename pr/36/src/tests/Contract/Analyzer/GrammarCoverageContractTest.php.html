375    public function testAllPhpParserNodeTypesAreClassified(): void
376    {
377        $discovered = self::discoverPhpParserNodeTypes();
378        $dependencyProducing = array_keys(self::DEPENDENCY_PRODUCING);
379        $notDependencyProducing = self::NOT_DEPENDENCY_PRODUCING;
380        $classified = array_merge($dependencyProducing, $notDependencyProducing);
381
382        $unclassified = array_diff($discovered, $classified);
383        $extra = array_diff($classified, $discovered);
384
385        self::assertSame(
386            [],
387            array_values($unclassified),
388            'The following PhpParser node types are not classified in either DEPENDENCY_PRODUCING or NOT_DEPENDENCY_PRODUCING: '
389            .implode(', ', $unclassified),
390        );
391
392        self::assertSame(
393            [],
394            array_values($extra),
395            'The following classified types were not discovered as concrete PhpParser node types: '
396            .implode(', ', $extra),
397        );
398    }

437    /**
438     * @return list<class-string<Node>>
439     */
440    private static function discoverPhpParserNodeTypes(): array
441    {
442        $fileName = (new \ReflectionClass(Expr::class))->getFileName();
443        assert($fileName !== false);
444        $nodeDir = dirname($fileName);
445        $iterator = new \RecursiveIteratorIterator(
446            new \RecursiveDirectoryIterator($nodeDir, \FilesystemIterator::SKIP_DOTS)
447        );
448        $types = [];
449
450        /** @var \SplFileInfo $file */
451        foreach ($iterator as $file) {
452            if ($file->getExtension() !== 'php') {
453                continue;
454            }
455            $relative = str_replace('/', '\\', substr($file->getPathname(), strlen($nodeDir) + 1, -4));
456            $fqcn = 'PhpParser\Node\\'.$relative;
457            if (!class_exists($fqcn)) {
458                continue;
459            }
460            $rc = new \ReflectionClass($fqcn);
461            if (!$rc->isAbstract() && !$rc->isInterface() && $rc->implementsInterface(Node::class)) {
462                /** @var class-string<Node> $fqcn */
463                $types[] = $fqcn;
464            }
465        }
466        sort($types);
467
468        return $types;
469    }
