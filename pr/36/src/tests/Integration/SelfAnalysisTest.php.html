219    {

49        self::assertSame([], array_values(array_map($spell, array_filter($edges, static fn (Edge $edge): bool => $graph->node($edge->from()) === null || $graph->node($edge->to()) === null))), 'Relations pointing at a symbol the graph does not hold');
