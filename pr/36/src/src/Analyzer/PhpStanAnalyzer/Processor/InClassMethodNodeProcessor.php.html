106    public static function dispatch(PhpParserNode $node, Scope $scope, Node $source): array
107    {
157        $ast = self::getParsedAst($scope->getFile());
158        if ($ast === null) {
159            return null;
160        }
161
162        $methodNode = $node->getOriginalNode();
163        $nodeFinder = self::$nodeFinder ??= new NodeFinder();
164        $classReflection = $scope->getClassReflection();
165        if ($classReflection === null) {
166            return null;
167        }
168
169        $className = $classReflection->getName();
170        $classNode = $nodeFinder->findFirst($ast, function (PhpParserNode $n) use ($className) {
171            if (($n instanceof Class_ || $n instanceof Interface_ || $n instanceof Trait_ || $n instanceof Enum_)
172                && isset($n->namespacedName)) {
173                return $n->namespacedName->toString() === $className;
174            }
175
176            return false;
177        });
178
179        if ($classNode instanceof Class_ || $classNode instanceof Trait_ || $classNode instanceof Enum_) {
180            $methodName = $methodNode->name->toString();
181            $foundMethod = $nodeFinder->findFirst($classNode->stmts, function (PhpParserNode $n) use ($methodName) {
182                return $n instanceof ClassMethod && $n->name->toString() === $methodName;
183            });
184
185            if ($foundMethod instanceof ClassMethod) {
186                return $foundMethod->stmts;
187            }
188        }
189
190        return null;

109            $node instanceof PhpParserNode\Expr\ClassConstFetch => ConstFetchProcessor::process($node, $scope, $source),
110            $node instanceof PhpParserNode\Expr\New_ => InstantiationProcessor::process($node, $scope, $source),
111            $node instanceof PhpParserNode\Expr\StaticCall => StaticCallProcessor::process($node, $scope, $source),
112            $node instanceof PhpParserNode\Stmt\Catch_ => CatchProcessor::process($node, $scope, $source),
113            $node instanceof PhpParserNode\Expr\Instanceof_ => InstanceofProcessor::process($node, $scope, $source),
114            $node instanceof PhpParserNode\Expr\FuncCall => FunctionCallProcessor::process($node, $scope, $source),
115            $node instanceof PhpParserNode\Expr\NullsafeMethodCall => MethodCallProcessor::process($node, $scope, $source),
116            $node instanceof PhpParserNode\Expr\MethodCall => MethodCallProcessor::process($node, $scope, $source),
117            $node instanceof PhpParserNode\Expr\NullsafePropertyFetch => PropertyAccessProcessor::process($node, $scope, $source),
118            $node instanceof PhpParserNode\Expr\PropertyFetch => PropertyAccessProcessor::process($node, $scope, $source),
119            $node instanceof PhpParserNode\Expr\StaticPropertyFetch => StaticPropertyAccessProcessor::process($node, $scope, $source),
120            default => [],
121        };
122    }
