17/**
18 * Records the cases an enum declares.
19 *
20 * An enum case is a symbol of its own — code names it directly — so it becomes a node
21 * related to the enum that declares it, along with any attributes written on it.
22 *
23 * @visibility App\Analyzer\PhpStanAnalyzer
24 */
25final class EnumCaseProcessor
26{
27    /**
28     * Records the case and the enum that declares it.
29     *
30     * @param EnumCase $node  The syntax node met during analysis
31     * @param Scope    $scope The analyser scope it was written in
32     *
33     * @return list<AttributeEdge|EnumCaseEdge|EnumCaseNode> The relations it describes
34     */
35    public static function process(EnumCase $node, Scope $scope): array
36    {
37        $classReflection = $scope->getClassReflection();
38        if ($classReflection === null) {
39            return [];
40        }
41
42        $className = $classReflection->getName();
43        $meta = new FileMeta($scope->getFile(), $node->getStartLine(), 1);
44        $declared = new EnumCaseNode(EnumCaseNodeId::of($className, $node->name->toString()), true, $meta);
45
46        return [
47            $declared,
48            new EnumCaseEdge(new EnumNode(EnumNodeId::of($className), true, null), $declared, $meta),
49            ...AttributeProcessor::process($node->attrGroups, $declared, $scope),
50        ];
51    }
52}
