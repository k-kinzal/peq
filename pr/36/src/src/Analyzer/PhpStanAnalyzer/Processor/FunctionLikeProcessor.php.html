36    {
37        $items = [];
38        $meta = new FileMeta($scope->getFile(), $node->getStartLine(), 1);
39
40        $sourceNode = null;
41
42        if ($node instanceof ClassMethod) {
43            $classReflection = $scope->getClassReflection();
44            if ($classReflection === null) {
45                return [];
46            }
47            $className = $classReflection->getName();
48            $methodName = $node->name->toString();
49            $sourceNode = new MethodNode(
50                new MethodNodeId(self::getNamespace($className), self::getShortName($className), $methodName),
51                true,
52                $meta
53            );
54
55            $namespace = self::getNamespace($className);
56            $shortName = self::getShortName($className);
57            $parentSourceNode = null;
58
59            if ($classReflection->isInterface()) {
60                $parentSourceNode = new GraphInterfaceNode(new InterfaceNodeId($namespace, $shortName), true, null);
61            } elseif ($classReflection->isTrait()) {
62                $parentSourceNode = new TraitNode(new TraitNodeId($namespace, $shortName), true, null);
63            } elseif ($classReflection->isEnum()) {
64                $parentSourceNode = new EnumNode(new EnumNodeId($namespace, $shortName), true, null);
65            } else {
66                $parentSourceNode = new ClassNode(new ClassNodeId($namespace, $shortName), true, null);
67            }
68
69            $items[] = new DeclarationMethodEdge($parentSourceNode, $sourceNode, $meta);
70        } elseif ($node instanceof Function_) {
71            if (!isset($node->namespacedName)) {
72                return [];
73            }
74            $functionName = $node->namespacedName->toString();
75            $sourceNode = new FunctionNode(
76                new FunctionNodeId(self::getNamespace($functionName), self::getShortName($functionName)),
77                true,
78                $meta
79            );
80        }
81
82        if ($sourceNode === null) {
83            return [];
84        }
85
86        $items[] = $sourceNode;
87
88        // Attributes on Function/Method
89        foreach ($node->attrGroups as $attrGroup) {
90            foreach ($attrGroup->attrs as $attr) {
91                $attrName = $scope->resolveName($attr->name);
92                $attrNode = new ClassNode(new ClassNodeId(self::getNamespace($attrName), self::getShortName($attrName)), false, null);
93                $items[] = new AttributeEdge($sourceNode, $attrNode, $meta);
94            }
95        }
96
97        // Return Type
98        $returnType = $node->getReturnType();
99        foreach (TypeResolver::resolveNames($returnType) as $typeName) {
100            $name = $typeName->toString();
101            if (!self::isBuiltin($name)) {
102                $typeNode = new ClassNode(new ClassNodeId(self::getNamespace($name), self::getShortName($name)), false, null);
103                $items[] = new DeclarationTypeReturnEdge($sourceNode, $typeNode, new FileMeta($scope->getFile(), $typeName->getStartLine(), 1));
104            }
105        }
106
107        // Parameters
108        foreach ($node->getParams() as $param) {
109            // Attributes on Parameter
110            foreach ($param->attrGroups as $attrGroup) {
111                foreach ($attrGroup->attrs as $attr) {
112                    $attrName = $scope->resolveName($attr->name);
113                    $attrNode = new ClassNode(new ClassNodeId(self::getNamespace($attrName), self::getShortName($attrName)), false, null);
114                    // Attribute on Parameter
115                    $items[] = new AttributeEdge($sourceNode, $attrNode, new FileMeta($scope->getFile(), $param->getStartLine(), 1));
116                }
117            }
118
119            foreach (TypeResolver::resolveNames($param->type) as $typeName) {
120                $name = $typeName->toString();
121                if (!self::isBuiltin($name)) {
122                    $typeNode = new ClassNode(new ClassNodeId(self::getNamespace($name), self::getShortName($name)), false, null);
123                    $items[] = new DeclarationTypeParameterEdge($sourceNode, $typeNode, new FileMeta($scope->getFile(), $typeName->getStartLine(), 1));
124                }
125            }
126        }
127
128        return $items;
129    }
