48/**
49 * Faker provider for generating complete dependency graphs.
50 *
51 * This provider creates fully-formed Graph objects with randomly generated
52 * nodes and edges using a recursive, hierarchical approach that mimics
53 * real PHP code structure.
54 *
55 * @property GraphGenerator $generator
56 */
57final class GraphProvider extends Base
58{
59    /**
60     * Generates a complete dependency graph with nodes and edges.
61     *
62     * Creates a graph starting from a random root node type (Class, Interface, Trait, or Function)
63     * and recursively generates related nodes and edges to form a complete dependency structure.
64     *
65     * @param int $depth Maximum recursion depth (default: 5)
66     *
67     * @return Graph A complete dependency graph with nodes and edges
68     */
69    public function graph(int $depth = 5): Graph
70    {
71        $rootKinds = [NodeKind::Klass, NodeKind::Interface, NodeKind::Trait, NodeKind::Function];
72
73        /** @var NodeKind $rootKind */
74        $rootKind = $this->generator->randomElement($rootKinds);
75
76        return match ($rootKind) {
77            NodeKind::Klass => $this->classGraph(null, $depth - 1),
78            NodeKind::Interface => $this->interfaceGraph(null, $depth - 1),
79            NodeKind::Trait => $this->traitGraph(null, $depth - 1),
80            NodeKind::Function => $this->functionGraph(null, $depth - 1),
81            default => throw new \LogicException('Unsupported root node kind.'),
82        };
83    }

85    public function typeGraph(
86        BuiltinNodeId|ClassNodeId|EnumNodeId|InterfaceNodeId|null $nodeId = null,
87        int $depth = 5
88    ): Graph {
89        if ($nodeId === null) {
90            /** @var BuiltinNodeId|ClassNodeId|EnumNodeId|InterfaceNodeId $nodeId */
91            $nodeId = $this->generator->randomElement([
92                $this->generator->classNodeId(),
93                $this->generator->interfaceNodeId(),
94                $this->generator->enumNodeId(),
95                $this->generator->builtinNodeId(),
96            ]);
97        }
98
99        return match (true) {
100            $nodeId instanceof ClassNodeId => $this->classGraph($nodeId, $depth),
101            $nodeId instanceof InterfaceNodeId => $this->interfaceGraph($nodeId, $depth),
102            $nodeId instanceof EnumNodeId => $this->enumGraph($nodeId, $depth),
103            $nodeId instanceof BuiltinNodeId => $this->builtinGraph($nodeId),
104        };
105    }

107    /**
108     * Generates a class graph with all its members and relationships.
109     *
110     * @param null|ClassNodeId $symbol Optional class NodeId to use as root
111     * @param int              $depth  Maximum recursion depth (default: 20)
112     *
113     * @return Graph the class graph with methods, properties, constants, inheritance, etc
114     */
115    private function classGraph(?ClassNodeId $symbol = null, int $depth = 5): Graph
116    {
117        $graph = new Graph();
118
119        $classId = $symbol ?? $this->generator->classNodeId();
120        $class = new ClassNode(
121            id: $classId,
122            resolved: $this->generator->boolean(),
123            meta: $this->generator->fileMeta()
124        );
125        $graph->addNode($class);
126
127        if ($depth === 0) {
128            return $graph;
129        }
130
131        $methodIds = $this->generator->array(fn () => $this->generator->methodNodeId(), 0, 5);
132        foreach ($methodIds as $methodId) {
133            $methodGraph = $this->methodGraph($methodId, $depth - 1);
134            $graph = $graph->merge($methodGraph);
135            $method = $graph->node($methodId);
136            assert($method instanceof MethodNode);
137            $graph->addEdge(new DeclarationMethodEdge($class, $method, $this->generator->fileMeta()));
138        }
139
140        $propertyIds = $this->generator->array(fn () => $this->generator->propertyNodeId(), 0, 5);
141        foreach ($propertyIds as $propertyId) {
142            $propertyGraph = $this->propertyGraph($propertyId, $depth - 1);
143            $graph = $graph->merge($propertyGraph);
144            $property = $graph->node($propertyId);
145            assert($property instanceof PropertyNode);
146            $graph->addEdge(new DeclarationPropertyEdge($class, $property, $this->generator->fileMeta()));
147        }
148
149        $constantIds = $this->generator->array(fn () => $this->generator->constantNodeId(), 0, 5);
150        foreach ($constantIds as $constantId) {
151            $constantGraph = $this->constantGraph($constantId);
152            $graph = $graph->merge($constantGraph);
153            $constant = $graph->node($constantId);
154            assert($constant instanceof ConstantNode);
155            $graph->addEdge(new DeclarationConstantEdge($class, $constant, $this->generator->fileMeta()));
156        }
157
158        /** @var null|ClassNodeId $parentId */
159        // @phpstan-ignore-next-line
160        $parentId = $this->generator->optional()->classNodeId();
161        if ($parentId !== null) {
162            $parentGraph = $this->classGraph($parentId, $depth - 1);
163            $graph = $graph->merge($parentGraph);
164            $parent = $graph->node($parentId);
165            assert($parent instanceof ClassNode);
166            $graph->addEdge(new DeclarationExtendsEdge($class, $parent, $this->generator->fileMeta()));
167        }
168
169        $interfaceIds = $this->generator->array(fn () => $this->generator->interfaceNodeId(), 0, 5);
170        foreach ($interfaceIds as $interfaceId) {
171            $interfaceGraph = $this->interfaceGraph($interfaceId, $depth - 1);
172            $graph = $graph->merge($interfaceGraph);
173            $interface = $graph->node($interfaceId);
174            assert($interface instanceof GraphInterfaceNode);
175            $graph->addEdge(new DeclarationImplementsEdge($class, $interface, $this->generator->fileMeta()));
176        }
177
178        $traitIds = $this->generator->array(fn () => $this->generator->traitNodeId(), 0, 5);
179        foreach ($traitIds as $traitId) {
180            $traitGraph = $this->traitGraph($traitId, $depth - 1);
181            $graph = $graph->merge($traitGraph);
182            $trait = $graph->node($traitId);
183            assert($trait instanceof TraitNode);
184            $graph->addEdge(new DeclarationTraitUseEdge($class, $trait, $this->generator->fileMeta()));
185        }
186
187        return $graph;
188    }
