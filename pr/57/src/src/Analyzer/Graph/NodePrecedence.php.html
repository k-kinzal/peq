5namespace App\Analyzer\Graph;

7/**
8 * Decides which of two descriptions of the same symbol the graph keeps.
9 *
10 * One symbol is met many times during an analysis, and the meetings are not equally
11 * informative. A class is met where it is declared — with its file, its keywords, its
12 * attributes — and met again as the owner of each of its methods, as the type of a
13 * parameter, as the target of a `new`. Those later meetings know the name and nothing
14 * more, and they arrive in whatever order the collectors happen to report them.
15 *
16 * Keeping the first description would therefore make what the graph knows about a
17 * symbol depend on the order its files were read, which is not a property a graph
18 * should have. Keeping the most informative one instead makes the result the same
19 * whichever way the analysis walked, and it is what lets a declaration recorded by
20 * one processor survive a bare reference recorded by another.
21 */
22final class NodePrecedence
23{
24    /**
25     * How much a description says about the symbol, in facts a reference cannot know.
26     *
27     * The weights are ordered rather than counted: knowing what kind of symbol it is
28     * outranks having been resolved, which outranks having read its declaration,
29     * which outranks knowing where it is written. That order is the order in which
30     * the facts stop being guesses — a placeholder knows none of them, and each later
31     * one is only available to something that actually read the source.
32     *
33     * @param Node $node The description to weigh
34     *
35     * @example A symbol only ever referred to says the least about itself
36     *     $named = \App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Domain\\Invoice');
37     *     \App\Analyzer\Graph\NodePrecedence::describes(\App\Analyzer\Graph\Node\UnknownNode::standingInFor($named)) // => 0
38     * @example A symbol read where it is declared says the most
39     *     $read = new \App\Analyzer\Graph\Node\ClassNode(
40     *         \App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Domain\\Invoice'),
41     *         true,
42     *         new \App\Analyzer\Graph\FileMeta('/project/src/Invoice.php', 12, 1),
43     *         new \App\Analyzer\Graph\Declaration\SymbolDeclaration(
44     *             modifiers: new \App\Analyzer\Graph\Declaration\Modifiers(final: true),
45     *         ),
46     *     );
47     *     \App\Analyzer\Graph\NodePrecedence::describes($read) // => 15
48     *
49     * @return int The weight of the description, zero for one that says nothing
50     */
51    public static function describes(Node $node): int
52    {
53        $declared = $node->declaration();
54
55        return ($node->kind() === NodeKind::Unknown ? 0 : 8)
56            + ($node->resolved() ? 4 : 0)
57            + ($declared !== null && !$declared->empty() ? 2 : 0)
58            + ($node->meta() !== null ? 1 : 0);
59    }
60
61    /**
62     * Reports whether a newly met description replaces the recorded one.
63     *
64     * Equal descriptions leave the recorded one in place. Two meetings that say the
65     * same amount say the same thing, and preferring the later one would reintroduce
66     * the order dependence this class exists to remove.
67     *
68     * @param Node $recorded The description the graph already holds
69     * @param Node $offered  The description analysis has just met
70     *
71     * @example A declaration replaces the placeholder an edge left behind
72     *     $named = \App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Domain\\Invoice');
73     *     $placeholder = \App\Analyzer\Graph\Node\UnknownNode::standingInFor($named);
74     *     \App\Analyzer\Graph\NodePrecedence::prefers($placeholder, new \App\Analyzer\Graph\Node\ClassNode($named, true)) // => true
75     * @example A bare reference does not replace a declaration
76     *     $named = \App\Analyzer\Graph\NodeId\ClassNodeId::of('App\\Domain\\Invoice');
77     *     $declared = new \App\Analyzer\Graph\Node\ClassNode($named, true);
78     *     \App\Analyzer\Graph\NodePrecedence::prefers($declared, new \App\Analyzer\Graph\Node\ClassNode($named, false)) // => false
79     *
80     * @return bool True when the offered description says strictly more
81     */
82    public static function prefers(Node $recorded, Node $offered): bool
83    {
84        return self::describes($offered) > self::describes($recorded);
85    }
86}
