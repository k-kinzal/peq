22/**
23 * Records the properties a class or trait declares.
24 *
25 * One `public $a, $b;` statement declares several properties, each of which becomes a
26 * node of its own, related to the class-like that declares it and to the type it is
27 * declared as. The attributes and the type are written once for the statement, so
28 * each declared property carries the same ones.
29 *
30 * @visibility App\Analyzer\PhpStanAnalyzer
31 */
32final class PropertyProcessor
33{
34    /**
35     * Records every property one statement declares.
36     *
37     * @param Property $node  The syntax node met during analysis
38     * @param Scope    $scope The analyser scope it was written in
39     *
40     * @return list<AttributeEdge|PropertyEdge|PropertyNode|TypePropertyEdge> The relations it describes
41     */
42    public static function process(Property $node, Scope $scope): array
43    {
44        $classReflection = $scope->getClassReflection();
45        if ($classReflection === null) {
46            return [];
47        }
48
49        $className = $classReflection->getName();
50        $owner = $classReflection->isTrait()
51            ? new TraitNode(TraitNodeId::of($className), true, null)
52            : new ClassNode(ClassNodeId::of($className), true, null);
53
54        $declaration = DeclarationReader::forProperty($node, AttributeProcessor::usages($node->attrGroups, $scope));
55
56        $items = [];
57        foreach ($node->props as $property) {
58            $meta = new FileMeta($scope->getFile(), $property->getStartLine(), 1);
59            $declared = new PropertyNode(PropertyNodeId::of($className, $property->name->toString()), true, $meta, $declaration);
60
61            $items[] = $declared;
62            $items[] = new PropertyEdge($owner, $declared, $meta);
63            array_push($items, ...AttributeProcessor::process($node->attrGroups, $declared, $scope));
64
65            foreach (TypeResolver::references($node->type, $scope->getFile()) as $type) {
66                $items[] = new TypePropertyEdge($declared, $type->node, $type->meta);
67            }
68        }
69
70        return $items;
71    }
72}
