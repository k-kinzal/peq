11/**
12 * What a path mode forbids a match from doing twice.
13 *
14 * A dependency graph is full of cycles — mutual recursion, a class whose method
15 * returns its own type, a pair of services that call each other — and without a rule
16 * about repetition a pattern over one of them has infinitely many matches. These
17 * rules are what make a search end.
18 *
19 * A path that names no mode is a walk, and a walk forbids nothing — which is why GQL
20 * lets a walk repeat only a bounded number of times. `TRAIL`, which forbids crossing
21 * the same relation twice, is the weakest rule that makes an unbounded repetition
22 * finite, so it rules out as little as possible of what a reader might have meant.
23 *
24 * @visibility App\Gql
25 */
26final class PathModeRule
27{
28    /**
29     * Reports whether a match may cross a relation.
30     *
31     * @param PathMode   $mode  What the path may visit more than once
32     * @param MatchState $state How far the attempt has got
33     * @param EdgeDatum  $edge  The relation about to be crossed
34     *
35     * @example Crossing anything twice is allowed where nothing is forbidden
36     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
37     *     \App\Gql\Matching\PathModeRule::allowsEdge(\App\Gql\Syntax\Pattern\PathMode::Walk, $state, new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b')) // => true
38     * @example A relation already crossed is refused by the mode that forbids it
39     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'))->across(new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('b'));
40     *     \App\Gql\Matching\PathModeRule::allowsEdge(\App\Gql\Syntax\Pattern\PathMode::Trail, $state, new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b')) // => false
41     *
42     * @return bool True when the mode allows it
43     */
44    public static function allowsEdge(PathMode $mode, MatchState $state, EdgeDatum $edge): bool
45    {
46        return match ($mode) {
47            PathMode::Walk => true,
48            PathMode::Trail,
49            PathMode::Simple,
50            PathMode::Acyclic => !$state->crossed($edge->id),
51        };
52    }
53
54    /**
55     * Reports whether a match may arrive at a symbol.
56     *
57     * The difference between the two strictest modes is one case: a path that comes
58     * back to where it started. One allows it, because a cycle through a symbol is
59     * often exactly what a question about recursion is looking for; the other does
60     * not, because a question about layering is not.
61     *
62     * @param PathMode   $mode  What the path may visit more than once
63     * @param MatchState $state How far the attempt has got
64     * @param NodeDatum  $node  The symbol about to be arrived at
65     *
66     * @example Coming back to where the path started is allowed by one mode
67     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'))->across(new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('b'));
68     *     \App\Gql\Matching\PathModeRule::allowsNode(\App\Gql\Syntax\Pattern\PathMode::Simple, $state, new \App\Gql\Datum\NodeDatum('a')) // => true
69     * @example And forbidden by the other
70     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'))->across(new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('b'));
71     *     \App\Gql\Matching\PathModeRule::allowsNode(\App\Gql\Syntax\Pattern\PathMode::Acyclic, $state, new \App\Gql\Datum\NodeDatum('a')) // => false
72     *
73     * @return bool True when the mode allows it
74     */
75    public static function allowsNode(PathMode $mode, MatchState $state, NodeDatum $node): bool
76    {
77        return match ($mode) {
78            PathMode::Walk,
79            PathMode::Trail => true,
80            PathMode::Simple => !$state->met($node->id) || $state->startedAt($node->id),
81            PathMode::Acyclic => !$state->met($node->id),
82        };
83    }
84}
