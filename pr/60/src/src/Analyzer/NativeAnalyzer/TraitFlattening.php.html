14/**
15 * Which copy of a method a class taking traits on ends up with.
16 *
17 * A trait is not a symbol its methods belong to: the methods belong to every class
18 * that uses it, and the graph records them there. Reading a trait's statements once
19 * per using class is therefore the right thing to do — but only for the copies that
20 * survive PHP's own rules. A class that writes a method keeps its own; two traits
21 * offering the same method are settled by an `insteadof`; a method renamed by an `as`
22 * is taken on under the new name and only under that one; and a trait that demands a
23 * method rather than writing one is answered by whatever does write it, which may be
24 * another trait or a class further up.
25 *
26 * Reading a copy PHP discards would record a method the class does not have, at the
27 * line of a trait rather than of the class, which is why this is decided before a
28 * trait statement is read rather than after.
29 *
30 * @visibility namespace
31 */
32final class TraitFlattening
33{
34    /**
35     * Reads the renames one `use` statement applies to one trait.
36     *
37     * An adaptation naming no trait applies to every trait of the statement; one
38     * naming a trait applies only to that trait.
39     *
40     * @param TraitUse $use       The `use` statement met while walking
41     * @param string   $traitName Fully qualified name of the trait being read
42     *
43     * @return array<string, string> The new name of each renamed method, keyed by its lower-cased written name
44     */
45    public static function renames(TraitUse $use, string $traitName): array
46    {
47        $renames = [];
48        foreach ($use->adaptations as $adaptation) {
49            if (!$adaptation instanceof Alias || $adaptation->newName === null) {
50                continue;
51            }
52            if ($adaptation->trait !== null && strtolower($adaptation->trait->toString()) !== strtolower($traitName)) {
53                continue;
54            }
55            $renames[strtolower($adaptation->method->toString())] = $adaptation->newName->toString();
56        }
57
58        return $renames;
59    }

61    /**
62     * Reports whether a class takes a trait's copy of a method on, or discards it.
63     *
64     * @param ClassLike   $class      The class-like reading the trait
65     * @param ClassMethod $method     The method as the trait writes it
66     * @param string      $methodName The name it is taken on as, after any rename
67     * @param string      $traitName  Fully qualified name of the trait the copy is written in
68     * @param SourceIndex $index      What the analysed files declare
69     *
70     * @return bool True when this copy is the one the class ends up with
71     */
72    public static function keepsMethod(ClassLike $class, ClassMethod $method, string $methodName, string $traitName, SourceIndex $index): bool
73    {
74        if (self::writesMethod($class, $methodName)) {
75            return false;
76        }
77
78        $provider = self::providerOf($class, $methodName, $index);
79        if ($provider === null || strtolower($provider) !== strtolower($traitName)) {
80            return false;
81        }
82
83        return $method->stmts !== null || !self::inherits($class, $methodName, $index, []);
84    }

86    /**
87     * Names the trait a class ends up taking a method from.
88     *
89     * A trait that writes the method answers for it before one that only demands it,
90     * whatever order they are used in, because a demand is not an answer.
91     *
92     * @param ClassLike   $class      The class-like reading the traits
93     * @param string      $methodName The method name, after any rename
94     * @param SourceIndex $index      What the analysed files declare
95     *
96     * @return null|string The trait the method comes from, or null when no trait offers it
97     */
98    public static function providerOf(ClassLike $class, string $methodName, SourceIndex $index): ?string
99    {
100        $offers = self::offersOf($class, $methodName, $index);
101        foreach ($offers as $offered) {
102            if (!$offered->demanded) {
103                return $offered->declaringTrait;
104            }
105        }
106
107        return $offers === [] ? null : $offers[0]->declaringTrait;
108    }

110    /**
111     * Reads every offer of a method the traits a class uses make, in the order they are used.
112     *
113     * An `insteadof` settles which trait a method is taken from, so the offers of
114     * every other trait of that name are not offers at all.
115     *
116     * @param ClassLike   $class      The class-like reading the traits
117     * @param string      $methodName The method name, after any rename
118     * @param SourceIndex $index      What the analysed files declare
119     *
120     * @return list<TraitMethod> The offers that stand, in the order the traits are used
121     */
122    public static function offersOf(ClassLike $class, string $methodName, SourceIndex $index): array
123    {
124        $wanted = strtolower($methodName);
125        $settled = null;
126        $offers = [];
127
128        foreach ($class->getTraitUses() as $use) {
129            foreach ($use->adaptations as $adaptation) {
130                if ($adaptation instanceof Precedence && $adaptation->trait !== null && strtolower($adaptation->method->toString()) === $wanted) {
131                    $settled = strtolower($adaptation->trait->toString());
132                }
133            }
134            foreach ($use->traits as $trait) {
135                $through = $trait->toString();
136                $offered = self::methodsOf($through, self::renames($use, $through), $index, [])[$wanted] ?? null;
137                if ($offered !== null && ($settled === null || strtolower($through) === $settled)) {
138                    $offers[] = $offered;
139                }
140            }
141        }
142
143        return $offers;
144    }

159    /**
160     * Returns the method of that name a class-like writes, when it writes one.
161     *
162     * @param ClassLike $class      The class-like being read
163     * @param string    $methodName The method name, in any casing
164     *
165     * @return null|ClassMethod The method it writes, or null when it writes none
166     */
167    public static function methodWritten(ClassLike $class, string $methodName): ?ClassMethod
168    {
169        $wanted = strtolower($methodName);
170        foreach ($class->stmts as $statement) {
171            if ($statement instanceof ClassMethod && strtolower($statement->name->toString()) === $wanted) {
172                return $statement;
173            }
174        }
175
176        return null;
177    }

179    /**
180     * Reports whether a class has a written method of that name from further up.
181     *
182     * A trait that demands a method is answered by anything that writes one, and a
183     * class that inherits a written method has already answered. Only what the
184     * analysed files declare can be looked up; a class whose parent lives outside
185     * them has nothing to be read here, which is also all the reference engine can
186     * say about it.
187     *
188     * @param ClassLike    $class      The class-like being read
189     * @param string       $methodName The method name, after any rename
190     * @param SourceIndex  $index      What the analysed files declare
191     * @param list<string> $seen       The ancestors already read, which stops a cycle
192     *
193     * @return bool True when something above the class writes that method
194     */
195    public static function inherits(ClassLike $class, string $methodName, SourceIndex $index, array $seen): bool
196    {
197        $parentName = $class instanceof Class_ && $class->extends !== null ? $class->extends->toString() : null;
198        if ($parentName === null || in_array(strtolower($parentName), $seen, true)) {
199            return false;
200        }
201        $seen[] = strtolower($parentName);
202
203        $parent = $index->classLike($parentName);
204        if ($parent === null) {
205            return false;
206        }
207
208        $written = self::methodWritten($parent->node, $methodName);
209        if ($written !== null) {
210            return $written->stmts !== null;
211        }
212
213        $provider = self::providerOf($parent->node, $methodName, $index);
214        foreach (self::offersOf($parent->node, $methodName, $index) as $offered) {
215            if ($offered->declaringTrait === $provider && !$offered->demanded) {
216                return true;
217            }
218        }
219
220        return self::inherits($parent->node, $methodName, $index, $seen);
221    }

223    /**
224     * Reads the methods a trait offers, following the traits it uses in turn.
225     *
226     * @param string                $traitName Fully qualified name of the trait
227     * @param array<string, string> $renames   The renames the using statement applies to it
228     * @param SourceIndex           $index     What the analysed files declare
229     * @param list<string>          $reading   The traits already being read, which stops a cycle
230     *
231     * @return array<string, TraitMethod> What the trait offers, keyed by lower-cased method name
232     */
233    public static function methodsOf(string $traitName, array $renames, SourceIndex $index, array $reading): array
234    {
235        $declaration = $index->classLike($traitName);
236        if ($declaration === null || in_array(strtolower($traitName), $reading, true)) {
237            return [];
238        }
239        $reading[] = strtolower($traitName);
240
241        $methods = [];
242        foreach ($declaration->node->stmts as $statement) {
243            if ($statement instanceof ClassMethod) {
244                $written = strtolower($statement->name->toString());
245                $methods[strtolower($renames[$written] ?? $written)] = new TraitMethod($declaration->name, $statement->stmts === null);
246            }
247        }
248
249        foreach ($declaration->node->getTraitUses() as $use) {
250            foreach ($use->traits as $nested) {
251                $nestedName = $nested->toString();
252                foreach (self::methodsOf($nestedName, self::renames($use, $nestedName), $index, $reading) as $name => $offered) {
253                    $methods[$name] ??= $offered;
254                }
255            }
256        }
257
258        return $methods;
259    }
