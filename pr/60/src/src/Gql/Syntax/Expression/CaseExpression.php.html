9/**
10 * A choice between values, written the way SQL writes one.
11 *
12 * GQL has both forms, and they are one shape here. The searched form tests a
13 * predicate per branch; the simple form tests one subject against a value per branch.
14 * Holding the subject as something that may be absent is what makes them the same
15 * shape: with no subject, every branch is its own question.
16 *
17 * A `CASE` that matches nothing and was written without an `ELSE` is worth nothing,
18 * which is GQL's rule and the reason the fallback may be absent too.
19 */
20final readonly class CaseExpression implements Expression
21{
22    /**
23     * @param null|Expression  $subject   What each branch is tested against, or null when branches test themselves
24     * @param list<CaseBranch> $branches  The branches, in the order they are tried
25     * @param null|Expression  $otherwise What the expression is worth when no branch is taken, or null when nothing is
26     */
27    public function __construct(
28        public ?Expression $subject,
29        public array $branches,
30        public ?Expression $otherwise = null,
31    ) {
32        assert($this->branches !== [], 'A choice offers at least one branch');
33    }
34}
