12/**
13 * Reading a name, and telling apart the three things GQL means by one.
14 *
15 * The grammar writes three productions where a reader sees a word, and they accept
16 * different things:
17 *
18 * - a keyword — `upper` in `upper(x)` — is any word written plainly, reserved or not,
19 *   because which words may stand there is settled by the production and not by GQL's
20 *   reserved words;
21 * - an `<identifier>` — a label, a property, a field, an alias — is a word the standard
22 *   does not reserve, or any word at all in back quotes;
23 * - a `<regular identifier>` — which is all a binding variable may be — is a word the
24 *   standard does not reserve, and back quotes do not make one.
25 *
26 * Collapsing the three into "a name" is what lets `(:Function)` parse in an
27 * implementation that means to be GQL, and it is why they are separated here rather
28 * than left to each parser to remember.
29 *
30 * @visibility App\Gql
31 */
32final class NameReader
33{
34    /**
35     * Reports whether a reader stands at a word written plainly.
36     *
37     * @param TokenReader $tokens The pieces of the query being read
38     *
39     * @example A word is one whether or not GQL reserves it
40     *     \App\Gql\Parsing\NameReader::atWord(\App\Gql\Parsing\TokenReader::of('upper(x)')) // => true
41     * @example A name in back quotes is not a word
42     *     \App\Gql\Parsing\NameReader::atWord(\App\Gql\Parsing\TokenReader::of('`upper`(x)')) // => false
43     *
44     * @return bool True when it does
45     */
46    public static function atWord(TokenReader $tokens): bool
47    {
48        return $tokens->current()->kind === TokenKind::Name;
49    }
50
51    /**
52     * Reports whether a reader stands at a name GQL would let a query bind.
53     *
54     * This is what lets a pattern tell `(n:Method)` from `(:Method)` without a list of
55     * special cases: `WHERE` is reserved, so `(n WHERE ...)` never looked like two
56     * variables, and neither does any other keyword a later clause begins with.
57     *
58     * @param TokenReader $tokens The pieces of the query being read
59     *
60     * @example A plain name could be bound
61     *     \App\Gql\Parsing\NameReader::atVariable(\App\Gql\Parsing\TokenReader::of('p.firstName')) // => true
62     * @example A word GQL reserves could not
63     *     \App\Gql\Parsing\NameReader::atVariable(\App\Gql\Parsing\TokenReader::of('WHERE p.line > 1')) // => false
64     *
65     * @return bool True when it does
66     */
67    public static function atVariable(TokenReader $tokens): bool
68    {
69        $token = $tokens->current();
70
71        return $token->kind === TokenKind::Name && !ReservedWords::reserves($token->value);
72    }
73
74    /**
75     * Moves past an identifier: a label, a property, a field or an alias.
76     *
77     * A graph with a label called `Function` is matched by `` (:`Function`) ``, and
78     * `(:Function)` is a query that was written wrong — `FUNCTION` is a word the
79     * standard holds back for a later edition. GQL's `<delimited identifier>` may be
80     * written in double quotes as well as back quotes, so `(:"Function")` says the same;
81     * a double-quoted sequence is a string only where a value is written.
82     *
83     * @param TokenReader $tokens The pieces of the query being read
84     *
85     * @example A name GQL does not reserve is an identifier
86     *     \App\Gql\Parsing\NameReader::identifier(\App\Gql\Parsing\TokenReader::of('firstName')) // => 'firstName'
87     * @example A word GQL reserves is one when it is written in back quotes
88     *     \App\Gql\Parsing\NameReader::identifier(\App\Gql\Parsing\TokenReader::of('`value`')) // => 'value'
89     * @example Or in double quotes
90     *     \App\Gql\Parsing\NameReader::identifier(\App\Gql\Parsing\TokenReader::of('"value"')) // => 'value'
91     * @example And is not one when it is not
92     *     \App\Gql\Parsing\NameReader::identifier(\App\Gql\Parsing\TokenReader::of('value')) // throws \App\Gql\GqlException: reserves
93     *
94     * @return string The identifier, as it stands for itself
95     *
96     * @throws GqlException If the query does not have one here
97     */
98    public static function identifier(TokenReader $tokens): string
99    {
100        $token = $tokens->current();
101        if ($token->kind === TokenKind::Text && str_starts_with($token->lexeme, '"')) {
102            return $tokens->take()->value;
103        }
104        if (!$tokens->atName()) {
105            $tokens->fail('a name');
106        }
107        if ($token->kind === TokenKind::Name && ReservedWords::reserves($token->value)) {
108            self::refuseReserved($token, 'write it in back quotes to use it as a name');
109        }
110
111        return $tokens->take()->value;
112    }
113
114    /**
115     * Moves past the name a query binds something to.
116     *
117     * A binding variable is narrower than an identifier in both directions: a reserved
118     * word is not one, and neither is a name in back quotes. A variable is therefore
119     * always a word a reader can read.
120     *
121     * @param TokenReader $tokens The pieces of the query being read
122     *
123     * @example A name GQL does not reserve can be bound
124     *     \App\Gql\Parsing\NameReader::variable(\App\Gql\Parsing\TokenReader::of('person')) // => 'person'
125     * @example A word GQL reserves cannot, back quotes or not
126     *     \App\Gql\Parsing\NameReader::variable(\App\Gql\Parsing\TokenReader::of('`value`')) // throws \App\Gql\GqlException: syntax error
127     *
128     * @return string The variable's name
129     *
130     * @throws GqlException If the query does not have one here
131     */
132    public static function variable(TokenReader $tokens): string
133    {
134        $token = $tokens->current();
135        if ($token->kind === TokenKind::Name && ReservedWords::reserves($token->value)) {
136            self::refuseReserved($token, 'a query binds a name to a word GQL leaves free');
137        }
138        if (!self::atVariable($tokens)) {
139            $tokens->fail('a name');
140        }
141
142        return $tokens->take()->value;
143    }
144
145    /**
146     * Reports that a word GQL reserves was written where a name was wanted.
147     *
148     * @param Token  $token  The word, and where it stands
149     * @param string $remedy What the reader can write instead
150     *
151     * @example A reserved word is refused by name, and told what to write instead
152     *     $token = \App\Gql\Parsing\TokenReader::of('value')->current();
153     *     \App\Gql\Parsing\NameReader::refuseReserved($token, 'write it in back quotes') // throws \App\Gql\GqlException: GQL reserves "VALUE"
154     *
155     * @throws GqlException Always
156     */
157    public static function refuseReserved(Token $token, string $remedy): never
158    {
159        throw GqlException::syntax(
160            'GQL reserves "'.strtoupper($token->value).'", so it is not a name here: '.$remedy,
161            $token->line,
162            $token->column,
163            $token->describe(),
164        );
165    }
