30/**
31 * Generates the symbols a class-like declares: methods, functions and properties.
32 *
33 * A method and a function have the same signature shape — a return type and some
34 * parameter types — so that part is generated once for both. What differs is the
35 * body: a method may reach a method, a function, a property, a constant or a
36 * constructor call, while a function only calls functions, and a property has no
37 * body at all and only a declared type.
38 *
39 * @visibility parent
40 */
41final class MemberGraphGenerator
42{
43    /**
44     * @param NodeGenerator   $nodes  The source of the nodes these graphs hold
45     * @param NodeIdGenerator $ids    The source of the source locations relations carry
46     * @param RandomSource    $random The random source counts and optional relations are drawn from
47     */
48    public function __construct(
49        private readonly NodeGenerator $nodes,
50        private readonly NodeIdGenerator $ids,
51        private readonly RandomSource $random,
52    ) {}
53
54    /**
55     * Generates the graph of a method: its signature and what its body reaches.
56     *
57     * @param SymbolGraphGenerator $symbols The contract the referenced symbols are generated through
58     * @param null|MethodNodeId    $symbol  The method to generate, or null to draw one
59     * @param int                  $depth   How many further levels of symbols to generate
60     *
61     * @return GeneratedGraph<MethodNode> The method graph
62     */
63    public function methodGraph(SymbolGraphGenerator $symbols, ?MethodNodeId $symbol = null, int $depth = 5): GeneratedGraph
64    {
65        $result = GeneratedGraph::rootedAt($this->nodes->methodNode($symbol));
66        if ($depth === 0) {
67            return $result;
68        }
69
70        $result = $this->callableSignature($symbols, $result, $depth);
71
72        return $this->methodAccesses($symbols, $this->methodCalls($symbols, $result, $depth), $depth);
73    }

75    /**
76     * Generates the graph of a function: its signature and the functions it calls.
77     *
78     * @param SymbolGraphGenerator $symbols The contract the referenced symbols are generated through
79     * @param null|FunctionNodeId  $symbol  The function to generate, or null to draw one
80     * @param int                  $depth   How many further levels of symbols to generate
81     *
82     * @return GeneratedGraph<FunctionNode> The function graph
83     */
84    public function functionGraph(SymbolGraphGenerator $symbols, ?FunctionNodeId $symbol = null, int $depth = 5): GeneratedGraph
85    {
86        $result = GeneratedGraph::rootedAt($this->nodes->functionNode($symbol));
87        if ($depth === 0) {
88            return $result;
89        }
90
91        $result = $this->callableSignature($symbols, $result, $depth);
92        if ($this->random->boolean()) {
93            $result = $result->relatedTo(
94                $symbols->functionGraph(null, $depth - 1),
95                fn (FunctionNode $caller, FunctionNode $called): Edge => new FunctionCallEdge($caller, $called, $this->ids->fileMeta()),
96            );
97        }
98
99        return $result;
100    }

102    /**
103     * Generates the graph of a property: its declared type.
104     *
105     * @param SymbolGraphGenerator $symbols The contract the declared type is generated through
106     * @param null|PropertyNodeId  $symbol  The property to generate, or null to draw one
107     * @param int                  $depth   How many further levels of symbols to generate
108     *
109     * @return GeneratedGraph<PropertyNode> The property graph
110     */
111    public function propertyGraph(SymbolGraphGenerator $symbols, ?PropertyNodeId $symbol = null, int $depth = 5): GeneratedGraph
112    {
113        $result = GeneratedGraph::rootedAt($this->nodes->propertyNode($symbol));
114        if ($depth === 0) {
115            return $result;
116        }
117
118        return $result->relatedTo(
119            $symbols->typeGraph(null, $depth - 1),
120            fn (PropertyNode $property, BuiltinNode|ClassNode|EnumNode|GraphInterfaceNode $type): Edge => new TypePropertyEdge($property, $type, $this->ids->fileMeta()),
121        );
122    }

124    /**
125     * Adds the return type and the parameter types a callable declares.
126     *
127     * @template TCallable of FunctionNode|MethodNode
128     *
129     * @param SymbolGraphGenerator      $symbols The contract the declared types are generated through
130     * @param GeneratedGraph<TCallable> $result  The callable graph so far
131     * @param int                       $depth   How many further levels of symbols to generate
132     *
133     * @return GeneratedGraph<TCallable> The callable graph with its declared types
134     */
135    public function callableSignature(SymbolGraphGenerator $symbols, GeneratedGraph $result, int $depth): GeneratedGraph
136    {
137        $result = $result->relatedTo(
138            $symbols->typeGraph(null, $depth - 1),
139            fn (FunctionNode|MethodNode $callable, BuiltinNode|ClassNode|EnumNode|GraphInterfaceNode $type): Edge => new TypeReturnEdge($callable, $type, $this->ids->fileMeta()),
140        );
141
142        for ($remaining = $this->random->numberBetween(0, 5); $remaining > 0; --$remaining) {
143            $result = $result->relatedTo(
144                $symbols->typeGraph(null, $depth - 1),
145                fn (FunctionNode|MethodNode $callable, BuiltinNode|ClassNode|EnumNode|GraphInterfaceNode $type): Edge => new TypeParameterEdge($callable, $type, $this->ids->fileMeta()),
146            );
147        }
148
149        return $result;
150    }

152    /**
153     * Adds the methods and functions a method body calls.
154     *
155     * @param SymbolGraphGenerator       $symbols The contract the called symbols are generated through
156     * @param GeneratedGraph<MethodNode> $result  The method graph so far
157     * @param int                        $depth   How many further levels of symbols to generate
158     *
159     * @return GeneratedGraph<MethodNode> The method graph with the calls its body makes
160     */
161    public function methodCalls(SymbolGraphGenerator $symbols, GeneratedGraph $result, int $depth): GeneratedGraph
162    {
163        if ($this->random->boolean()) {
164            $result = $result->relatedTo(
165                $symbols->methodGraph(null, $depth - 1),
166                fn (MethodNode $caller, MethodNode $called): Edge => new MethodCallEdge($caller, $called, $this->ids->fileMeta()),
167            );
168        }
169        if ($this->random->boolean()) {
170            $result = $result->relatedTo(
171                $symbols->methodGraph(null, $depth - 1),
172                fn (MethodNode $caller, MethodNode $called): Edge => new StaticCallEdge($caller, $called, $this->ids->fileMeta()),
173            );
174        }
175        if ($this->random->boolean()) {
176            $result = $result->relatedTo(
177                $symbols->functionGraph(null, $depth - 1),
178                fn (MethodNode $caller, FunctionNode $called): Edge => new FunctionCallEdge($caller, $called, $this->ids->fileMeta()),
179            );
180        }
181
182        return $result;
183    }

185    /**
186     * Adds the properties, constants and constructors a method body reaches.
187     *
188     * @param SymbolGraphGenerator       $symbols The contract the reached symbols are generated through
189     * @param GeneratedGraph<MethodNode> $result  The method graph so far
190     * @param int                        $depth   How many further levels of symbols to generate
191     *
192     * @return GeneratedGraph<MethodNode> The method graph with what its body reaches
193     */
194    public function methodAccesses(SymbolGraphGenerator $symbols, GeneratedGraph $result, int $depth): GeneratedGraph
195    {
196        if ($this->random->boolean()) {
197            $result = $result->relatedTo(
198                $symbols->propertyGraph(null, $depth - 1),
199                fn (MethodNode $reader, PropertyNode $property): Edge => new PropertyAccessEdge($reader, $property, $this->ids->fileMeta()),
200            );
201        }
202        if ($this->random->boolean()) {
203            $result = $result->relatedTo(
204                $symbols->propertyGraph(null, $depth - 1),
205                fn (MethodNode $reader, PropertyNode $property): Edge => new StaticPropertyAccessEdge($reader, $property, $this->ids->fileMeta()),
206            );
207        }
208        if ($this->random->boolean()) {
209            $result = $result->relatedTo(
210                $symbols->constantGraph(),
211                fn (MethodNode $reader, ConstantNode $constant): Edge => new ConstFetchEdge($reader, $constant, $this->ids->fileMeta()),
212            );
213        }
214        if ($this->random->boolean()) {
215            $result = $result->relatedTo(
216                $symbols->classGraph(null, $depth - 1),
217                fn (MethodNode $caller, ClassNode $instantiated): Edge => new InstantiationEdge($caller, $instantiated, $this->ids->fileMeta()),
218            );
219        }
220
221        return $result;
222    }
