9/**
10 * Selects the PHP files an analysis covers.
11 *
12 * A path may name a file or a directory, and a directory is searched for PHP files
13 * subject to the project's include and exclude patterns. Every result is an absolute
14 * path, and a path that cannot be resolved on disk is left out rather than passed on
15 * — an analysis run should not fail on a symlink that was removed underneath it.
16 *
17 * @visibility namespace
18 */
19final class PhpFileCollector
20{
21    /**
22     * Collects the PHP files under the given paths.
23     *
24     * Patterns are matched against paths relative to the scanned directories, so
25     * `src` selects files below a `src/` directory and `vendor` excludes the
26     * `vendor/` directory together with everything in it.
27     *
28     * @param list<string> $paths    Files or directories to scan
29     * @param list<string> $includes Path patterns to include, or none to include everything
30     * @param list<string> $excludes Path patterns to exclude
31     *
32     * @return list<string> Absolute paths of the collected PHP files, without duplicates
33     */
34    public function collect(array $paths, array $includes = [], array $excludes = []): array
35    {
36        $collected = [];
37        $directories = [];
38
39        foreach ($paths as $path) {
40            if (is_dir($path)) {
41                $directories[] = $path;
42
43                continue;
44            }
45
46            $resolved = is_file($path) ? realpath($path) : false;
47            if ($resolved !== false) {
48                $collected[$resolved] = true;
49            }
50        }
51
52        if ($directories === []) {
53            return array_keys($collected);
54        }
55
56        $finder = (new Finder())->files()->in($directories)->name('*.php');
57        if ($includes !== []) {
58            $finder->path($includes);
59        }
60        if ($excludes !== []) {
61            $finder->notPath($excludes);
62        }
63
64        foreach ($finder as $file) {
65            $resolved = $file->getRealPath();
66            if ($resolved !== false) {
67                $collected[$resolved] = true;
68            }
69        }
70
71        return array_keys($collected);
72    }
73}
