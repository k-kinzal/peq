29/**
30 * Records a class-like declaration and what it is built from.
31 *
32 * A class, interface, trait or enum brings its own node into the graph together with
33 * what it extends, implements and uses, and the attributes written on it. These are
34 * the relations that make an inheritance chain walkable in both directions.
35 *
36 * @visibility App\Analyzer\PhpStanAnalyzer
37 */
38final class ClassLikeProcessor
39{
40    /**
41     * Records the declaration and everything it takes on.
42     *
43     * An anonymous class has no name to key a node by, so nothing is recorded for it.
44     *
45     * @param ClassLike $node  The syntax node met during analysis
46     * @param Scope     $scope The analyser scope it was written in
47     *
48     * @return list<AttributeEdge|ClassNode|EnumNode|ExtendsEdge|GraphInterfaceNode|ImplementsEdge|TraitNode|TraitUseEdge> The relations it describes
49     */
50    public static function process(ClassLike $node, Scope $scope): array
51    {
52        if ($node->namespacedName === null) {
53            return [];
54        }
55
56        $meta = new FileMeta($scope->getFile(), $node->getStartLine(), 1);
57        $declared = self::declaredNode(
58            $node,
59            $node->namespacedName->toString(),
60            $meta,
61            DeclarationReader::forClassLike($node, AttributeProcessor::usages($node->attrGroups, $scope)),
62        );
63        if ($declared === null) {
64            return [];
65        }
66
67        return [
68            $declared,
69            ...AttributeProcessor::process($node->attrGroups, $declared, $scope),
70            ...self::inheritance($node, $declared, $meta),
71        ];
72    }
73
74    /**
75     * Builds the node for the declaration itself.
76     *
77     * @param ClassLike         $node        The syntax node met during analysis
78     * @param string            $className   The fully qualified name it declares
79     * @param FileMeta          $meta        Where the declaration is written
80     * @param SymbolDeclaration $declaration What the declaration says about itself
81     *
82     * @return null|ClassNode|EnumNode|GraphInterfaceNode|TraitNode The declared node, or null for a kind with no node
83     */
84    public static function declaredNode(ClassLike $node, string $className, FileMeta $meta, SymbolDeclaration $declaration): ClassNode|EnumNode|GraphInterfaceNode|TraitNode|null
85    {
86        return match (true) {
87            $node instanceof Class_ => new ClassNode(ClassNodeId::of($className), true, $meta, $declaration),
88            $node instanceof Interface_ => new GraphInterfaceNode(InterfaceNodeId::of($className), true, $meta, $declaration),
89            $node instanceof Trait_ => new TraitNode(TraitNodeId::of($className), true, $meta, $declaration),
90            $node instanceof Enum_ => new EnumNode(EnumNodeId::of($className), true, $meta, $declaration),
91            default => null,
92        };
93    }
94
95    /**
96     * Records what a declaration extends, implements and uses.
97     *
98     * Which of those a declaration may have is decided by its kind, and PHP's own
99     * rules are what the branches here say: only a class extends a class, only an
100     * interface extends several, an enum implements but never extends, and every
101     * class-like except an interface may use a trait — an enum included, which the
102     * previous shape of this code asserted against.
103     *
104     * @param ClassLike                                       $node     The syntax node met during analysis
105     * @param ClassNode|EnumNode|GraphInterfaceNode|TraitNode $declared The node of the declaration itself
106     * @param FileMeta                                        $meta     Where the declaration is written
107     *
108     * @return list<ExtendsEdge|ImplementsEdge|TraitUseEdge> The inheritance relations
109     */
110    public static function inheritance(ClassLike $node, ClassNode|EnumNode|GraphInterfaceNode|TraitNode $declared, FileMeta $meta): array
111    {
112        $items = [];
113
114        if ($node instanceof Class_ && $node->extends !== null && $declared instanceof ClassNode) {
115            $items[] = new ExtendsEdge($declared, new ClassNode(ClassNodeId::of($node->extends->toString()), false, null), $meta);
116        }
117
118        if ($node instanceof Interface_ && $declared instanceof GraphInterfaceNode) {
119            foreach ($node->extends as $parent) {
120                $items[] = new ExtendsEdge($declared, new GraphInterfaceNode(InterfaceNodeId::of($parent->toString()), false, null), $meta);
121            }
122        }
123
124        if (($node instanceof Class_ || $node instanceof Enum_) && ($declared instanceof ClassNode || $declared instanceof EnumNode)) {
125            foreach ($node->implements as $contract) {
126                $items[] = new ImplementsEdge($declared, new GraphInterfaceNode(InterfaceNodeId::of($contract->toString()), false, null), $meta);
127            }
128        }
129
130        if (!$declared instanceof GraphInterfaceNode) {
131            foreach ($node->getTraitUses() as $traitUse) {
132                foreach ($traitUse->traits as $trait) {
133                    $items[] = new TraitUseEdge($declared, new TraitNode(TraitNodeId::of($trait->toString()), false, null), $meta);
134                }
135            }
136        }
137
138        return $items;
139    }
