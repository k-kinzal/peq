7/**
8 * Which format a report is written in.
9 *
10 * peq answers one question — what does this symbol reach, or what reaches it —
11 * and the answer is read by very different readers. A person scanning a terminal
12 * wants the shape of the graph at a glance; a program wants fields it can address;
13 * a picture wants nodes and arrows; a review wants a flat list it can sort and
14 * grep. Those are four ways of writing down the same walk, not four analyses, and
15 * naming them as a closed type is what lets the reporter factory answer the choice
16 * with one arm per format rather than a branch inside a command.
17 */
18enum OutputFormat: string
19{
20    /** Draws the walk as an indented tree, the way `tree` draws a directory */
21    case Tree = 'tree';
22
23    /** Writes the walk as a JSON document, for a program or an agent to read */
24    case Json = 'json';
25
26    /** Writes the walk as a Graphviz digraph, for rendering as a picture */
27    case Dot = 'dot';
28
29    /** Writes the walk as a table, one row per symbol it reached */
30    case Table = 'table';
31
32    /** Draws the graph itself in the terminal: every symbol once, joined by arrows, left to right */
33    case Graph = 'graph';
34
35    /** Writes the graph itself as a Mermaid flowchart, for a page that renders one */
36    case Mermaid = 'mermaid';
37
38    /**
39     * Writes out the formats a report can be asked for, as they are written.
40     *
41     * The command line documents its own choices from this rather than from a
42     * sentence someone has to remember to update, so a format that exists is a
43     * format the help text offers.
44     *
45     * @example The choices are written the way the command line spells them
46     *     \App\Config\OutputFormat::spell() // => 'tree|json|dot|table|graph|mermaid'
47     *
48     * @return string The formats, separated by the character that separates them on the command line
49     */
50    public static function spell(): string
51    {
52        return implode('|', array_map(static fn (self $format): string => $format->value, self::cases()));
53    }
54}
