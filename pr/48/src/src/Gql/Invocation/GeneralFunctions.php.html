16/**
17 * The functions GQL offers that belong to no one kind of value.
18 *
19 * `coalesce` and `nullif` are the two moves a query makes around absent values: the
20 * first supplies one, the second withdraws one. Between them they are how a reader
21 * escapes three-valued logic without writing a `CASE`.
22 *
23 * `to_json_string` is the one that matters most for the reader peq is built for. An
24 * agent that has matched a pattern and wants the whole of what it found — labels,
25 * properties and all — asks for it as JSON in one column rather than enumerating
26 * properties it would have to know the names of first.
27 *
28 * @visibility App\Gql
29 */
30final class GeneralFunctions
31{
32    /**
33     * Returns the first value that is there.
34     *
35     * @param list<Datum> $values The values, in the order they are preferred
36     *
37     * @example The first value that is there is the answer
38     *     $values = [new \App\Gql\Datum\NullDatum(), new \App\Gql\Datum\StringDatum('Unknown')];
39     *     \App\Gql\Invocation\GeneralFunctions::coalesce($values)->toText() // => 'Unknown'
40     * @example When none of them is there, neither is the answer
41     *     \App\Gql\Invocation\GeneralFunctions::coalesce([new \App\Gql\Datum\NullDatum()])->kind() // => \App\Gql\Datum\DatumKind::Null
42     *
43     * @return Datum The first value that is there, or the absence of one
44     */
45    public static function coalesce(array $values): Datum
46    {
47        foreach ($values as $value) {
48            if ($value->kind() !== DatumKind::Null) {
49                return $value;
50            }
51        }
52
53        return new NullDatum();
54    }
55
56    /**
57     * Returns a value, unless it equals another, in which case nothing.
58     *
59     * This is how a query turns a stand-in value back into an absent one: a
60     * visibility recorded as `unknown`, a file path recorded as an empty string.
61     *
62     * @param Datum $value    The value
63     * @param Datum $withdraw What it must not equal
64     *
65     * @example A value that equals the one withdrawn becomes absent
66     *     $value = new \App\Gql\Datum\StringDatum('unknown');
67     *     \App\Gql\Invocation\GeneralFunctions::nullif($value, new \App\Gql\Datum\StringDatum('unknown'))->kind() // => \App\Gql\Datum\DatumKind::Null
68     * @example Any other value is itself
69     *     $value = new \App\Gql\Datum\StringDatum('public');
70     *     \App\Gql\Invocation\GeneralFunctions::nullif($value, new \App\Gql\Datum\StringDatum('unknown'))->toText() // => 'public'
71     *
72     * @return Datum The value, or the absence of one
73     *
74     * @throws GqlException If the two values cannot be compared
75     */
76    public static function nullif(Datum $value, Datum $withdraw): Datum
77    {
78        return DatumOrder::equals($value, $withdraw) === true ? new NullDatum() : $value;
79    }
80
81    /**
82     * Returns a moment in time: the one written, or the one it is now.
83     *
84     * @param list<Datum> $values The moment as written, or nothing to mean now
85     *
86     * @example A moment written in ISO 8601 is read as that moment
87     *     $written = [new \App\Gql\Datum\StringDatum('2024-01-15T10:30:00+00:00')];
88     *     \App\Gql\Invocation\GeneralFunctions::zonedDatetime($written)->toText() // => '2024-01-15T10:30:00+00:00'
89     * @example Something that is not a moment is reported rather than guessed at
90     *     \App\Gql\Invocation\GeneralFunctions::zonedDatetime([new \App\Gql\Datum\StringDatum('yesterday-ish')]) // throws \App\Gql\GqlException: invalid value type
91     *
92     * @return Datum The moment
93     *
94     * @throws GqlException If what is written is not a moment
95     */
96    public static function zonedDatetime(array $values): Datum
97    {
98        if ($values !== [] && $values[0]->kind() === DatumKind::Null) {
99            return new NullDatum();
100        }
101
102        $written = $values === [] ? 'now' : TextArgument::of($values[0]);
103        $moment = date_create_immutable($written);
104        if ($moment === false) {
105            throw GqlException::because(StatusCode::InvalidType, sprintf('"%s" is not a moment in time', $written));
106        }
107
108        return new DateTimeDatum($moment);
109    }
110}
