13/**
14 * How far one attempt at matching a pattern has got.
15 *
16 * Matching a graph pattern is a search, and a search needs to remember four things at
17 * once: what it has bound so far, where in the graph it is standing, the path it took
18 * to get there, and what it has already been through. The last two are not the same:
19 * the path is the answer a query can ask for, and what has been been through is what
20 * keeps the search finite.
21 *
22 * Every step produces a new state rather than changing this one, because a search
23 * that finds three ways forward has to carry on down all three from the same place.
24 *
25 * @visibility App\Gql
26 */
27final readonly class MatchState
28{
29    /**
30     * @param BindingRow          $row     What the attempt has bound so far
31     * @param null|NodeDatum      $current Where in the graph it is standing, or null before it has started
32     * @param null|PathDatum      $path    The path it took to get there, or null before it has started
33     * @param array<string, true> $edges   The relations it has crossed, by what identifies them
34     * @param array<string, true> $nodes   The symbols it has met, by what identifies them
35     * @param null|string         $start   What identifies the symbol it started from, or null before it has started
36     */
37    public function __construct(
38        public BindingRow $row,
39        public ?NodeDatum $current = null,
40        public ?PathDatum $path = null,
41        private array $edges = [],
42        private array $nodes = [],
43        private ?string $start = null,
44    ) {}

46    /**
47     * Returns the attempt as it stands before it has gone anywhere.
48     *
49     * @param BindingRow $row What is already bound when the pattern is reached
50     *
51     * @example An attempt that has not started stands nowhere
52     *     \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->current // => null
53     *
54     * @return self The attempt
55     */
56    public static function before(BindingRow $row): self
57    {
58        return new self($row);
59    }
60
61    /**
62     * Returns the attempt standing at a symbol it has just started from.
63     *
64     * @param NodeDatum $node The symbol
65     *
66     * @example Starting somewhere begins a path there
67     *     $started = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
68     *     $started->path?->length() // => 0
69     *
70     * @return self The attempt, standing there
71     */
72    public function startingAt(NodeDatum $node): self
73    {
74        return new self($this->row, $node, PathDatum::at($node), $this->edges, [$node->id => true], $node->id);
75    }
76
77    /**
78     * Returns the attempt having crossed a relation to another symbol.
79     *
80     * @param EdgeDatum $edge The relation crossed
81     * @param NodeDatum $node The symbol arrived at
82     *
83     * @example Crossing a relation makes the path one longer
84     *     $started = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
85     *     $crossed = $started->across(new \App\Gql\Datum\EdgeDatum('e', [], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('b'));
86     *     $crossed->path?->length() // => 1
87     *
88     * @return self The attempt, standing at the other symbol
89     */
90    public function across(EdgeDatum $edge, NodeDatum $node): self
91    {
92        $path = $this->path ?? PathDatum::at($node);
93
94        return new self(
95            $this->row,
96            $node,
97            $this->path === null ? $path : $path->continuedBy($edge, $node),
98            [...$this->edges, $edge->id => true],
99            [...$this->nodes, $node->id => true],
100            $this->start ?? $node->id,
101        );
102    }

104    /**
105     * Returns the attempt with what it has bound replaced, and everything else about it kept.
106     *
107     * A repeated group binds its names afresh on every repetition, so each repetition
108     * is matched against a row without them and the row is put back together after.
109     *
110     * @param BindingRow $row What the attempt has bound
111     *
112     * @example The path an attempt has walked is kept
113     *     $state = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
114     *     $state->withRow(\App\Gql\Binding\BindingRow::unit())->current?->id // => 'a'
115     *
116     * @return self The attempt, with that row
117     */
118    public function withRow(BindingRow $row): self
119    {
120        return new self($row, $this->current, $this->path, $this->edges, $this->nodes, $this->start);
121    }
122
123    /**
124     * Returns the attempt with one more name bound.
125     *
126     * @param string $name  The name
127     * @param Datum  $value What to bind it to
128     *
129     * @example Binding a name leaves everything else about the attempt alone
130     *     $before = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit());
131     *     $before->bind('n', new \App\Gql\Datum\IntegerDatum(1))->current // => null
132     *
133     * @return self The attempt, with that binding
134     */
135    public function bind(string $name, Datum $value): self
136    {
137        return new self($this->row->with($name, $value), $this->current, $this->path, $this->edges, $this->nodes, $this->start);
138    }

140    /**
141     * Reports whether the attempt has already crossed a relation.
142     *
143     * @param string $id What identifies the relation
144     *
145     * @example An attempt that has gone nowhere has crossed nothing
146     *     \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->crossed('e') // => false
147     *
148     * @return bool True when it has
149     */
150    public function crossed(string $id): bool
151    {
152        return isset($this->edges[$id]);
153    }
154
155    /**
156     * Reports whether the attempt has already met a symbol.
157     *
158     * @param string $id What identifies the symbol
159     *
160     * @example An attempt has met the symbol it started from
161     *     $started = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
162     *     $started->met('a') // => true
163     *
164     * @return bool True when it has
165     */
166    public function met(string $id): bool
167    {
168        return isset($this->nodes[$id]);
169    }

171    /**
172     * Reports whether a symbol is the one the attempt started from.
173     *
174     * A path that comes back to where it started is a cycle, and one mode allows
175     * exactly that and nothing else repeated, which is why this is asked separately
176     * from whether a symbol has been met.
177     *
178     * @param string $id What identifies the symbol
179     *
180     * @example The symbol an attempt started from is where it started
181     *     $started = \App\Gql\Matching\MatchState::before(\App\Gql\Binding\BindingRow::unit())->startingAt(new \App\Gql\Datum\NodeDatum('a'));
182     *     $started->startedAt('a') // => true
183     *
184     * @return bool True when it is
185     */
186    public function startedAt(string $id): bool
187    {
188        return $this->start === $id;
189    }
