9/**
10 * The clause and subclause numbers of ISO/IEC 39075, read off its table of contents.
11 *
12 * Every scenario of this suite says which subclause of the standard it states, and a
13 * number is the easiest thing in a specification to get wrong: the digits look right,
14 * the topic beside them is one a reader recognises, and nothing complains. So the
15 * numbers are looked up, and a scenario pointing at a subclause the standard does not
16 * number fails the run.
17 *
18 * ISO publishes no machine-readable form of its own table of contents, so this is the
19 * one list under `spec/iso/` that is transcribed rather than downloaded. It keeps the
20 * numbers only: they are facts about the document, where the titles are ISO's text.
21 *
22 * @see spec/iso/subclauses.txt Where the numbers are read from
23 */
24final class IsoSubclauses
25{
26    /**
27     * Where the transcription sits.
28     */
29    private const TRANSCRIPTION = __DIR__.'/../iso/subclauses.txt';
30
31    /**
32     * Reports whether the standard numbers a clause or subclause.
33     *
34     * @param string $number The number, as a tag writes it
35     *
36     * @return bool True when it does
37     *
38     * @throws RuntimeException If the transcription cannot be read
39     */
40    public static function numbers(string $number): bool
41    {
42        return in_array($number, self::all(), true);
43    }
44
45    /**
46     * Returns every clause and subclause number the standard has.
47     *
48     * @return list<string> The numbers, in document order
49     *
50     * @throws RuntimeException If the transcription cannot be read
51     */
52    public static function all(): array
53    {
54        static $numbers = null;
55        if ($numbers !== null) {
56            return $numbers;
57        }
58
59        $read = @file_get_contents(self::TRANSCRIPTION);
60        if ($read === false) {
61            throw new RuntimeException(sprintf('Cannot read the subclause numbers at %s', self::TRANSCRIPTION));
62        }
63
64        $numbers = [];
65        foreach (explode("\n", $read) as $line) {
66            $line = trim($line);
67            if ($line !== '' && !str_starts_with($line, '#')) {
68                $numbers[] = $line;
69            }
70        }
71
72        return $numbers;
73    }
74}
