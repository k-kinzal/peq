20/**
21 * Steps that state the conformance claim itself, against ISO's own enumerations.
22 *
23 * Clause 24 of ISO/IEC 39075 does not let an implementation say "it conforms" and stop.
24 * A claim names the class of conformance and then, under 24.3, the optional features
25 * implemented; 24.5.3 requires any extension to be stated too. That makes the claim a
26 * register, and a register can be checked: the standard publishes the feature codes and
27 * the condition codes as artifacts, so every line can be looked up rather than believed.
28 *
29 * The check that matters most is the last one. A register listing features nothing
30 * exercises would be a longer way of asserting conformance; requiring every claimed
31 * feature to be stated by a scenario of this suite is what makes the claim cost
32 * something to make.
33 */
34final class ConformanceContext implements Context

210    /**
211     * States that every word the implementation spells is a word the grammar writes.
212     *
213     * A word of an implementation's own is the cheapest extension there is to make and
214     * the most expensive to have made: `CONTAINS` is useful, reads well, and quietly
215     * turns the language into a dialect that has to be explained. So every upper-case
216     * word written as a literal anywhere in the engine is looked up in ISO's grammar
217     * artifact, which is where the answer to "is that a GQL word?" actually lives.
218     */
219    #[Then('every word the implementation spells is one ISO\/IEC 39075\'s grammar writes')]
220    public function everyWordTheImplementationSpellsIsOneTheGrammarWrites(): void
221    {
222        $written = IsoGrammar::keywords();
223        $invented = [];
224        foreach (EngineVocabulary::words() as $word => $found) {
225            if (!in_array($word, $written, true)) {
226                $invented[] = $word.' ('.implode(', ', $found[1]).')';
227            }
228        }
229
230        Assert::assertSame(
231            [],
232            $invented,
233            'The implementation spells words ISO/IEC 39075 does not write. '
234            .'A word of its own makes the query language a dialect rather than the one its readers know.',
235        );
236    }

334     * States that every function the implementation offers is one GQL calls.
335     *
336     * A word being in the grammar does not make it a function: `LABELS` is in there,
337     * in the syntax for declaring a graph type, and `labels(x)` is still not GQL. So
338     * each name is looked up among the words the grammar actually writes a left
339     * parenthesis after, which is the only reading of "GQL has this function" that
340     * does not come down to somebody's memory.
341     */
342    #[Then('every function the implementation offers is one ISO\/IEC 39075\'s grammar calls')]
343    public function everyFunctionOfferedIsOneTheGrammarCalls(): void
344    {
345        $called = IsoGrammar::calledNames();
346        $invented = [];
347        foreach ([...FunctionCatalog::all(), ...AggregateCatalog::all()] as $name) {
348            if (!in_array(strtoupper($name), $called, true)) {
349                $invented[] = $name;

351        }
352
353        Assert::assertSame(

361    /**
362     * States that every name the schema reports back is a name a query can write.
363     *
364     * A graph is named by whoever wrote the code it was read from, so some of its
365     * labels and properties spell words GQL reserves. `--schema` is the only thing an
366     * agent has to write a query from, which makes a name it reports that no query can
367     * write a worse answer than no answer at all.
368     */
369    #[Then('every name the schema offers is one a query can write')]
370    public function everyNameTheSchemaOffersIsOneAQueryCanWrite(): void
371    {
372        foreach (self::schemaNames() as $category => $names) {
373            foreach ($names as $name) {
374                $program = str_contains($category, 'label')
375                    ? sprintf('MATCH (x:%s) RETURN 1 AS n', $name)
376                    : sprintf('MATCH (x) RETURN x.%s AS n', $name);
377
378                Assert::assertTrue(
379                    self::reads($program),
380                    sprintf('The schema offers the %s "%s", which no query can write.', $category, $name),
381                );
382            }
383        }
384    }

386    /**
387     * Returns the labels and properties the schema offers, by category.
388     *
389     * @return array<string, list<string>> The names, by category
390     */
391    private static function schemaNames(): array
392    {
393        $names = [];
394        foreach (GraphSchema::table()->rows as $row) {
395            $category = $row->values[0]->toText();
396            if (!str_contains($category, 'label') && !str_contains($category, 'property')) {
397                continue;
398            }
399            $names[$category][] = $row->values[1]->toText();
400        }
401
402        return $names;
403    }
