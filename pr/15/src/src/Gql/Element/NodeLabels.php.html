10/**
11 * The labels a symbol carries when a query looks at it.
12 *
13 * A GQL pattern selects by label, so what labels exist decides what questions are
14 * easy to ask. Each symbol therefore carries three kinds at once: what it is —
15 * `Method`, `Class` — what family it belongs to — `ClassLike`, `Member`, `Callable` —
16 * and whether analysis actually found it.
17 *
18 * The families are the useful part. "Every callable that a controller reaches"
19 * covers methods and functions in one pattern; "every class-like that is not an
20 * interface" is `(:ClassLike&!Interface)` rather than a disjunction someone has to
21 * remember to keep up to date. Without them, a query would have to enumerate kinds,
22 * and would quietly stop being right when a kind was added.
23 *
24 * @visibility App\Gql
25 */
26final class NodeLabels
27{
28    /**
29     * Returns the labels a symbol carries.
30     *
31     * @param Node $node The symbol
32     *
33     * @example A method belongs to two families at once
34     *     $named = \App\Analyzer\Graph\NodeId\MethodNodeId::of('App\\Invoice', 'total');
35     *     \App\Gql\Element\NodeLabels::of(new \App\Analyzer\Graph\Node\MethodNode($named, true)) // => ['Method', 'Member', 'Callable', 'Resolved']
36     * @example A symbol analysis never found says so
37     *     $named = new \App\Analyzer\Graph\NodeId\UnknownNodeId('App\\Missing');
38     *     \App\Gql\Element\NodeLabels::of(new \App\Analyzer\Graph\Node\UnknownNode($named)) // => ['Unknown', 'Unresolved']
39     *
40     * @return list<string> The labels, most specific first
41     */
42    public static function of(Node $node): array
43    {
44        return [...self::forKind($node->kind()), $node->resolved() ? 'Resolved' : 'Unresolved'];
45    }
46
47    /**
48     * Returns the labels a kind of symbol carries, apart from whether it was found.
49     *
50     * @param NodeKind $kind The kind of symbol
51     *
52     * @example A class is a class-like
53     *     \App\Gql\Element\NodeLabels::forKind(\App\Analyzer\Graph\NodeKind::Klass) // => ['Class', 'ClassLike']
54     * @example A function is callable without being a member of anything
55     *     \App\Gql\Element\NodeLabels::forKind(\App\Analyzer\Graph\NodeKind::Function) // => ['Function', 'Callable']
56     *
57     * @return list<string> The labels, most specific first
58     */
59    public static function forKind(NodeKind $kind): array
60    {
61        return match ($kind) {
62            NodeKind::Klass => ['Class', 'ClassLike'],
63            NodeKind::Interface => ['Interface', 'ClassLike'],
64            NodeKind::Trait => ['Trait', 'ClassLike'],
65            NodeKind::Enum => ['Enum', 'ClassLike'],
66            NodeKind::Method => ['Method', 'Member', 'Callable'],
67            NodeKind::Function => ['Function', 'Callable'],
68            NodeKind::Property => ['Property', 'Member'],
69            NodeKind::Constant => ['Constant', 'Member'],
70            NodeKind::EnumCase => ['EnumCase', 'Member'],
71            NodeKind::Builtin => ['Builtin'],
72            NodeKind::Unknown => ['Unknown'],
73        };
74    }
75
76    /**
77     * Returns every label a symbol can carry, for a reader asking what there is.
78     *
79     * An agent writing a query against a graph it has not seen needs to know what
80     * the labels are before it can ask anything, and working them out by looking at
81     * the answer to a query is a poor substitute for being told.
82     *
83     * @example The families are among the labels a query can ask for
84     *     in_array('Callable', \App\Gql\Element\NodeLabels::all(), true) // => true
85     *
86     * @return list<string> Every label, without repetition
87     */
88    public static function all(): array
89    {
90        $labels = [];
91        foreach (NodeKind::cases() as $kind) {
92            foreach (self::forKind($kind) as $label) {
93                $labels[$label] = true;
94            }
95        }
96
97        return [...array_keys($labels), 'Resolved', 'Unresolved'];
98    }
99}
