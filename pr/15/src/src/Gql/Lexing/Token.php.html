5namespace App\Gql\Lexing;

7/**
8 * One piece of a query text, with where it was written.
9 *
10 * A token carries two forms of itself. The lexeme is what the query said, and it is
11 * what an error message quotes back so the reader recognises their own text. The
12 * value is what it means: the characters inside a string once its escapes are
13 * resolved, the name inside a pair of backticks. Keeping both is what lets a message
14 * say `found "\n"` while the engine works with a newline.
15 */
16final readonly class Token
17{
18    /**
19     * @param TokenKind $kind   What kind of thing it is
20     * @param string    $lexeme The text exactly as the query wrote it
21     * @param string    $value  What that text stands for
22     * @param int       $line   The line it starts on, counting from one
23     * @param int       $column The column it starts at, counting from one
24     * @param int       $offset How far into the query text it starts
25     */
26    public function __construct(
27        public TokenKind $kind,
28        public string $lexeme,
29        public string $value,
30        public int $line,
31        public int $column,
32        public int $offset = 0,
33    ) {
34        assert($this->line > 0, 'A token is written on a line, counted from one');
35        assert($this->column > 0, 'A token is written at a column, counted from one');
36    }
37
38    /**
39     * Reports whether this token is a particular piece of punctuation or operator.
40     *
41     * @param string $lexeme The symbol to test for, as it is written
42     *
43     * @example A symbol is recognised by what it is written as
44     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::Symbol, '(', '(', 1, 1))->isSymbol('(') // => true
45     * @example A name that happens to read like one is not a symbol
46     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::Name, 'p', 'p', 1, 1))->isSymbol('p') // => false
47     *
48     * @return bool True when this token is that symbol
49     */
50    public function isSymbol(string $lexeme): bool
51    {
52        return $this->kind === TokenKind::Symbol && $this->lexeme === $lexeme;
53    }
54
55    /**
56     * Reports whether this token is a bare name spelling a given keyword.
57     *
58     * GQL keywords are matched without regard to case, and a name written in
59     * backticks is never one, which is how a query talks about a property called
60     * `end` without the grammar taking it for the end of a CASE.
61     *
62     * @param string $keyword The keyword to test for
63     *
64     * @example A keyword is recognised however it is cased
65     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::Name, 'match', 'match', 1, 1))->isKeyword('MATCH') // => true
66     * @example A name written in backticks is a name
67     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::QuotedName, '`match`', 'match', 1, 1))->isKeyword('MATCH') // => false
68     *
69     * @return bool True when this token spells that keyword
70     */
71    public function isKeyword(string $keyword): bool
72    {
73        return $this->kind === TokenKind::Name && strcasecmp($this->value, $keyword) === 0;
74    }
75
76    /**
77     * Describes the token the way an error message quotes it back.
78     *
79     * @example A token is quoted as the reader wrote it
80     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::Name, 'RETRUN', 'RETRUN', 1, 1))->describe() // => '"RETRUN"'
81     * @example The end of a query has nothing to quote
82     *     (new \App\Gql\Lexing\Token(\App\Gql\Lexing\TokenKind::End, '', '', 1, 1))->describe() // => 'the end of the query'
83     *
84     * @return string The token, as an error message shows it
85     */
86    public function describe(): string
87    {
88        if ($this->kind === TokenKind::End) {
89            return 'the end of the query';
90        }
91
92        return '"'.$this->lexeme.'"';
93    }
94}
