7/**
8 * The kinds of value a GQL query can hold.
9 *
10 * GQL names its types the way SQL does, and a query that fails because a value was
11 * of the wrong kind should say so in those names rather than in PHP's. Keeping the
12 * mapping here means one place decides that a whole number is an `INT64` and a path
13 * is a `PATH`, and every message, every result column heading and every type error
14 * agrees about it.
15 */
16enum DatumKind
17{
18    /** The absence of a value, which GQL also reads as the unknown truth value */
19    case Null;
20
21    /** A truth value */
22    case Boolean;
23
24    /** A whole number */
25    case Integer;
26
27    /** An exact number with digits after the point */
28    case Decimal;
29
30    /** An approximate number */
31    case Float;
32
33    /** A character string */
34    case Text;
35
36    /** A list of values */
37    case ListOf;
38
39    /** A reference to a node of the graph */
40    case Node;
41
42    /** A reference to an edge of the graph */
43    case Edge;
44
45    /** A path through the graph */
46    case Path;
47
48    /** A moment in time, with the offset it is written in */
49    case DateTime;
50
51    /**
52     * Names the kind the way GQL writes the type.
53     *
54     * @example A whole number is what GQL calls an INT64
55     *     \App\Gql\Datum\DatumKind::Integer->typeName() // => 'INT64'
56     * @example A moment carries its offset, and GQL says so in the name
57     *     \App\Gql\Datum\DatumKind::DateTime->typeName() // => 'ZONED DATETIME'
58     *
59     * @return string The GQL name of the type
60     */
61    public function typeName(): string
62    {
63        return match ($this) {
64            self::Null => 'NULL',
65            self::Boolean => 'BOOL',
66            self::Integer => 'INT64',
67            self::Decimal => 'DECIMAL',
68            self::Float => 'FLOAT64',
69            self::Text => 'STRING',
70            self::ListOf => 'LIST',
71            self::Node => 'NODE',
72            self::Edge => 'EDGE',
73            self::Path => 'PATH',
74            self::DateTime => 'ZONED DATETIME',
75        };
76    }
77
78    /**
79     * Reports whether values of this kind are numbers.
80     *
81     * Arithmetic and numeric comparison both need the same answer, and GQL's rule is
82     * that the two numeric kinds mix freely and nothing else joins them.
83     *
84     * @example Both numeric kinds are numbers
85     *     \App\Gql\Datum\DatumKind::Float->numeric() // => true
86     * @example A string that happens to spell one is not
87     *     \App\Gql\Datum\DatumKind::Text->numeric() // => false
88     *
89     * @return bool True when the kind is one of the numeric kinds
90     */
91    public function numeric(): bool
92    {
93        return match ($this) {
94            self::Integer,
95            self::Decimal,
96            self::Float => true,
97
98            self::Null,
99            self::Boolean,
100            self::Text,
101            self::ListOf,
102            self::Node,
103            self::Edge,
104            self::Path,
105            self::DateTime => false,
106        };
107    }
108}
