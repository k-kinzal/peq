10/**
11 * Reads one attribute off a declaration, under the name it resolves to.
12 *
13 * Resolving the written name into a fully qualified one is the analyser's job and
14 * differs between the ways peq reads sources, so the resolved name is handed in
15 * rather than worked out here. What is worked out here is the rest: the arguments,
16 * written back as the text that gave them.
17 *
18 * @visibility App\Analyzer
19 */
20final class WrittenAttribute
21{
22    /**
23     * Reads one attribute as it was written on a declaration.
24     *
25     * @param Attribute $attribute    The attribute as it was written
26     * @param string    $resolvedName The fully qualified name it resolves to
27     *
28     * @example An attribute keeps the arguments it was given
29     *     $written = new \PhpParser\Node\Attribute(
30     *         new \PhpParser\Node\Name('Route'),
31     *         [new \PhpParser\Node\Arg(new \PhpParser\Node\Scalar\String_('/users'))],
32     *     );
33     *     \App\Analyzer\Declaration\WrittenAttribute::usage($written, 'App\\Http\\Route')->toString() // => "App\\Http\\Route('/users')"
34     *
35     * @return AttributeUsage The attribute, under the name it resolves to
36     */
37    public static function usage(Attribute $attribute, string $resolvedName): AttributeUsage
38    {
39        $arguments = [];
40        foreach ($attribute->args as $argument) {
41            $arguments[] = ExpressionText::argument($argument);
42        }
43
44        return new AttributeUsage($resolvedName, $arguments);
45    }
46}
