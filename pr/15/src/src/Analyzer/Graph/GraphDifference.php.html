5namespace App\Analyzer\Graph;

7/**
8 * What one graph holds and another does not, in both directions.
9 *
10 * A difference is reported rather than merely detected, because the question it
11 * answers is never "are these two graphs equal" alone: when a second analyzer is
12 * checked against the first, the answer has to say which symbol or relation one of
13 * them invented and which one it lost, or the check cannot be acted on.
14 *
15 * The two directions are kept apart for the same reason. A relation missing from an
16 * impact analysis is a symbol that will be reported as safe to change when it is not,
17 * which is the failure that matters; a relation too many is noise. Telling them apart
18 * is the difference between a bug and a nuisance.
19 */
20final class GraphDifference
21{
22    /**
23     * How many lines of each kind are written out before the rest is summarised.
24     *
25     * A difference between two analyses of a whole codebase can run to thousands of
26     * lines, and reading the first few of each kind is what locates the cause.
27     */
28    private const REPORTED_LINES = 12;
29
30    /**
31     * @param list<string> $missingNodes    Symbols the expected graph holds and this one does not
32     * @param list<string> $unexpectedNodes Symbols this graph holds and the expected one does not
33     * @param list<string> $missingEdges    Relations the expected graph holds and this one does not
34     * @param list<string> $unexpectedEdges Relations this graph holds and the expected one does not
35     */
36    public function __construct(
37        public readonly array $missingNodes,
38        public readonly array $unexpectedNodes,
39        public readonly array $missingEdges,
40        public readonly array $unexpectedEdges,
41    ) {}
42
43    /**
44     * Reports whether the two graphs say exactly the same thing.
45     *
46     * @example Nothing on either side is no difference
47     *     (new \App\Analyzer\Graph\GraphDifference([], [], [], []))->isEmpty() // => true
48     * @example One symbol on one side is a difference
49     *     (new \App\Analyzer\Graph\GraphDifference(['class A resolved=yes at nowhere'], [], [], []))->isEmpty() // => false
50     *
51     * @return bool True when neither graph holds anything the other lacks
52     */
53    public function isEmpty(): bool
54    {
55        return $this->missingNodes === []
56            && $this->unexpectedNodes === []
57            && $this->missingEdges === []
58            && $this->unexpectedEdges === [];
59    }
60
61    /**
62     * Counts everything the two graphs disagree about.
63     *
64     * @example Agreement counts as nothing
65     *     (new \App\Analyzer\Graph\GraphDifference([], [], [], []))->count() // => 0
66     *
67     * @return int The number of symbols and relations only one of the graphs holds
68     */
69    public function count(): int
70    {
71        return count($this->missingNodes)
72            + count($this->unexpectedNodes)
73            + count($this->missingEdges)
74            + count($this->unexpectedEdges);
75    }
76
77    /**
78     * Describes the difference in a form a failing check can print.
79     *
80     * @example Agreement describes itself as such
81     *     (new \App\Analyzer\Graph\GraphDifference([], [], [], []))->describe() // => 'the graphs are identical'
82     *
83     * @return string The difference, written out
84     */
85    public function describe(): string
86    {
87        if ($this->isEmpty()) {
88            return 'the graphs are identical';
89        }
90
91        return implode("\n", array_merge(
92            [sprintf('%d differences', $this->count())],
93            self::section('symbols only the reference engine found', $this->missingNodes),
94            self::section('symbols only this engine found', $this->unexpectedNodes),
95            self::section('relations only the reference engine found', $this->missingEdges),
96            self::section('relations only this engine found', $this->unexpectedEdges),
97        ));
98    }
99
100    /**
101     * Writes out one side of the difference, shortened when it is long.
102     *
103     * @param string       $title The heading of the section
104     * @param list<string> $lines What that side holds
105     *
106     * @example A side that holds nothing is left out
107     *     \App\Analyzer\Graph\GraphDifference::section('lost', []) // => []
108     *
109     * @return list<string> The lines of that section, or none when the side is empty
110     */
111    public static function section(string $title, array $lines): array
112    {
113        if ($lines === []) {
114            return [];
115        }
116
117        $shown = array_slice($lines, 0, self::REPORTED_LINES);
118        $rest = count($lines) - count($shown);
119
120        return array_merge(
121            [sprintf('  %s (%d):', $title, count($lines))],
122            array_map(static fn (string $line): string => '    '.$line, $shown),
123            $rest > 0 ? [sprintf('    ... and %d more', $rest)] : [],
124        );
125    }
126}
