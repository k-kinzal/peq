7/**
8 * How the arrows between two neighbouring columns get across, without two of them
9 * ever reading as one.
10 *
11 * Lines that meet in a drawing join, and a reader follows a join either way. So the
12 * arrows between two columns are routed in two stages, which keeps every join between
13 * arrows that belong together:
14 *
15 * - on the left, a lane for each place with several arrows leaving it, where they fan
16 *   out: everything on it leaves that one place;
17 * - on the right, a lane for each place several arrows arrive at, where they gather:
18 *   everything on it arrives at that one place.
19 *
20 *         ┌──▶ B ──┐
21 *     A ──┤        ├──▶ D
22 *         └──▶ C ──┘
23 *
24 * An arrow from a place that fans out to a place that gathers goes over from one lane
25 * to the other on a line of its own: places are only ever drawn on even lines, and
26 * each such arrow is given an odd one no other arrow between the two columns uses.
27 * An arrow between two places on the same line, neither of which has another arrow on
28 * this side, is drawn straight across.
29 *
30 * What is left to go wrong is the order of the fanning lanes: an arrow entering one on
31 * a line another leaves on, further left, would share that stretch of line. That
32 * order is avoided above any number of crossings, and where no order avoids it the
33 * arrow leaving is sent over on a line of its own instead.
34 *
35 * @visibility App\Reporter\Diagram
36 */
37final readonly class LaneRouting

55    /**
56     * Routes the arrows that leave one column for the next.
57     *
58     * @param DiagramLayout $layout Where everything goes
59     * @param int           $layer  The column the arrows leave
60     *
61     * @example Two arrows fanning out from one place share its lane
62     *     $diagram = new \App\Reporter\Diagram\Diagram();
63     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('a'));
64     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('b'));
65     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('c'));
66     *     $diagram->relate(new \App\Reporter\Diagram\DiagramEdge('a', 'b'));
67     *     $diagram->relate(new \App\Reporter\Diagram\DiagramEdge('a', 'c'));
68     *     $layout = \App\Reporter\Diagram\Layout\LayeredLayout::of($diagram);
69     *     count(\App\Reporter\Diagram\Layout\LaneRouting::of($layout, 0)->lanes) // => 1
70     * @example An arrow between two places on one line is drawn straight
71     *     $diagram = new \App\Reporter\Diagram\Diagram();
72     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('a'));
73     *     $diagram->add(new \App\Reporter\Diagram\DiagramNode('b'));
74     *     $diagram->relate(new \App\Reporter\Diagram\DiagramEdge('a', 'b'));
75     *     $layout = \App\Reporter\Diagram\Layout\LayeredLayout::of($diagram);
76     *     count(\App\Reporter\Diagram\Layout\LaneRouting::of($layout, 0)->straight) // => 1
77     *
78     * @return self The routes
79     */
80    public static function of(DiagramLayout $layout, int $layer): self
81    {
82        $links = $layout->linksFrom($layer);
83        $leaving = array_count_values(array_column($links, 0));
84        $arriving = array_count_values(array_column($links, 1));
85        $beside = self::beside($layout, $layer + 1);
86        $sends = [];
87        $receives = [];
88        $crossings = [];
89        $straight = [];
90        foreach ($links as [$origin, $target]) {
91            $row = $layout->rows[$origin];
92            $sends[$origin] ??= ['targets' => [], 'crossings' => []];
93            $receives[$target] ??= ['origins' => [], 'crossings' => []];
94            if ($arriving[$target] === 1 && $leaving[$origin] === 1 && $row === $layout->rows[$target]) {
95                $straight[] = [$origin, $target];
96            } elseif ($arriving[$target] === 1) {
97                $sends[$origin]['targets'][] = $target;
98            } elseif ($leaving[$origin] === 1 && ($beside[$row] ?? $target) === $target) {
99                $receives[$target]['origins'][] = $origin;
100            } else {
101                $crossing = self::spare($row, $layout->rows[$target], $crossings);
102                $crossings[] = [$origin, $target, $crossing];
103                $sends[$origin]['crossings'][] = $crossing;
104                $receives[$target]['crossings'][] = $crossing;
105            }
106        }
107
108        return self::assemble($layout->rows, $sends, $receives, $crossings, $straight);
109    }

133    /**
134     * Returns an odd line no other arrow crosses over on, as near the place arrived at as can be.
135     *
136     * @param int                              $from      The line of the place the arrow leaves
137     * @param int                              $to        The line of the place it arrives at
138     * @param list<array{string, string, int}> $crossings The arrows already given a line
139     *
140     * @example The line just above the place arrived at, when coming from above
141     *     \App\Reporter\Diagram\Layout\LaneRouting::spare(0, 4, []) // => 3
142     * @example The next one along, when that is taken
143     *     \App\Reporter\Diagram\Layout\LaneRouting::spare(0, 4, [['a', 'b', 3]]) // => 5
144     *
145     * @return int The line
146     */
147    public static function spare(int $from, int $to, array $crossings): int
148    {
149        $taken = array_column($crossings, 2);
150        $toward = $from < $to ? -1 : 1;
151        for ($distance = 1;; $distance += 2) {
152            foreach ([$to + $toward * $distance, $to - $toward * $distance] as $line) {
153                if (!in_array($line, $taken, true)) {
154                    return $line;
155                }
156            }
157        }
158    }

160    /**
161     * Makes the lanes out of what each place sends and receives, and puts them in order.
162     *
163     * The lanes where arrows fan out all go left of the lanes where they gather, so
164     * that an arrow going over from one to the other always goes right.
165     *
166     * @param array<string, int>                                                $rows      The line of every place
167     * @param array<string, array{targets: list<string>, crossings: list<int>}> $sends     What each place sends along a lane: the places it arrives at, and the lines it goes over on
168     * @param array<string, array{origins: list<string>, crossings: list<int>}> $receives  What each place receives along a lane: the places it comes from, and the lines it comes over on
169     * @param list<array{string, string, int}>                                  $crossings The arrows that go over, by the places they join and the line they cross on
170     * @param list<array{string, string}>                                       $straight  The arrows drawn straight across
171     *
172     * @example A place that fans out and one that gathers are joined by a crossing
173     *     $routing = \App\Reporter\Diagram\Layout\LaneRouting::assemble(
174     *         ['a' => 0, 'b' => 4],
175     *         ['a' => ['targets' => [], 'crossings' => [3]]],
176     *         ['b' => ['origins' => [], 'crossings' => [3]]],
177     *         [['a', 'b', 3]],
178     *         [],
179     *     );
180     *     $routing->transfers // => [[0, 1, 3]]
181     *
182     * @return self The routes
183     */
184    public static function assemble(array $rows, array $sends, array $receives, array $crossings, array $straight): self
185    {
186        $fans = [];
187        foreach ($sends as $origin => $sent) {
188            if ($sent['targets'] !== [] || $sent['crossings'] !== []) {
189                $fans[] = new Lane([$origin], $sent['targets'], [], $sent['crossings']);
190            }
191        }
192        [$fans, $gathers, $crossings] = self::untangle(self::arrange($fans, $rows), $rows, $crossings);
193        foreach ($receives as $target => $received) {
194            if ($received['origins'] !== [] || $received['crossings'] !== []) {
195                $gathers[] = new Lane($received['origins'], [$target], $received['crossings']);
196            }
197        }
198        $lanes = [...$fans, ...self::arrange($gathers, $rows)];
199
200        $transfers = [];
201        foreach ($crossings as [, , $line]) {
202            $transfers[] = [self::laneWith($lanes, $line, true), self::laneWith($lanes, $line, false), $line];
203        }
204
205        return new self($lanes, $straight, $transfers);
206    }

208    /**
209     * Sends an arrow that would share a stretch of line with another over a line of its own.
210     *
211     * An arrow leaving a fanning lane for the place on some line runs right along that
212     * line; if an arrow enters a fanning lane further right on the same line, the two
213     * share the stretch between the lanes and read as one. Ordering the lanes avoids
214     * that when it can, and cannot when two lanes each leave on the line the other
215     * enters on. Such an arrow is sent over on an odd line of its own instead, into a
216     * lane of its own right of every fanning lane, which no arrow entering a fanning
217     * lane ever reaches.
218     *
219     * @param list<Lane>                       $fans      The fanning lanes, in order
220     * @param array<string, int>               $rows      The line of every place
221     * @param list<array{string, string, int}> $crossings The arrows already going over
222     *
223     * @example An arrow leaving on the line another enters on, further right, goes over instead
224     *     $left = new \App\Reporter\Diagram\Layout\Lane(['a'], ['c']);
225     *     $right = new \App\Reporter\Diagram\Layout\Lane(['b'], ['d']);
226     *     [$fans, $gathers] = \App\Reporter\Diagram\Layout\LaneRouting::untangle([$left, $right], ['a' => 0, 'b' => 2, 'c' => 2, 'd' => 4], []);
227     *     $gathers[0]->targets // => ['c']
228     *
229     * @return array{list<Lane>, list<Lane>, list<array{string, string, int}>} The fanning lanes, the lanes the arrows sent over arrive in, and every arrow going over
230     */
231    public static function untangle(array $fans, array $rows, array $crossings): array
232    {
233        $entering = [];
234        foreach ($fans as $index => $lane) {
235            foreach ($lane->entries($rows) as $row) {
236                $entering[$row] = $index;
237            }
238        }
239
240        $untangled = [];
241        $gathers = [];
242        foreach ($fans as $index => $lane) {
243            $kept = [];
244            $departures = $lane->departures;
245            foreach ($lane->targets as $target) {
246                if (($entering[$rows[$target]] ?? -1) <= $index) {
247                    $kept[] = $target;
248
249                    continue;
250                }
251                $line = self::spare($rows[$lane->origins[0]], $rows[$target], $crossings);
252                $crossings[] = [$lane->origins[0], $target, $line];
253                $departures[] = $line;
254                $gathers[] = new Lane([], [$target], [$line]);
255            }
256            $untangled[] = new Lane($lane->origins, $kept, $lane->arrivals, $departures);
257        }
258
259        return [$untangled, $gathers, $crossings];
260    }

262    /**
263     * Returns the lane an arrow going over on a line leaves, or the one it arrives in.
264     *
265     * Every arrow going over between two columns has a line of its own, so the line is
266     * enough to find both of its lanes.
267     *
268     * @param list<Lane> $lanes   The lanes, left to right
269     * @param int        $line    The line the arrow goes over on
270     * @param bool       $leaving Whether to find the lane it leaves rather than the one it arrives in
271     *
272     * @example The lane an arrow leaves is the one it departs on that line from
273     *     $lanes = [new \App\Reporter\Diagram\Layout\Lane(['a'], [], [], [3]), new \App\Reporter\Diagram\Layout\Lane([], ['b'], [3])];
274     *     \App\Reporter\Diagram\Layout\LaneRouting::laneWith($lanes, 3, true) // => 0
275     *
276     * @return int Where the lane stands, counting from the left from zero
277     */
278    public static function laneWith(array $lanes, int $line, bool $leaving): int
279    {
280        foreach ($lanes as $index => $lane) {
281            if (in_array($line, $leaving ? $lane->departures : $lane->arrivals, true)) {
282                return $index;
283            }
284        }
285
286        return 0;
287    }

289    /**
290     * Puts lanes in the order that costs least, by swapping neighbours while a swap helps.
291     *
292     * What two lanes cost depends only on which of the two is on the left, so each
293     * swap that helps lowers the cost of the whole order, and the swapping ends.
294     *
295     * @param list<Lane>         $lanes The lanes
296     * @param array<string, int> $rows  The line of every place
297     *
298     * @example Two lanes that would cross each other twice are swapped, and cross nothing
299     *     $first = new \App\Reporter\Diagram\Layout\Lane(['a'], ['c']);
300     *     $second = new \App\Reporter\Diagram\Layout\Lane(['b'], ['d']);
301     *     \App\Reporter\Diagram\Layout\LaneRouting::arrange([$first, $second], ['a' => 0, 'b' => 2, 'c' => 4, 'd' => 6]) === [$second, $first] // => true
302     *
303     * @return list<Lane> The lanes, left to right
304     */
305    public static function arrange(array $lanes, array $rows): array
306    {
307        usort($lanes, static fn (Lane $left, Lane $right): int => $left->span($rows) <=> $right->span($rows));
308        for ($pass = 0; $pass < count($lanes); ++$pass) {
309            $swapped = false;
310            for ($index = 0; $index + 1 < count($lanes); ++$index) {
311                if (self::cost($lanes[$index + 1], $lanes[$index], $rows) < self::cost($lanes[$index], $lanes[$index + 1], $rows)) {
312                    array_splice($lanes, $index, 2, [$lanes[$index + 1], $lanes[$index]]);
313                    $swapped = true;
314                }
315            }
316            if (!$swapped) {
317                break;
318            }
319        }
320
321        return $lanes;
322    }

324    /**
325     * Returns what it costs to put one lane left of another.
326     *
327     * An arrow leaving the left lane for the right crosses the right lane where the
328     * right lane runs past that line; an arrow entering the right lane from the left
329     * crosses the left lane where the left lane runs past that line. An arrow leaving
330     * the left lane on a line an arrow enters the right lane on shares a stretch of
331     * line with it, which costs more than any number of crossings.
332     *
333     * @param Lane               $left  The lane on the left
334     * @param Lane               $right The lane on the right
335     * @param array<string, int> $rows  The line of every place
336     *
337     * @example An arrow leaving on the line another enters on shares a stretch of line with it
338     *     $left = new \App\Reporter\Diagram\Layout\Lane(['a'], ['c']);
339     *     $right = new \App\Reporter\Diagram\Layout\Lane(['b'], ['d']);
340     *     \App\Reporter\Diagram\Layout\LaneRouting::cost($left, $right, ['a' => 0, 'b' => 2, 'c' => 2, 'd' => 4]) // => 1000
341     * @example Lanes that never meet cost nothing
342     *     $left = new \App\Reporter\Diagram\Layout\Lane(['a'], ['c']);
343     *     $right = new \App\Reporter\Diagram\Layout\Lane(['b'], ['d']);
344     *     \App\Reporter\Diagram\Layout\LaneRouting::cost($left, $right, ['a' => 0, 'b' => 4, 'c' => 2, 'd' => 6]) // => 0
345     *
346     * @return int The cost
347     */
348    public static function cost(Lane $left, Lane $right, array $rows): int
349    {
350        [$leftTop, $leftBottom] = $left->span($rows);
351        [$rightTop, $rightBottom] = $right->span($rows);
352        $leaving = $left->exits($rows);
353        $entering = $right->entries($rows);
354        $arriving = $right->exits($rows);
355        $cost = 0;
356        foreach ($leaving as $row) {
357            $shared = in_array($row, $entering, true);
358            $crossed = !$shared && !in_array($row, $arriving, true) && $row > $rightTop && $row < $rightBottom;
359            $cost += ($shared ? self::SHARED_LINE : 0) + ($crossed ? 1 : 0);
360        }
361        foreach ($entering as $row) {
362            $cost += $row > $leftTop && $row < $leftBottom && !in_array($row, $leaving, true) ? 1 : 0;
363        }
364
365        return $cost;
366    }
