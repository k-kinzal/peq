14/**
15 * Writes the paths a query bound as one tree.
16 *
17 * Paths are how a query answers "how does this reach that", and a list of them
18 * repeats their shared beginnings once per path — which, for a controller that
19 * reaches forty things through the same two calls, is most of the answer. Drawing
20 * them as a tree writes each shared beginning once, which is the same reason peq's
21 * own inspection draws a tree.
22 *
23 * The difference is what decides the shape. The inspection walks whatever the graph
24 * holds; this draws whatever the query chose to bind, so the tree is of the reader's
25 * own question rather than of the whole graph.
26 *
27 * A nonempty answer without paths cannot be drawn as a tree. Report that mismatch
28 * so a reader can bind and return a path or choose a table for the existing result.
29 *
30 * @visibility App\Reporter
31 */
32final class TreeWriter implements QueryReporter
33{
34    /**
35     * Writes the paths the answer holds as a tree.
36     *
37     * @param ResultTable     $result What the query answered
38     * @param OutputInterface $output Where it is written
39     *
40     * @throws QueryOutputException When a nonempty result contains no paths
41     */
42    #[Override]
43    public function report(ResultTable $result, OutputInterface $output): void
44    {
45        $paths = self::paths($result);
46        if ($paths === []) {
47            if ($result->rows !== []) {
48                throw new QueryOutputException('Tree output requires paths, but the result contains none. Bind and return a path (for example, MATCH p = (a)-->(b) RETURN p), or use --output=table or --output=json.');
49            }
50
51            return;
52        }
53
54        $previous = [];
55        $continues = [];
56        foreach ($paths as $place => $path) {
57            $shared = self::sharedLength($previous, $path);
58            for ($depth = $shared; $depth < count($path); ++$depth) {
59                $continues[$depth] = self::hasSibling($paths, $place, $depth);
60                $output->writeln(self::line($path, $depth, $continues), OutputInterface::OUTPUT_RAW);
61            }
62            $previous = $path;
63        }
64    }

66    /**
67     * Returns the paths an answer holds, in the order a tree draws them.
68     *
69     * They are sorted so that paths sharing a beginning stand together, which is what
70     * lets the tree be drawn in one pass without building one.
71     *
72     * @param ResultTable $result What the query answered
73     *
74     * @example An answer that binds no path holds no paths
75     *     \App\Reporter\Query\TreeWriter::paths(\App\Gql\Result\ResultTable::nothing()) // => []
76     *
77     * @return list<list<TreeStep>> The paths, sorted and without repetition
78     */
79    public static function paths(ResultTable $result): array
80    {
81        $found = [];
82        foreach ($result->rows as $row) {
83            foreach (ResultElements::flattened($row->values) as $value) {
84                if ($value instanceof PathDatum) {
85                    $steps = self::steps($value);
86                    $found[self::signature($steps)] = $steps;
87                }
88            }
89        }
90        ksort($found);
91
92        return array_values($found);
93    }

95    /**
96     * Returns one path as the steps a tree draws it in.
97     *
98     * @param PathDatum $path The path
99     *
100     * @example A path that crosses nothing is one step
101     *     count(\App\Reporter\Query\TreeWriter::steps(\App\Gql\Datum\PathDatum::at(new \App\Gql\Datum\NodeDatum('a')))) // => 1
102     * @example A path that crosses one relation is two
103     *     $path = \App\Gql\Datum\PathDatum::at(new \App\Gql\Datum\NodeDatum('a'))->continuedBy(new \App\Gql\Datum\EdgeDatum('e', ['calls'], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('b'));
104     *     count(\App\Reporter\Query\TreeWriter::steps($path)) // => 2
105     * @example A relation crossed against its direction is a step taken backwards
106     *     $path = \App\Gql\Datum\PathDatum::at(new \App\Gql\Datum\NodeDatum('b'))->continuedBy(new \App\Gql\Datum\EdgeDatum('e', ['calls'], [], 'a', 'b'), new \App\Gql\Datum\NodeDatum('a'));
107     *     \App\Reporter\Query\TreeWriter::steps($path)[1]->backwards // => true
108     *
109     * @return list<TreeStep> The steps, beginning where the path begins
110     */
111    public static function steps(PathDatum $path): array
112    {
113        $steps = [];
114        $crossed = null;
115        foreach ($path->elements as $element) {
116            if ($element instanceof EdgeDatum) {
117                $crossed = $element;
118
119                continue;
120            }
121            if ($element instanceof NodeDatum) {
122                $steps[] = $crossed === null
123                    ? new TreeStep('', $element->id)
124                    : new TreeStep($crossed->label(), $element->id, $crossed->origin === $element->id && $crossed->target !== $element->id);
125                $crossed = null;
126            }
127        }
128
129        return $steps;
130    }

132    /**
133     * Returns what tells one path apart from another, and sorts them sensibly.
134     *
135     * @param list<TreeStep> $steps The steps of the path
136     *
137     * @example A path with no steps has nothing to tell it apart
138     *     \App\Reporter\Query\TreeWriter::signature([]) // => ''
139     *
140     * @return string What identifies the path
141     */
142    public static function signature(array $steps): string
143    {
144        return implode("\1", array_map(static fn (TreeStep $step): string => $step->key(), $steps));
145    }

147    /**
148     * Returns how many steps two paths begin with in common.
149     *
150     * @param list<TreeStep> $left  The path drawn before
151     * @param list<TreeStep> $right The path being drawn
152     *
153     * @example A path shares nothing with nothing
154     *     \App\Reporter\Query\TreeWriter::sharedLength([], [new \App\Reporter\Query\TreeStep('', 'a')]) // => 0
155     * @example Two paths from the same symbol share their first step
156     *     $left = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'b')];
157     *     $right = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'c')];
158     *     \App\Reporter\Query\TreeWriter::sharedLength($left, $right) // => 1
159     *
160     * @return int How many steps they share
161     */
162    public static function sharedLength(array $left, array $right): int
163    {
164        $shared = 0;
165        $shortest = min(count($left), count($right));
166        while ($shared < $shortest && $left[$shared]->key() === $right[$shared]->key()) {
167            ++$shared;
168        }
169
170        return $shared;
171    }

173    /**
174     * Reports whether another path branches off beside this one at a depth.
175     *
176     * A branch is drawn as closed or open depending on whether anything follows it at
177     * the same depth, and what follows it is a later path that shares everything above
178     * that depth and differs at it.
179     *
180     * @param list<list<TreeStep>> $paths The paths, sorted
181     * @param int                  $place Which path is being drawn
182     * @param int                  $depth Which step of it
183     *
184     * @example The only path there is has nothing beside it
185     *     $paths = [[new \App\Reporter\Query\TreeStep('', 'a')]];
186     *     \App\Reporter\Query\TreeWriter::hasSibling($paths, 0, 0) // => false
187     * @example A path that differs at a depth stands beside the one before it
188     *     $first = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'b')];
189     *     $second = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'c')];
190     *     \App\Reporter\Query\TreeWriter::hasSibling([$first, $second], 0, 1) // => true
191     *
192     * @return bool True when a later path branches off at that depth
193     */
194    public static function hasSibling(array $paths, int $place, int $depth): bool
195    {
196        for ($later = $place + 1; $later < count($paths); ++$later) {
197            $shared = self::sharedLength($paths[$place], $paths[$later]);
198            if ($shared === $depth) {
199                return true;
200            }
201            if ($shared < $depth) {
202                return false;
203            }
204        }
205
206        return false;
207    }

209    /**
210     * Writes one step of one path.
211     *
212     * @param list<TreeStep>   $path      The path
213     * @param int              $depth     Which step of it
214     * @param array<int, bool> $continues Whether the branch at each depth has more beside it
215     *
216     * @example The start of a path carries no branch drawing
217     *     \App\Reporter\Query\TreeWriter::line([new \App\Reporter\Query\TreeStep('', 'a')], 0, []) // => 'a'
218     * @example A step is drawn as the relation that reached it
219     *     $path = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'b')];
220     *     \App\Reporter\Query\TreeWriter::line($path, 1, [1 => false]) // => '└── calls ──> b'
221     * @example A relation followed backwards points back, so that `a` is read as what `b` calls
222     *     $path = [new \App\Reporter\Query\TreeStep('', 'a'), new \App\Reporter\Query\TreeStep('calls', 'b', true)];
223     *     \App\Reporter\Query\TreeWriter::line($path, 1, [1 => false]) // => '└── calls <── b'
224     *
225     * @return string The line, without a trailing newline
226     */
227    public static function line(array $path, int $depth, array $continues): string
228    {
229        $step = $path[$depth];
230        if ($depth === 0) {
231            return $step->id;
232        }
233
234        $line = '';
235        for ($above = 1; $above < $depth; ++$above) {
236            $line .= ($continues[$above] ?? false) ? '│   ' : '    ';
237        }
238
239        return $line.(($continues[$depth] ?? false) ? '├── ' : '└── ').$step->label.($step->backwards ? ' <── ' : ' ──> ').$step->id;
240    }
