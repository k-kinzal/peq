27     * @throws InspectionException If the requested analysis or encoding is rejected
28     */
29    public function inspect(SourceIndex $index, string $target): DependencyGraph
30    {
31        if (str_contains($target, '::$')) {
32            return (new Properties())->inspect($index, $target);
33        }
34        $found = null;
35        foreach ($index->sources() as $source) {
36            foreach ($this->callables($source->statements) as $name => $callable) {
37                if (strcasecmp(ltrim($target, '\\'), $name) === 0) {
38                    if ($found !== null) {
39                        throw new InspectionException('Ambiguous callable: '.$target.'. Analyze a narrower path.');
40                    }
41                    $text = file_get_contents($source->path);
42                    if ($text !== false) {
43                        $found = $this->analyze($callable, new DependencyGraph($name, $source->path, $text));
44                    }
45                }
46            }
47        }
48        if ($found === null) {
49            throw new InspectionException('Callable not found: '.$target.'. Select a method or function declared in a readable, valid PHP source file.');
50        }
51
52        return $found;
53    }

60    public function callables(array $statements): array
61    {
62        $callables = [];
63        foreach ((new NodeFinder())->find($statements, static fn (Node $node): bool => $node instanceof ClassLike || $node instanceof Function_) as $node) {
64            if ($node instanceof ClassLike && $node->namespacedName !== null) {
65                foreach ($node->getMethods() as $method) {
66                    if ($method->stmts !== null) {
67                        $callables[$node->namespacedName->toString().'::'.$method->name->toString()] = $method;
68                    }
69                }
70            } elseif ($node instanceof Function_ && $node->namespacedName !== null) {
71                $callables[$node->namespacedName->toString()] = $node;
72            }
73        }
74
75        return $callables;
76    }

87        $recording = new Recording($graph);
88        $state = new State();
89        foreach ((new NodeFinder())->findInstanceOf($body, Variable::class) as $variable) {
90            if (is_string($variable->name)) {

92                if (!isset($state->definitions[$name])) {

96                }
97            }
98        }
99        foreach ($callable->params as $parameter) {
100            if ($parameter->var instanceof Variable) {
101                $id = $recording->write($parameter->var, [], $state, 'parameter');
102                $graph->scalars[$id] = !$parameter->variadic && \App\Analyzer\ExperimentAnalyzer\Resolution\ScalarOrigins::parameter($parameter->type);
103            }
104        }
105        $unstructured = (new NodeFinder())->findFirst($body, static fn (Node $node): bool => $node instanceof Node\Stmt\Goto_ || $node instanceof Node\Stmt\Label);
106        $aliases = array_filter($callable->params, static fn (Node\Param $parameter): bool => $parameter->byRef);
107        if ($unstructured !== null || $aliases !== []) {
108            (new \App\Analyzer\ExperimentAnalyzer\Resolution\Boundary($graph))->read($callable, $state, 'NONLOCAL_FLOW', 'Jump targets or aliased parameters require a whole-callable model.');
109        } else {

111        }
112
