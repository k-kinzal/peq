14/**
15 * Writes a query's answer as a JSON document.
16 *
17 * This is the form written for a program, and for the reader peq's query language
18 * exists for: an agent that asked a question and has to decide what to do next. It
19 * carries the status first, because the standard says a status is the part a program
20 * is allowed to test for, and because "this found nothing" and "this could not be
21 * read" need different next moves.
22 *
23 * Symbols and relations serialise as objects rather than as names, so an agent that
24 * matched a pattern gets the labels and properties it selected by without having to
25 * ask a second query for them.
26 *
27 * @visibility App\Reporter
28 */
29final class JsonWriter implements QueryReporter
30{
31    /**
32     * Writes the answer as JSON.
33     *
34     * @param ResultTable     $result What the query answered
35     * @param OutputInterface $output Where it is written
36     */
37    #[Override]
38    public function report(ResultTable $result, OutputInterface $output): void
39    {
40        $output->writeln(self::document($result), OutputInterface::OUTPUT_RAW);
41    }
42
43    /**
44     * Writes the whole answer as one JSON document.
45     *
46     * @param ResultTable $result What the query answered
47     *
48     * @example An answer carries its status, its columns and its rows
49     *     $row = \App\Gql\Binding\BindingRow::unit()->with('n', new \App\Gql\Datum\IntegerDatum(1));
50     *     $result = \App\Gql\Result\ResultTable::of(['n'], new \App\Gql\Binding\BindingTable([$row]));
51     *     \App\Reporter\Query\JsonWriter::document($result) // => '{"status":"00000","condition":"note: successful completion","columns":[{"name":"n","type":"INT64"}],"rows":[[1]]}'
52     * @example An answer that found nothing says so rather than failing
53     *     \App\Reporter\Query\JsonWriter::document(\App\Gql\Result\ResultTable::nothing()) // => '{"status":"02000","condition":"note: no data","columns":[],"rows":[]}'
54     *
55     * @return string The document
56     */
57    public static function document(ResultTable $result): string
58    {
59        $status = $result->status();
60
61        return '{"status":'.DatumJson::text($status->value)
62            .',"condition":'.DatumJson::text($status->condition())
63            .',"columns":['.implode(',', array_map(self::column(...), $result->columns))
64            .'],"rows":['.implode(',', array_map(self::row(...), $result->rows)).']}';
65    }
66
67    /**
68     * Writes one column heading as JSON.
69     *
70     * @param ResultColumn $column The column
71     *
72     * @example A column says what it is called and what it holds
73     *     \App\Reporter\Query\JsonWriter::column(new \App\Gql\Result\ResultColumn('n', 'INT64')) // => '{"name":"n","type":"INT64"}'
74     *
75     * @return string The column, written as JSON
76     */
77    public static function column(ResultColumn $column): string
78    {
79        return '{"name":'.DatumJson::text($column->heading).',"type":'.DatumJson::text($column->type).'}';
80    }
81
82    /**
83     * Writes one row as JSON.
84     *
85     * A row is an array rather than an object, because the columns are already named
86     * once for the whole table and naming them again on every row would multiply the
87     * size of an answer by the length of its headings.
88     *
89     * @param ResultRow $row The row
90     *
91     * @example A row is the values of its columns, in order
92     *     \App\Reporter\Query\JsonWriter::row(new \App\Gql\Result\ResultRow([new \App\Gql\Datum\IntegerDatum(1)])) // => '[1]'
93     *
94     * @return string The row, written as JSON
95     */
96    public static function row(ResultRow $row): string
97    {
98        return '['.implode(',', array_map(static fn (Datum $value): string => DatumJson::of($value), $row->values)).']';
99    }
100}
