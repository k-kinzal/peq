7/**
8 * Every word GQL reserves, and the rule that makes reserving them matter.
9 *
10 * ISO/IEC 39075 writes a regular identifier and a keyword out of the same characters
11 * and tells the two apart with a single syntax rule: a regular identifier is not a
12 * reserved word. Without that rule `MATCH (match)` has two readings and a grammar has
13 * to guess between them; with it, a query means one thing.
14 *
15 * The price is that a reserved word cannot be a name written plainly. GQL's own answer
16 * is the delimited identifier — `value` in back quotes is a name whatever it spells —
17 * so nothing becomes unsayable, only differently said. That is why this list narrows
18 * what peq reads rather than what a query can ask about.
19 *
20 * The standard lists the words in two productions and reserves both: `<reserved word>`,
21 * and the `<pre-reserved word>`s held back for editions to come. The second is why
22 * `Function` needs its back quotes although nothing in GQL uses the word yet.
23 *
24 * What follows is those two productions of ISO's published grammar artifact, in the
25 * order the artifact writes them. `spec/features/lexical-elements.feature` fails the
26 * build if it ever stops being exactly that.
27 *
28 * @see https://www.iso.org/standard/76120.html ISO/IEC 39075:2024, subclause 21.1
29 *
30 * @visibility App\Gql
31 */

304    /**
305     * Reports whether a word is one GQL reserves.
306     *
307     * A reserved word is recognised however it is cased, since GQL compares a regular
308     * identifier against a keyword without regard to case.
309     *
310     * @param string $word The word, as it was written
311     *
312     * @example A word the grammar uses is reserved
313     *     \App\Gql\ReservedWords::reserves('MATCH') // => true
314     * @example However it is cased
315     *     \App\Gql\ReservedWords::reserves('match') // => true
316     * @example A word held back for a later edition is reserved now
317     *     \App\Gql\ReservedWords::reserves('function') // => true
318     * @example Anything else is a name
319     *     \App\Gql\ReservedWords::reserves('firstName') // => false
320     *
321     * @return bool True when GQL reserves it
322     */
323    public static function reserves(string $word): bool
324    {
325        return in_array(strtoupper($word), self::WORDS, true);
326    }
327
328    /**
329     * Returns a name the way a query has to write it.
330     *
331     * A graph is named by whoever wrote the code it was read from, and GQL's reserved
332     * words were not consulted. Anything peq tells a reader a label or a property is
333     * called therefore has to come back as something they can paste into a query, or
334     * the answer to "what can I ask about?" is a query that does not parse.
335     *
336     * @param string $name The name, as the graph carries it
337     *
338     * @example A name GQL leaves free is written as it is
339     *     \App\Gql\ReservedWords::asWritten('Method') // => 'Method'
340     * @example A name GQL reserves is written in back quotes
341     *     \App\Gql\ReservedWords::asWritten('Function') // => '`Function`'
342     *
343     * @return string The name, as an identifier
344     */
345    public static function asWritten(string $name): string
346    {
347        return self::reserves($name) ? '`'.$name.'`' : $name;
348    }
