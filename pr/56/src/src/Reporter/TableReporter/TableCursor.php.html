14/**
15 * The rows one table report is accumulating as the walk goes on.
16 *
17 * A table says the same things a tree line says, in columns instead of in prefixes
18 * and suffixes. Two of them survive the change of medium unaltered: how deep a symbol
19 * sits, and what it is called. Two are things the tree can only hint at — the kind of
20 * symbol, which the tree writes as a parenthesis for the two kinds it has a word for,
21 * and where the symbol is written, which the tree has no room for at all.
22 *
23 * What does not survive is the shape. A table is a list, and a list is what it is for:
24 * something to sort, grep, paste into a review and count the length of.
25 *
26 * @visibility namespace
27 */
28final class TableCursor
29{
30    /**
31     * Written in a cell the walk has nothing to put in.
32     */
33    private const ABSENT = '-';
34
35    /**
36     * @var list<list<string>> One row per node the walk reached, in the order it reached them
37     */
38    private array $rows = [];
39
40    /** @var array<int, Node> */
41    private array $parents = [];
42
43    /**
44     * How far this report has already expanded the graph.
45     */
46    private readonly Expansion $expansion;
47
48    /**
49     * @param null|int $level Deepest level to report, or null for the whole graph
50     */
51    public function __construct(?int $level = null, private readonly ?Graph $graph = null, private readonly Direction $direction = Direction::Uses)
52    {
53        $this->expansion = new Expansion($level);
54    }
55
56    /**
57     * Records one node as a row and reports whether the walk should continue below it.
58     *
59     * A branch the walk cuts is named in the symbol cell, the way the tree names it
60     * at the end of a line. Without that, a row with nothing under it would read as a
61     * symbol that depends on nothing, which is a different statement.
62     *
63     * @param Node $node  The node the traversal reached
64     * @param int  $depth How far below the root symbol it sits
65     *
66     * @return bool True when the traversal should descend into this node
67     */
68    public function visit(Node $node, int $depth): bool
69    {
70        $continuation = $this->expansion->reach($node, $depth);
71        if (!$continuation->reported()) {
72            return false;
73        }
74
75        $marker = $continuation->marker();
76        $meta = $node->meta();
77
78        $possible = $this->graph !== null && RelationNotice::possible($this->graph, $depth > 0 ? ($this->parents[$depth - 1] ?? null) : null, $node, $this->direction);
79        $label = $node->id()->toString().($possible ? ' (possible)' : '');
80        $parent = $depth > 0 ? ($this->parents[$depth - 1] ?? null) : null;
81        $calls = $this->graph === null ? [] : CallOccurrences::between($this->graph, $parent?->id(), $node->id(), $this->direction);
82        $label .= $calls === [] ? '' : ' ['.implode('; ', array_map(CallOccurrences::text(...), $calls)).']';
83        $this->parents[$depth] = $node;
84        $this->rows[] = [
85            (string) $depth,
86            $marker === null ? $label : sprintf('%s (%s)', $label, $marker),
87            $node->kind()->value,
88            $meta === null ? self::ABSENT : sprintf('%s:%d', $meta->path, $meta->line),
89        ];
90
91        return $continuation->descends();
92    }
93
94    /**
95     * Returns the rows written for the nodes the walk reached.
96     *
97     * @return list<list<string>> The rows, in the order the walk reached them
98     */
99    public function rows(): array
100    {
101        return $this->rows;
102    }
103}
