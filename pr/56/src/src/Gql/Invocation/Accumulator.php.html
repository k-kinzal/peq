10/**
11 * Something that summarises a run of values into one.
12 *
13 * Aggregates are written as function calls and behave like nothing else in the
14 * language: they see every row of a group rather than one row, and they are the only
15 * expression whose answer depends on rows other than the one being evaluated. Giving
16 * them a shape of their own — offered values one at a time, asked for an answer once —
17 * is what lets the same six summaries be used over a whole table, over a group, and
18 * over the list of edges a variable-length pattern bound.
19 *
20 * Every one of them ignores values that are not there, as GQL requires, and answers
21 * with nothing when it was offered nothing material. Counting is the exception the
22 * standard makes: nothing counted is zero, not nothing.
23 */
24interface Accumulator
25{
26    /**
27     * Offers one value to the summary.
28     *
29     * @param Datum $value The value
30     *
31     * @throws GqlException If the value is not one this summary can take
32     */
33    public function accept(Datum $value): void;
34
35    /**
36     * Returns what the values offered so far summarise to.
37     *
38     * @return Datum The summary
39     */
40    public function result(): Datum;
41}
