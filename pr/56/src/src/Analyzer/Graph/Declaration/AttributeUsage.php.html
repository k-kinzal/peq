5namespace App\Analyzer\Graph\Declaration;

7/**
8 * One attribute written on a declaration.
9 *
10 * The graph already records an attribute as a relation, because writing one is a
11 * dependency on the class it names. That relation answers "what breaks if this
12 * attribute class changes"; it does not answer "which methods are routes", which is
13 * the question a framework-shaped codebase actually asks. Recording the attribute on
14 * the declaration as well is what makes the second question one lookup instead of a
15 * walk through the relations of every candidate.
16 *
17 * The arguments are kept as they are written, not evaluated. `#[Route('/users')]` is
18 * recognised by the text between its parentheses, and evaluating it would need a
19 * runtime this analysis deliberately does not have.
20 */
21final readonly class AttributeUsage
22{
23    /**
24     * @param string       $name      The fully qualified name of the attribute class
25     * @param list<string> $arguments The arguments as they are written, in source order
26     */
27    public function __construct(
28        public string $name,
29        public array $arguments = [],
30    ) {
31        assert($this->name !== '', 'An attribute must name a class');
32    }
33
34    /**
35     * Writes the attribute the way its declaration writes it.
36     *
37     * @example An attribute with arguments reads as it was written
38     *     (new \App\Analyzer\Graph\Declaration\AttributeUsage('Route', ["'/users'"]))->toString() // => "Route('/users')"
39     * @example An attribute without arguments carries no parentheses
40     *     (new \App\Analyzer\Graph\Declaration\AttributeUsage('Override'))->toString() // => 'Override'
41     *
42     * @return string The attribute as source code writes it, without its leading marker
43     */
44    public function toString(): string
45    {
46        if ($this->arguments === []) {
47            return $this->name;
48        }
49
50        return $this->name.'('.implode(', ', $this->arguments).')';
51    }
52}
