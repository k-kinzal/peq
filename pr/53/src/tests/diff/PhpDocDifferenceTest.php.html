37    #[DataProvider('providerSupportedVersions')]
38    public function testReferencesRetainTheirOwnersAndLocations(int $version): void
39    {
40        $source = <<<'PHP'
41            <?php
42            namespace Doc;
43            class Item {}
44            /** @mixin Item */
45            class Subject {
46                /** @var Item */
47                public $first, $second;
48                /** @return Item */
49                public function run() {
50                    /** @var Item $value */
51                    $value = null;
52                }
53            }
54            PHP;
55        $directory = sys_get_temp_dir().'/peq-phpdoc-owners-'.$version;
56        is_dir($directory) || mkdir($directory, 0o777, true);
57        file_put_contents($directory.'/source.php', $source);
58        $graph = (new PhpStanAnalyzer(phpVersion: $version))->analyze($directory);
59        $references = array_filter($graph->forwardEdges(), static fn (Edge $edge): bool => $edge->kind() === EdgeKind::PhpDoc);
60        $locations = array_map(static fn (Edge $edge): array => [$edge->from()->toString(), $edge->to()->toString(), $edge->meta()->line], $references);
61        sort($locations);
62        $native = new Process([PHP_BINARY, '-r', <<<'PHP'
63            require $argv[1];
64            $graph = (new App\Analyzer\NativeAnalyzer\NativeAnalyzer(phpVersion: (int) $argv[3]))->analyze($argv[2]);
65            echo App\Analyzer\Graph\GraphSnapshot::of($graph)->toString();
66            PHP, dirname(__DIR__, 2).'/vendor/autoload.php', $directory, (string) $version]);
67        $native->mustRun();
68
69        self::assertSame([
70            ['Doc\Subject', 'Doc\Item', 4],
71            ['Doc\Subject::first', 'Doc\Item', 6],
72            ['Doc\Subject::run', 'Doc\Item', 8],
73            ['Doc\Subject::run', 'Doc\Item', 10],
74            ['Doc\Subject::second', 'Doc\Item', 6],
75        ], $locations);
76        self::assertCount(5, $graph->edges(\App\Analyzer\Graph\NodeId\ClassNodeId::of('Doc\Item')));
77        self::assertSame(GraphSnapshot::of($graph)->toString(), $native->getOutput());
78    }

109    #[DataProvider('providerSources')]
110    public function testBothEnginesReadTheSameDocumentedSource(string $source, array $expectedTypes): void
111    {
112        $directory = sys_get_temp_dir().'/peq-phpdoc-diff-'.hash('sha256', $source);
113        is_dir($directory) || mkdir($directory, 0o777, true);
114        file_put_contents($directory.'/source.php', $source);
115
116        $referenceGraph = (new PhpStanAnalyzer())->analyze($directory);
117        $reference = GraphSnapshot::of($referenceGraph);
118        $references = array_filter($referenceGraph->forwardEdges(), static fn (Edge $edge): bool => $edge->kind() === EdgeKind::PhpDoc);
119        $types = array_values(array_unique(array_map(static fn (Edge $edge): string => $edge->to()->toString(), $references)));
120        sort($types);
121        $native = new Process([PHP_BINARY, '-r', <<<'PHP'
122            require $argv[1];
123            $graph = (new App\Analyzer\NativeAnalyzer\NativeAnalyzer())->analyze($argv[2]);
124            if (class_exists(PHPStan\DependencyInjection\ContainerFactory::class, false)) {
125                throw new RuntimeException('Native analysis loaded the PHPStan engine.');
126            }
127            echo App\Analyzer\Graph\GraphSnapshot::of($graph)->toString();
128            PHP, dirname(__DIR__, 2).'/vendor/autoload.php', $directory]);
129        $native->mustRun();
130
131        self::assertNotSame([], $reference->nodes);
132        self::assertSame($expectedTypes, $types);
133        self::assertSame($reference->toString(), $native->getOutput());
134    }
