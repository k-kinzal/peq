13/**
14 * Finding every way a whole MATCH matches the graph.
15 *
16 * Paths written side by side are matched one after another, each starting from what
17 * the ones before it bound. That is all the joining there is: a name written in two
18 * paths is bound by the first and constrains the second, so `(p)-[:declaresMethod]->
19 * (m), (p)-[:extends]->(q)` finds a class with both a method and a parent without any
20 * join condition being written or worked out.
21 *
22 * Matching each path against the rows the previous one produced also means the order
23 * paths are written in decides how much work the search does. A selective path
24 * written first narrows everything after it, which is worth knowing when a pattern
25 * turns out to be slow.
26 *
27 * @visibility App\Gql
28 */
29final readonly class PatternMatching
30{
31    /**
32     * Finding every way one path matches.
33     */
34    private PathMatching $paths;
35
36    /**
37     * @param ElementGraph         $graph      The graph being matched against
38     * @param ExpressionEvaluation $evaluation How a predicate written in a pattern is worked out
39     */
40    public function __construct(
41        ElementGraph $graph,
42        ExpressionEvaluation $evaluation,
43    ) {
44        $this->paths = new PathMatching($graph, $evaluation);
45    }
46
47    /**
48     * Returns every way a pattern matches, as rows of what it bound.
49     *
50     * @param GraphPattern $pattern The pattern
51     * @param BindingRow   $row     What is already bound when the pattern is reached
52     *
53     * @example A pattern over a graph with no symbols matches nothing
54     *     $matching = new \App\Gql\Matching\PatternMatching(new \App\Gql\Element\ElementGraph([], [], []), new \App\Gql\Evaluation\ExpressionEvaluation());
55     *     $pattern = new \App\Gql\Syntax\Pattern\GraphPattern([new \App\Gql\Syntax\Pattern\PathPattern([new \App\Gql\Syntax\Pattern\NodePattern('p')])]);
56     *     $matching->match($pattern, \App\Gql\Binding\BindingRow::unit()) // => []
57     *
58     * @return list<BindingRow> One row per way the pattern matched
59     *
60     * @throws GqlException If a requirement written in the pattern cannot be worked out
61     */
62    public function match(GraphPattern $pattern, BindingRow $row): array
63    {
64        $rows = [$row];
65        foreach ($pattern->paths as $path) {
66            $next = [];
67            foreach ($rows as $standing) {
68                array_push($next, ...$this->paths->match($path, $standing));
69            }
70            $rows = $next;
71            if ($rows === []) {
72                return [];
73            }
74        }
75
76        return $rows;
77    }
78}
