44    /**
45     * @param list<string> $expanding Prevents recursive aliases and bounds from looping
46     *
47     * @return list<string>
48     */
49    public static function of(Node $node, DocScope $scope, array $expanding = []): array
50    {
51        if ($node instanceof InvalidTagValueNode || $node instanceof DoctrineTagValueNode) {
52            return [];
53        }
54        if ($node instanceof CallableTypeNode || $node instanceof MethodTagValueNode) {
55            $scope = $scope->withTypes(new PhpDocNode(array_map(static fn (TemplateTagValueNode $template): PhpDocTagNode => new PhpDocTagNode('@template', $template), $node->templateTypes)));
56        }
57        if ($node instanceof GenericTypeNode && strtolower($node->type->name) === 'int') {
58            $bounds = array_filter($node->genericTypes, static fn (TypeNode $type): bool => !$type instanceof IdentifierTypeNode || !in_array($type->name, ['min', 'max'], true));
59
60            return array_values(array_unique(array_merge([], ...array_map(static fn (TypeNode $type): array => self::of($type, $scope, $expanding), $bounds))));
61        }
62        if ($node instanceof IdentifierTypeNode) {
63            return self::identifier($node->name, $scope, $expanding);
64        }
65        if ($node instanceof ThisTypeNode) {
66            return $scope->class === null ? [] : [$scope->class];
67        }
68        if ($node instanceof ConstFetchNode) {
69            return $node->className === '' ? [] : self::identifier($node->className, $scope, $expanding);
70        }
71        if ($node instanceof ArrayShapeItemNode && $node->keyName instanceof ConstFetchNode) {
72            return array_values(array_unique([...self::of($node->keyName, $scope, $expanding), ...self::of($node->valueType, $scope, $expanding)]));
73        }
74        if ($node instanceof ArrayShapeItemNode || $node instanceof ObjectShapeItemNode) {
75            return self::of($node->valueType, $scope, $expanding);
76        }
77        $names = [];
78        foreach (get_object_vars($node) as $child) {
79            foreach (is_array($child) ? $child : [$child] as $value) {
80                if ($value instanceof Node) {
81                    array_push($names, ...self::of($value, $scope, $expanding));
82                }
83            }
84        }
85
86        return array_values(array_unique($names));
87    }

89    /**
90     * @param list<string> $expanding
91     *
92     * @return list<string>
93     */
94    public static function identifier(string $name, DocScope $scope, array $expanding): array
95    {
96        if (array_key_exists($name, $scope->localTypes)) {
97            $type = $scope->localTypes[$name];
98
99            return $type === null || in_array($name, $expanding, true) ? [] : self::of($type, $scope, [...$expanding, $name]);
100        }
101        if (in_array(strtolower($name), self::BUILTINS, true)) {
102            return [];
103        }
104        if (in_array(strtolower($name), ['pure-closure', 'static-closure', 'static-pure-closure'], true)) {
105            return ['Closure'];
106        }
107        if (str_contains($name, '-') && !str_starts_with($name, 'OCI-')) {
108            return [];
109        }
110        $resolved = $scope->resolve($name);
111
112        return $resolved === null ? [] : [$resolved];
113    }
