21/**
22 * The syntax tree of a file as it is written, read again from disk.
23 *
24 * PHPStan hands a collector a cleaned tree: its CleaningVisitor strips expressions
25 * out of method and closure bodies, and it can do so partially, leaving a body that
26 * looks present but has lost the calls and property accesses peq is there to record.
27 * Reading the file again is therefore not an optimisation to be removed — it is the
28 * only way to see the whole body.
29 *
30 * A file is parsed once per instance and kept, because a file holds many methods and
31 * every one of them would otherwise pay for the same parse. The lifetime of that
32 * memory is the lifetime of this object, which is one analysis run.
33 *
34 * @visibility namespace
35 */
36final class ReparsedSource
37{
38    /**
39     * @var array<string, null|list<Stmt>> Statements per file, with null recording a file that could not be read
40     */
41    private array $parsedFiles = [];
42
43    /**
44     * The parser, built on first use because building it is not free.
45     */
46    private ?Parser $parser = null;
47
48    /**
49     * The traverser that resolves names, reused across files.
50     */
51    private ?NodeTraverser $traverser = null;
52
53    /**
54     * The finder used to locate declarations inside a parsed file.
55     */
56    private ?NodeFinder $finder = null;
57
58    /**
59     * Returns the statements of a file, with every name resolved to its full form.
60     *
61     * A file that cannot be read or cannot be parsed yields null, and the failure is
62     * remembered so the same file is not retried for each of its methods.
63     *
64     * @param string $file Absolute path of the file
65     *
66     * @return null|list<Stmt> The parsed statements, or null when the file is unusable
67     */
68    public function statements(string $file): ?array
69    {
70        if (array_key_exists($file, $this->parsedFiles)) {
71            return $this->parsedFiles[$file];
72        }
73
74        $contents = is_readable($file) ? file_get_contents($file) : false;
75        if ($contents === false) {
76            return $this->parsedFiles[$file] = null;
77        }
78
79        $this->parser ??= (new ParserFactory())->createForHostVersion();
80
81        $syntaxErrors = new Collecting();
82        $parsed = $this->parser->parse($contents, $syntaxErrors);
83
84        if ($parsed === null || $syntaxErrors->hasErrors()) {
85            return $this->parsedFiles[$file] = null;
86        }
87
88        if ($this->traverser === null) {
89            $this->traverser = new NodeTraverser();
90            $this->traverser->addVisitor(new NameResolver());
91        }
92
93        $resolved = [];
94        foreach ($this->traverser->traverse($parsed) as $statement) {
95            if ($statement instanceof Stmt) {
96                $resolved[] = $statement;
97            }
98        }
99
100        return $this->parsedFiles[$file] = $resolved;
101    }
102
103    /**
104     * Returns the body of one method as it is written in the file.
105     *
106     * @param string $file       Absolute path of the file the method is declared in
107     * @param string $className  Fully qualified name of the declaring class-like
108     * @param string $methodName Name of the method
109     *
110     * @return null|list<Stmt> The method body, or null when it has none or cannot be located
111     */
112    public function methodBody(string $file, string $className, string $methodName): ?array
113    {
114        $statements = $this->statements($file);
115        if ($statements === null) {
116            return null;
117        }
118
119        $finder = $this->finder ??= new NodeFinder();
120
121        $declaration = $finder->findFirst($statements, static function (PhpParserNode $node) use ($className): bool {
122            return ($node instanceof Class_ || $node instanceof Interface_ || $node instanceof Trait_ || $node instanceof Enum_)
123                && $node->namespacedName !== null
124                && $node->namespacedName->toString() === $className;
125        });
126
127        if (!$declaration instanceof Class_ && !$declaration instanceof Trait_ && !$declaration instanceof Enum_) {
128            return null;
129        }
130
131        $method = $finder->findFirst($declaration->stmts, static function (PhpParserNode $node) use ($methodName): bool {
132            return $node instanceof ClassMethod && $node->name->toString() === $methodName;
133        });
134
135        return $method instanceof ClassMethod && $method->stmts !== null ? array_values($method->stmts) : null;
136    }
137}
