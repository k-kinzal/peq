28/**
29 * Command to inspect the dependency tree of a PHP function, method, or class.
30 *
31 * The command is the input and output boundary. It declares the command line, stacks
32 * the configuration sources, runs the inspection and hands the result to the reporter
33 * the configuration asks for. It decides nothing about analysis or presentation: which
34 * analyzer runs is the use case's decision, and which reporter renders is the reporter
35 * factory's, so this class holds no branch on either.
36 */

79    /**
80     * Runs the inspection and writes its report.
81     *
82     * @param InputInterface  $input  The parsed command line
83     * @param OutputInterface $output Where the report and any error is written
84     *
85     * @return int Command::SUCCESS, Command::INVALID for unusable arguments, or
86     *             Command::FAILURE when the configuration or the symbol is rejected
87     */
88    #[Override]
89    protected function execute(InputInterface $input, OutputInterface $output): int
90    {
91        $memoryLimit = $input->getOption('memory-limit');
92        if (is_string($memoryLimit)) {
93            ini_set('memory_limit', $memoryLimit);
94        }
95
96        $target = $input->getArgument('target');
97        if (!is_string($target)) {
98            $output->writeln('<error>The target must be written as a single symbol name.</error>');
99
100            return Command::INVALID;
101        }
102
103        $configPath = $input->getOption('config');
104        if (!is_string($configPath)) {
105            $output->writeln('<error>The --config option must be a path to a file.</error>');
106
107            return Command::INVALID;
108        }
109
110        try {
111            $config = (new ConfigLoader([
112                new DefaultConfigReader(),
113                new YamlConfigLoader($configPath),
114                new EnvConfigReader(),
115                new InputConfigReader($input),
116            ]))->load();
117
118            $result = $this->action->execute(new InspectActionInput(
119                config: $config,
120                target: $target,
121            ));
122        } catch (ConfigException|SymbolNotFoundException $rejected) {
123            $output->writeln(sprintf('<error>%s</error>', $rejected->getMessage()));
124
125            return Command::FAILURE;
126        }
127
128        $this->reporters->create($config)->report($result->graph, $result->symbol->id(), $output);
129
130        return Command::SUCCESS;
131    }
