7/**
8 * The five-character GQLSTATUS a query execution reports.
9 *
10 * GQL specifies that a result carries a status code beside its data, and that the
11 * code — not the message — is the part a program is allowed to test for. Messages
12 * change with the query and with the version; the code is the promise.
13 *
14 * That promise matters most for the reader this command exists for. An agent that
15 * asks peq a question and gets a failure back has to decide whether it wrote the
16 * query wrongly, asked about something that is not there, or found nothing — and
17 * those need different next moves. A five-character code says which, without anyone
18 * having to parse English.
19 *
20 * Every code here is one ISO/IEC 39075 defines. The standard publishes its conditions
21 * as a digital artifact for implementers to report them by, and `composer spec` looks
22 * every code up in it, so a code peq invented would fail the build rather than reach a
23 * reader. A condition the standard gives no
24 * subclass for is reported under its class code, which is what the class codes are
25 * for: `42000` says the query was at fault without claiming to know a subcondition
26 * that does not exist.
27 *
28 * @see https://www.iso.org/standard/76120.html ISO/IEC 39075:2024, GQL
29 */
30enum StatusCode: string
31{
32    /** The query ran and produced at least one row */
33    case Success = '00000';
34
35    /** The query ran and produced no rows */
36    case NoData = '02000';
37
38    /** A number was worked out that its type cannot hold */
39    case NumericValueOutOfRange = '22003';
40
41    /** A substring was asked for that a string cannot have */
42    case SubstringError = '22011';
43
44    /** A division by zero was attempted */
45    case DivisionByZero = '22012';
46
47    /** A value was of a type the operation cannot accept */
48    case InvalidType = '22G03';
49
50    /** Two values were compared that have no order between them */
51    case ValuesNotComparable = '22G04';
52
53    /** The query named a function, an aggregate or a statement GQL does not define here */
54    case UnknownFeature = '42000';
55
56    /** The query could not be read as GQL */
57    case SyntaxError = '42001';
58
59    /** The query named something that is not bound where it names it */
60    case InvalidReference = '42002';
61
62    /**
63     * Names the condition the code stands for, in the wording GQL uses.
64     *
65     * The standard pairs every code with a fixed phrase, and reporting that phrase
66     * rather than an invented one is what makes two implementations of GQL
67     * recognisable as the same language when they fail. A code that names a subclass
68     * is written as its class and its subclass joined by a dash, which is how both
69     * the standard's own artifact and the implementations that follow it read.
70     *
71     * @example A syntax error is reported under the standard's own wording
72     *     \App\Gql\StatusCode::SyntaxError->condition() // => 'error: syntax error or access rule violation - invalid syntax'
73     * @example So is a query that simply found nothing
74     *     \App\Gql\StatusCode::NoData->condition() // => 'note: no data'
75     *
76     * @return string The condition the code names
77     */
78    public function condition(): string
79    {
80        return match ($this) {
81            self::Success => 'note: successful completion',
82            self::NoData => 'note: no data',
83            self::NumericValueOutOfRange => 'error: data exception - numeric value out of range',
84            self::SubstringError => 'error: data exception - substring error',
85            self::DivisionByZero => 'error: data exception - division by zero',
86            self::InvalidType => 'error: data exception - invalid value type',
87            self::ValuesNotComparable => 'error: data exception - values not comparable',
88            self::UnknownFeature => 'error: syntax error or access rule violation',
89            self::SyntaxError => 'error: syntax error or access rule violation - invalid syntax',
90            self::InvalidReference => 'error: syntax error or access rule violation - invalid reference',
91        };
92    }
93
94    /**
95     * Reports whether the code stands for an execution that succeeded.
96     *
97     * Finding nothing is a success: a query that asks which controllers cache a
98     * response and finds none has answered the question.
99     *
100     * @example Finding nothing is still finding out
101     *     \App\Gql\StatusCode::NoData->succeeded() // => true
102     * @example A query that could not be read is not
103     *     \App\Gql\StatusCode::SyntaxError->succeeded() // => false
104     *
105     * @return bool True when the code is in a success class
106     */
107    public function succeeded(): bool
108    {
109        return match ($this) {
110            self::Success,
111            self::NoData => true,
112
113            self::NumericValueOutOfRange,
114            self::SubstringError,
115            self::DivisionByZero,
116            self::InvalidType,
117            self::ValuesNotComparable,
118            self::UnknownFeature,
119            self::SyntaxError,
120            self::InvalidReference => false,
121        };
122    }
123}
