31/**
32 * Reports the relations written outside method bodies.
33 *
34 * Declarations — what a class declares, what a signature says, what a property is
35 * typed as — are read straight from the tree the analyser supplies, because nothing
36 * strips them. Usage expressions are different: inside a method body the analyser's
37 * tree is incomplete, so InClassMethodCollector handles those by reading the file
38 * again. This collector therefore stops at the door of a method body, which is what
39 * keeps a call from being recorded twice.
40 *
41 * @implements Collector<PhpParserNode, list<Edge|Node>>
42 *
43 * @visibility parent
44 */
45final class DependencyCollector implements Collector
46{
47    /**
48     * Names the analyser node this collector is called for.
49     *
50     * @return class-string<PhpParserNode> The class of the node kind this collector reads
51     */
52    #[Override]
53    public function getNodeType(): string
54    {
55        return PhpParserNode::class;
56    }
57
58    /**
59     * Reports what one node of the syntax tree declares or uses.
60     *
61     * A node is read as a declaration first. Only if it is none is it read as a
62     * usage, and then only outside a method body, because inside one the same
63     * expression is read from the re-parsed source instead.
64     *
65     * @param PhpParserNode $node  The node the analyser reached
66     * @param Scope         $scope The analyser scope it was reached in
67     *
68     * @return null|list<Edge|Node> The relations found, or null when there are none
69     */
70    #[Override]
71    public function processNode(PhpParserNode $node, Scope $scope): ?array
72    {
73        $items = self::declaration($node, $scope);
74        if ($items === null) {
75            $inClassMethod = $scope->isInClass() && $scope->getFunction() !== null;
76            $items = $inClassMethod ? [] : (self::usage($node, $scope) ?? []);
77        }
78
79        return $items === [] ? null : $items;
80    }
81
82    /**
83     * Reads a node as a declaration, if it is one.
84     *
85     * @param PhpParserNode $node  The node the analyser reached
86     * @param Scope         $scope The analyser scope it was reached in
87     *
88     * @return null|list<Edge|Node> What the declaration describes, or null when the node declares nothing
89     */
90    public static function declaration(PhpParserNode $node, Scope $scope): ?array
91    {
92        return match (true) {
93            $node instanceof ClassLike => ClassLikeProcessor::process($node, $scope),
94            $node instanceof PhpParserNode\FunctionLike => FunctionLikeProcessor::process($node, $scope),
95            $node instanceof Property => PropertyProcessor::process($node, $scope),
96            $node instanceof ClassConst => ClassConstProcessor::process($node, $scope),
97            $node instanceof EnumCase => EnumCaseProcessor::process($node, $scope),
98            $node instanceof PhpParserNode\Param => PromotedPropertyProcessor::process($node, $scope),
99            default => null,
100        };
101    }
102
103    /**
104     * Reads a node as a usage expression, if it is one.
105     *
106     * Calls and reads on `$this` are not among them: outside a method body there is
107     * no `$this` to write them on, and inside one they are read by the other collector.
108     *
109     * @param PhpParserNode $node  The node the analyser reached
110     * @param Scope         $scope The analyser scope it was reached in
111     *
112     * @return null|list<Edge|Node> What the usage describes, or null when the node uses nothing
113     */
114    public static function usage(PhpParserNode $node, Scope $scope): ?array
115    {
116        return match (true) {
117            $node instanceof PhpParserNode\Expr\ClassConstFetch => ConstFetchProcessor::process($node, $scope),
118            $node instanceof PhpParserNode\Expr\New_ => InstantiationProcessor::process($node, $scope),
119            $node instanceof PhpParserNode\Expr\StaticCall => StaticCallProcessor::process($node, $scope),
120            $node instanceof PhpParserNode\Stmt\Catch_ => CatchProcessor::process($node, $scope),
121            $node instanceof PhpParserNode\Expr\Instanceof_ => InstanceofProcessor::process($node, $scope),
122            $node instanceof PhpParserNode\Expr\FuncCall => FunctionCallProcessor::process($node, $scope),
123            $node instanceof PhpParserNode\Expr\StaticPropertyFetch => StaticPropertyAccessProcessor::process($node, $scope),
124            default => null,
125        };
126    }
127}
