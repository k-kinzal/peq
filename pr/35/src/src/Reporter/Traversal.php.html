13/**
14 * Interface for graph traversal strategies.
15 *
16 * A traversal decides which relations it follows out of a node and in what order
17 * it hands the nodes it reaches to its visitor. The direction it reads the graph
18 * in is part of its identity rather than a per-call argument, so a consumer that
19 * needs to know which edges a traversal would follow — a renderer working out
20 * whether a node is the last of its siblings, for instance — asks for it once.
21 */
22interface Traversal
23{
24    /**
25     * Traverses the graph starting from the given symbol.
26     *
27     * @param Graph                    $graph   The dependency graph to traverse
28     * @param NodeId<Node>             $symbol  The starting symbol identifier
29     * @param Closure(Node, int): bool $visitor Invoked for each visited node with the node and its
30     *                                          depth; returns true to descend into that node, false
31     *                                          to stop at it
32     */
33    public function traverse(Graph $graph, NodeId $symbol, Closure $visitor): void;
34
35    /**
36     * Returns the direction in which this traversal reads the graph.
37     *
38     * @return Direction The direction whose edge kinds this traversal follows
39     */
40    public function direction(): Direction;
41}
