7/**
8 * One step along a path, as a tree shows it.
9 *
10 * A path a query bound is a chain of relations and symbols, and a tree of paths is
11 * those chains with their common beginnings drawn once. A step is therefore the pair
12 * that makes a chain comparable: how the reader got here, and where here is.
13 *
14 * @visibility App\Reporter\Query
15 */
16final readonly class TreeStep
17{
18    /**
19     * @param string $label     What relation was followed to get here, empty at the start of a path
20     * @param string $id        What identifies the symbol arrived at
21     * @param bool   $backwards Whether the relation was followed against its direction, from where it points to where it starts
22     */
23    public function __construct(
24        public string $label,
25        public string $id,
26        public bool $backwards = false,
27    ) {}
28
29    /**
30     * Returns what tells this step apart from any other.
31     *
32     * Two paths share a beginning when they took the same relations to the same
33     * symbols, the same way round, which means every part of a step counts: a class
34     * reached by being extended and by being called is reached two different ways, and
35     * so is a method reached by calling it and by being called by it.
36     *
37     * @example A step is told apart by how it was reached and where it arrived
38     *     (new \App\Reporter\Query\TreeStep('calls', 'App\\Money::add'))->key() // => "calls\0>\0App\\Money::add"
39     * @example A relation followed backwards reaches it another way
40     *     (new \App\Reporter\Query\TreeStep('calls', 'App\\Money::add', true))->key() // => "calls\0<\0App\\Money::add"
41     *
42     * @return string What identifies the step
43     */
44    public function key(): string
45    {
46        return $this->label."\0".($this->backwards ? '<' : '>')."\0".$this->id;
47    }
48}
