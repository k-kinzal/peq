14/**
15 * Writes the piece of graph a query's answer holds, as a Graphviz digraph.
16 *
17 * The same piece of graph the terminal drawing shows, in the form a renderer takes.
18 * A query is what makes this worth doing: a digraph of a whole codebase is a hairball,
19 * and a digraph of the twelve symbols a question narrowed down to is a picture.
20 *
21 * @visibility App\Reporter
22 */
23final readonly class DotWriter implements QueryReporter
24{
25    /**
26     * @param null|ElementGraph $graph The graph the answer came from, so the arrows between its symbols can be drawn
27     */
28    public function __construct(
29        private ?ElementGraph $graph = null,
30    ) {}
31
32    /**
33     * Writes the answer as a digraph.
34     *
35     * @param ResultTable     $result What the query answered
36     * @param OutputInterface $output Where it is written
37     *
38     * @throws QueryOutputException When a nonempty result contains no graph elements
39     */
40    #[Override]
41    public function report(ResultTable $result, OutputInterface $output): void
42    {
43        $diagram = ResultElements::of($result, $this->graph);
44        if ($diagram->empty()) {
45            if ($result->rows !== []) {
46                throw new QueryOutputException('Graph output requires nodes, edges or paths, but the result contains none. Return elements (for example, RETURN n instead of RETURN n.id), or use --output=table or --output=json.');
47            }
48
49            return;
50        }
51
52        $output->writeln('digraph peq {', OutputInterface::OUTPUT_RAW);
53        $output->writeln('    rankdir=LR;', OutputInterface::OUTPUT_RAW);
54        foreach ($diagram->nodes() as $node) {
55            $output->writeln(self::vertex($node), OutputInterface::OUTPUT_RAW);
56        }
57        foreach ($diagram->nodes() as $node) {
58            foreach ($diagram->leaving($node->id) as $edge) {
59                $output->writeln(self::arrow($edge), OutputInterface::OUTPUT_RAW);
60            }
61        }
62        $output->writeln('}', OutputInterface::OUTPUT_RAW);
63    }
64
65    /**
66     * Writes one symbol as a Graphviz statement.
67     *
68     * @param DiagramNode $node The symbol
69     *
70     * @example A symbol is a box labelled with its name and what it is
71     *     \App\Reporter\Query\DotWriter::vertex(new \App\Reporter\Diagram\DiagramNode('App\\Invoice', 'class')) // => '    "App\\\\Invoice" [label="App\\\\Invoice\\nclass"];'
72     *
73     * @return string The statement
74     */
75    public static function vertex(DiagramNode $node): string
76    {
77        $label = $node->kind === '' ? self::quoted($node->id) : self::quoted($node->id).'\n'.self::quoted($node->kind);
78
79        return sprintf('    "%s" [label="%s"];', self::quoted($node->id), $label);
80    }
81
82    /**
83     * Writes one relation as a Graphviz statement.
84     *
85     * @param DiagramEdge $edge The relation
86     *
87     * @example A relation is an arrow labelled with what it is
88     *     \App\Reporter\Query\DotWriter::arrow(new \App\Reporter\Diagram\DiagramEdge('a', 'b', 'calls')) // => '    "a" -> "b" [label="calls"];'
89     *
90     * @return string The statement
91     */
92    public static function arrow(DiagramEdge $edge): string
93    {
94        return sprintf(
95            '    "%s" -> "%s" [label="%s"];',
96            self::quoted($edge->origin),
97            self::quoted($edge->target),
98            self::quoted($edge->label),
99        );
100    }
101
102    /**
103     * Writes a piece of text so that Graphviz reads it as text.
104     *
105     * A PHP name is full of backslashes and a Graphviz string treats a backslash as
106     * an escape, so a name written straight through would come back as something else
107     * — or not at all.
108     *
109     * @param string $text The text
110     *
111     * @example A namespace separator survives being written
112     *     \App\Reporter\Query\DotWriter::quoted('App\\Invoice') // => 'App\\\\Invoice'
113     * @example And so does a quotation mark
114     *     \App\Reporter\Query\DotWriter::quoted('say "hi"') // => 'say \\"hi\\"'
115     *
116     * @return string The text, escaped for Graphviz
117     */
118    public static function quoted(string $text): string
119    {
120        return str_replace(['\\', '"'], ['\\\\', '\"'], $text);
121    }
