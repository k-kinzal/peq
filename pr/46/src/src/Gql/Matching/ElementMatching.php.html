17/**
18 * Whether one element is what a pattern asked for.
19 *
20 * Three things have to hold, and the order they are asked in is the order they get
21 * cheaper to be wrong about. The labels first, because they rule out most of a graph
22 * for nothing; then the properties written as equalities, which are a lookup each;
23 * then the predicate, which is the only one that has to evaluate an expression.
24 *
25 * The predicate sees the element bound to its own variable, which is what makes
26 * `(p:Method WHERE p.visibility = 'public')` mean what it looks like. For an edge
27 * crossed by a repeating pattern the variable is bound to the one edge being
28 * considered rather than to the list of all of them, which is GQL's rule and the
29 * reason a repeating pattern can be narrowed edge by edge.
30 *
31 * @visibility App\Gql
32 */
33final class ElementMatching
34{
35    /**
36     * Reports whether a symbol satisfies what a pattern requires of it.
37     *
38     * @param NodeDatum            $node       The symbol
39     * @param null|string          $variable   The name the pattern binds it to, if any
40     * @param ElementFilter        $filter     What the pattern requires beyond its labels
41     * @param BindingRow           $row        What is already bound
42     * @param ExpressionEvaluation $evaluation How a predicate is worked out
43     *
44     * @example A symbol with the property a pattern asked for satisfies it
45     *     $node = new \App\Gql\Datum\NodeDatum('a', [], ['kind' => new \App\Gql\Datum\StringDatum('method')]);
46     *     $parser = new \App\Gql\Parsing\ExpressionParser(\App\Gql\Parsing\TokenReader::of("'method'"));
47     *     $filter = new \App\Gql\Syntax\Pattern\ElementFilter(['kind' => $parser->parse()]);
48     *     \App\Gql\Matching\ElementMatching::satisfies($node, null, $filter, \App\Gql\Binding\BindingRow::unit(), new \App\Gql\Evaluation\ExpressionEvaluation()) // => true
49     *
50     * @return bool True when it does
51     *
52     * @throws GqlException If a requirement cannot be worked out
53     */
54    public static function satisfies(
55        EdgeDatum|NodeDatum $node,
56        ?string $variable,
57        ElementFilter $filter,
58        BindingRow $row,
59        ExpressionEvaluation $evaluation,
60    ): bool {
61        if ($filter->empty()) {
62            return true;
63        }
64
65        $scope = $variable === null ? $row : $row->with($variable, $node);
66        foreach ($filter->properties as $name => $expected) {
67            if (DatumOrder::equals($node->property($name), $evaluation->evaluate($expected, $scope)) !== true) {
68                return false;
69            }
70        }
71        if ($filter->predicate === null) {
72            return true;
73        }
74
75        return Logic::holds($evaluation->evaluate($filter->predicate, $scope));
76    }
77
78    /**
79     * Reports whether an element agrees with what a name is already bound to.
80     *
81     * Writing the same name twice in one pattern says the two places must be the same
82     * element, which is how a query asks for two methods of the same class or two
83     * controllers that reach the same repository. It is also how a pattern joins to
84     * what an earlier clause found.
85     *
86     * @param null|string $variable The name the pattern binds, if any
87     * @param Datum       $element  The element being considered
88     * @param BindingRow  $row      What is already bound
89     *
90     * @example A name nothing bound agrees with anything
91     *     \App\Gql\Matching\ElementMatching::agrees('p', new \App\Gql\Datum\NodeDatum('a'), \App\Gql\Binding\BindingRow::unit()) // => true
92     * @example A name bound elsewhere agrees only with the same element
93     *     $row = \App\Gql\Binding\BindingRow::unit()->with('p', new \App\Gql\Datum\NodeDatum('a'));
94     *     \App\Gql\Matching\ElementMatching::agrees('p', new \App\Gql\Datum\NodeDatum('b'), $row) // => false
95     *
96     * @return bool True when it agrees
97     *
98     * @throws GqlException If the name is bound to a value that cannot be compared with an element
99     */
100    public static function agrees(?string $variable, Datum $element, BindingRow $row): bool
101    {
102        if ($variable === null || !$row->has($variable)) {
103            return true;
104        }
105
106        return DatumOrder::equals($row->value($variable), $element) === true;
107    }
108}
