14/**
15 * Reads what a node or an edge pattern writes between its brackets.
16 *
17 * A node pattern and an edge pattern hold the same three things, in the same order —
18 * the name the element is bound to, what its labels must be, and a predicate or the
19 * properties it must carry — so they are read here once for both.
20 *
21 * @visibility App\Gql\Parsing
22 */
23final readonly class ElementParser
24{
25    /**
26     * Reads what a pattern requires of an element's labels.
27     */
28    private LabelParser $labels;
29
30    /**
31     * @param TokenReader      $tokens      The pieces of the query being read
32     * @param ExpressionParser $expressions Where a value or a predicate is written inside a pattern
33     */
34    public function __construct(
35        private TokenReader $tokens,
36        private ExpressionParser $expressions,
37    ) {
38        $this->labels = new LabelParser($tokens);
39    }

41    /**
42     * Reads the name a matched element is bound to, if one is written here.
43     *
44     * What settles whether a name is written here is GQL's rule that a binding
45     * variable is a word the standard does not reserve. That is why `(n WHERE ...)`
46     * needs no special case: `WHERE` is reserved, so it was never a name.
47     *
48     * @example A name written first binds the element
49     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of('p:Method)'), new \App\Gql\Parsing\ExpressionParser($tokens));
50     *     $parser->parseName() // => 'p'
51     * @example A word GQL reserves is not a name, and is refused where it stands as one
52     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of('function)'), new \App\Gql\Parsing\ExpressionParser($tokens));
53     *     $parser->parseName() // throws \App\Gql\GqlException: GQL reserves "FUNCTION"
54     * @example A pattern that starts with a requirement binds nothing
55     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of(':Method)'), new \App\Gql\Parsing\ExpressionParser($tokens));
56     *     $parser->parseName() // => null
57     *
58     * @return null|string The name, or null when the match is not named
59     *
60     * @throws GqlException If the name cannot be read
61     */
62    public function parseName(): ?string
63    {
64        $token = $this->tokens->current();
65        $next = $this->tokens->peek();
66        $namedLikeAVariable = $next->isSymbol(')') || $next->isSymbol(']') || $next->isSymbol(':') || $next->isSymbol('{');
67        if ($token->kind === TokenKind::Name && ReservedWords::reserves($token->value) && $namedLikeAVariable) {
68            NameReader::refuseReserved($token, 'a query binds a name to a word GQL leaves free');
69        }
70        if (!NameReader::atVariable($this->tokens)) {
71            return null;
72        }
73
74        return NameReader::variable($this->tokens);
75    }

77    /**
78     * Reads what a pattern requires of an element's labels, if it requires anything.
79     *
80     * GQL's `<is or colon>` introduces the requirement, so `(n:Method)` and
81     * `(n IS Method)` say the same.
82     *
83     * @example A requirement is written after a colon
84     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of(':Method)'), new \App\Gql\Parsing\ExpressionParser($tokens));
85     *     $parser->parseLabels()?->name // => 'Method'
86     * @example A pattern with no colon requires nothing of them
87     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of(')'), new \App\Gql\Parsing\ExpressionParser($tokens));
88     *     $parser->parseLabels() // => null
89     *
90     * @return null|LabelPattern The requirement, or null when none is written
91     *
92     * @throws GqlException If what is written is not a requirement
93     */
94    public function parseLabels(): ?LabelPattern
95    {
96        return $this->tokens->acceptSymbol(':') || $this->tokens->acceptKeyword('IS') ? $this->labels->parse() : null;
97    }
98
99    /**
100     * Reads what a pattern requires of an element beyond its labels.
101     *
102     * GQL's `<element pattern predicate>` is a property specification or a `WHERE`
103     * clause, not both: `(n {name: 'x'} WHERE n.line > 1)` is written
104     * `(n WHERE n.name = 'x' AND n.line > 1)`.
105     *
106     * @example Properties are required to equal what they are written against
107     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of("{name: 'Invoice'})"), new \App\Gql\Parsing\ExpressionParser($tokens));
108     *     array_keys($parser->parseFilter()->properties) // => ['name']
109     * @example A predicate is required to hold
110     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of('WHERE p.line > 10)'), new \App\Gql\Parsing\ExpressionParser($tokens));
111     *     $parser->parseFilter()->predicate !== null // => true
112     *
113     * @return ElementFilter The requirement
114     *
115     * @throws GqlException If what is written is not a requirement
116     */
117    public function parseFilter(): ElementFilter
118    {
119        if ($this->tokens->atSymbol('{')) {
120            return new ElementFilter($this->parseProperties());
121        }
122
123        return new ElementFilter([], $this->tokens->acceptKeyword('WHERE') ? $this->expressions->parse() : null);
124    }
125
126    /**
127     * Reads the properties a pattern requires an element to carry.
128     *
129     * @example Every property written has to equal what it is written against
130     *     $parser = new \App\Gql\Parsing\ElementParser($tokens = \App\Gql\Parsing\TokenReader::of("{kind: 'method', visibility: 'public'}"), new \App\Gql\Parsing\ExpressionParser($tokens));
131     *     array_keys($parser->parseProperties()) // => ['kind', 'visibility']
132     *
133     * @return array<string, Expression> The properties, by name
134     *
135     * @throws GqlException If what is written is not a list of properties
136     */
137    public function parseProperties(): array
138    {
139        $this->tokens->expectSymbol('{');
140        if ($this->tokens->acceptSymbol('}')) {
141            return [];
142        }
143
144        $properties = [];
145        do {
146            $name = NameReader::identifier($this->tokens);
147            $this->tokens->expectSymbol(':');
148            $properties[$name] = $this->expressions->parse();
149        } while ($this->tokens->acceptSymbol(','));
150        $this->tokens->expectSymbol('}');
151
152        return $properties;
153    }
154}
