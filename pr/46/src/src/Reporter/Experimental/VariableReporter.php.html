25     * Writes the same selected graph in the requested format.
26     *
27     * @throws JsonException If the requested analysis or encoding is rejected
28     */
29    public function report(Slice $slice, OutputFormat $format, OutputInterface $output): void
30    {
31        $text = match ($format) {
32            OutputFormat::Json => json_encode([
33                'experimental' => true,
34                'semantics' => 'Possible intraprocedural dependencies; call-input and boundary-input edges describe inputs, not proven return or heap dependencies.',
35                'target' => $slice->target, 'file' => $slice->file, 'direction' => $slice->direction->value,
36                'roots' => $slice->roots, 'nodes' => $slice->nodes, 'edges' => $slice->edges, 'diagnostics' => $slice->diagnostics,
37            ], JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_INVALID_UTF8_SUBSTITUTE | JSON_THROW_ON_ERROR),
38            OutputFormat::Tree => (new VariableTree())->render($slice),
39            OutputFormat::Table => $this->table($slice),
40            OutputFormat::Dot => $this->dot($slice),
41            OutputFormat::Graph, OutputFormat::Mermaid => $this->diagram($slice, $format),
42        };
43        $output->writeln($text, OutputInterface::OUTPUT_RAW);
44        if ($format !== OutputFormat::Json && $slice->diagnostics !== []) {
45            $prefix = $format === OutputFormat::Dot ? '// ' : ($format === OutputFormat::Mermaid ? '%% ' : '');
46            foreach ($slice->diagnostics as $diagnostic) {
47                $output->writeln($prefix.$diagnostic, OutputInterface::OUTPUT_RAW);
48            }
49        }
50    }

52    /**
53     * Renders a selected graph with the existing diagram infrastructure.
54     */
55    public function diagram(Slice $slice, OutputFormat $format): string
56    {
57        $diagram = new Diagram();
58        foreach ($slice->nodes as $node) {
59            $diagram->add(new DiagramNode($node->id, $node->kind, $slice->file.':'.$node->line.':'.$node->column));
60        }
61        foreach ($slice->edges as $edge) {
62            $diagram->relate(new DiagramEdge($edge->from, $edge->to, $edge->kind.($edge->branch === null ? '' : ' ('.$edge->branch.')')));
63        }
64        $renderer = $format === OutputFormat::Mermaid ? new MermaidRenderer() : new TerminalRenderer((new Terminal())->getWidth());
65
66        $output = new BufferedOutput();
67        $renderer->render($diagram, $output);
68
69        return rtrim($output->fetch());
70    }

72    /**
73     * Writes one row per dependency of the selected graph.
74     */
75    public function table(Slice $slice): string
76    {
77        $lines = ['FROM | DEPENDENCY | TO'];
78        foreach ($slice->edges as $edge) {
79            $lines[] = $edge->from.' | '.$edge->kind.($edge->branch === null ? '' : ' ('.$edge->branch.')').' | '.$edge->to;
80        }
81        $connected = [];
82        foreach ($slice->edges as $edge) {
83            $connected[$edge->from] = true;
84            $connected[$edge->to] = true;
85        }
86        foreach ($slice->nodes as $node) {
87            if (!isset($connected[$node->id])) {
88                $lines[] = $node->id.' | (no selected edges) |';
89            }
90        }
91
92        return implode("\n", $lines);
93    }

108                .' [label='.json_encode($edge->kind.($edge->branch === null ? '' : ' ('.$edge->branch.')'), JSON_THROW_ON_ERROR).'];';
109        }
110        $lines[] = '}';
111
112        return implode("\n", $lines);
