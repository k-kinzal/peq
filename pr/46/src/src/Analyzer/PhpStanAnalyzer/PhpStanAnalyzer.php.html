20/**
21 * Builds the dependency graph by having PHPStan analyse real sources.
22 *
23 * Resolving what a name refers to is the hard part of reading PHP, and PHPStan
24 * already does it: this analyzer runs PHPStan over the selected files with peq's own
25 * collectors registered, and assembles the nodes and edges they report. Using
26 * PHPStan's internals is deliberate — its resolved types are the reason the graph
27 * can tell a call to one class from a call to another of the same method name.
28 */
29final readonly class PhpStanAnalyzer implements Analyzer
30{
31    /**
32     * The identifier PHPStan gives an error that stands for a failure of its own.
33     *
34     * Every other error is a diagnostic about the analysed code, which a dependency
35     * graph has no use for — including the syntax error of a file no PHP version
36     * would accept, which PHPStan reads past and so does peq.
37     */
38    private const string INTERNAL_ERROR = 'phpstan.internal';
39
40    /**
41     * @param list<string>       $includes         File path patterns to include in analysis
42     * @param list<string>       $excludes         File path patterns to exclude from analysis
43     * @param null|int           $phpVersion       The PHP version the sources are read as, in PHP_VERSION_ID
44     *                                             form, or null to leave the version to the analysis engine
45     * @param ContainerFactory   $containerFactory Builds the PHPStan container that runs the collectors
46     * @param PhpFileCollector   $fileCollector    Selects which files the analysis covers
47     * @param GraphBuilder       $graphBuilder     Assembles the graph from what the collectors reported
48     * @param list<class-string> $collectors       The collectors the graph is assembled from: declarations and method bodies unless narrowed
49     */
50    public function __construct(
51        private array $includes = [],
52        private array $excludes = [],
53        private ?int $phpVersion = null,
54        private ContainerFactory $containerFactory = new ContainerFactory(),
55        private PhpFileCollector $fileCollector = new PhpFileCollector(),
56        private GraphBuilder $graphBuilder = new GraphBuilder(),
57        private array $collectors = [DependencyCollector::class, InClassMethodCollector::class],
58    ) {}
59
60    /**
61     * Analyses the sources under the given path and builds their dependency graph.
62     *
63     * A path that holds no analysable file is not a failure: it produces an empty
64     * graph, which is what the caller would otherwise have to construct itself.
65     *
66     * @param string $path The file or directory to analyse
67     *
68     * @return Graph The dependency graph of the analysed sources
69     *
70     * @throws AnalysisFailedException If PHPStan cannot be configured to run the collectors,
71     *                                 or if it could not finish analysing one of the files
72     */
73    #[Override]
74    public function analyze(string $path): Graph
75    {
76        $realPath = realpath($path);
77        $files = $this->fileCollector->collect(
78            [$realPath !== false ? $realPath : $path],
79            $this->includes,
80            $this->excludes,
81        );
82
83        if ($files === []) {
84            return new Graph();
85        }
86
87        $report = $this->collect($this->containerFactory->create($files, $this->collectors, $this->phpVersion), $files);
88
89        return $this->graphBuilder->build($report->symbols());
90    }

92    /**
93     * Runs the analysis and reads back what peq's collectors reported.
94     *
95     * Running PHPStan in process is what gives peq the resolved types a dependency
96     * graph needs, and this is the only place that does it: the container, the
97     * analyser and the shape of its result are PHPStan's own vocabulary, and every
98     * other part of peq works on the symbols this hands back.
99     *
100     * PHPStan parses a file in full only when its parser has been told the file is
101     * one of those analysed; any other file is read with its function bodies
102     * stripped, which is enough for reflection but leaves nothing for a collector
103     * to see inside a function. PHPStan's own commands register the analysed files
104     * with the parser before analysing, and so does this.
105     *
106     * A file PHPStan could not finish is reported as a failure rather than left out:
107     * a graph that quietly lacks a file would claim that nothing depends on what
108     * that file declares, which is the one answer an impact analysis must never get
109     * wrong. PHPStan records such a failure as an internal error next to the
110     * diagnostics peq discards, so those are read before the collected data is. The
111     * message names the PHP version the sources were read as, because a file the
112     * analysis could not get through is most often one written for another version.
113     *
114     * @param Container    $container The container the analysis runs in
115     * @param list<string> $files     The files to analyse
116     *
117     * @return CollectorReport What the collectors reported
118     *
119     * @throws AnalysisFailedException If the container was built without a parser or an
120     *                                 analyser, or if PHPStan could not finish analysing one of the files
121     */
122    public function collect(Container $container, array $files): CollectorReport
123    {
124        try {
125            $parser = $container->getService('pathRoutingParser');
126            $analyser = $container->getByType(PhpStanAnalyser::class);
127        } catch (MissingServiceException $missing) {
128            throw new AnalysisFailedException(
129                'PHPStan was built without its parser or its analyser, so no source can be analysed.',
130                0,
131                $missing,
132            );
133        }
134        if (!is_object($parser) || $parser::class !== PathRoutingParser::class) {
135            throw new AnalysisFailedException('PHPStan has no parser to tell which files are analysed, so no source can be analysed.');
136        }
137        $parser->setAnalysedFiles($files);
138
139        $result = $analyser->analyse($files, null, null, false, $files);
140        $failures = array_values(array_filter(
141            $result->getErrors(),
142            static fn (Error $error): bool => $error->getIdentifier() === self::INTERNAL_ERROR,
143        ));
144        if ($failures !== []) {
145            throw new AnalysisFailedException(implode(PHP_EOL, array_map(
146                fn (Error $failure): string => sprintf(
147                    'PHPStan could not finish analysing %s as %s: %s',
148                    $failure->getFile(),
149                    $this->analysedVersion(),
150                    $failure->getMessage(),
151                ),
152                $failures,
153            )));
154        }
155
156        return CollectorReport::of($result->getCollectedData(), $this->collectors);
157    }
158
159    /**
160     * Names the PHP version the sources are read as.
161     *
162     * A file that could not be read is nearly always a file written for a different
163     * version than the one it was read as, so the message about it says which version
164     * that was — including when nobody chose it.
165     *
166     * @return string The version as a project would write it, or a name for the unstated one
167     */
168    public function analysedVersion(): string
169    {
170        if ($this->phpVersion === null) {
171            return 'the PHP version peq runs on';
172        }
173
174        return sprintf('PHP %d.%d', intdiv($this->phpVersion, 10000), intdiv($this->phpVersion, 100) % 100);
175    }
