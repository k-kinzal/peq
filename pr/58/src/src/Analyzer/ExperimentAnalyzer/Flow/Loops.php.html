44            return $outcome;

47        $inputs = $recurrence->open($node, $state, $back);
48        [$outcome, $back] = $this->pass($node, $state, $iterable);
49        $recurrence->close($node, $inputs, $back);

56     *
57     * @return array{Exits, list<State>}
58     */
59    public function pass(Stmt\Do_|Stmt\For_|Stmt\Foreach_|Stmt\While_ $node, State $state, ?string $iterable): array
60    {
61        $transfers = new LoopTransfers($this->statements->expressions);
62        $truth = $transfers->truth($node);
63        $exits = [];
64        if (!$node instanceof Stmt\Do_) {
65            $condition = $transfers->condition($node, $state, $iterable);
66            if ($truth !== true) {
67                $exit = clone $state;
68                $exit->controls[$condition] = $node instanceof Stmt\Foreach_ ? 'exhausted' : 'falsy';
69                $exits[] = $exit;
70            }
71            if ($truth === false) {
72                return [new Exits($exits[0]), []];
73            }
74            $state->controls[$condition] = $node instanceof Stmt\Foreach_ ? 'iterate' : 'truthy';
75            if ($node instanceof Stmt\Foreach_) {
76                $transfers->bind($node, $state, $condition);
77            }
78        }
79        $body = $this->statements->read($node->stmts, $state);
80        $back = $body->continues[1] ?? [];
81        if ($body->normal !== null) {
82            $back[] = $body->normal;
83        }
84        [$back, $tests] = $this->advance($node, $back, $iterable);
85        array_push($exits, ...($body->breaks[1] ?? []), ...$tests);
86        $result = new Exits($exits === [] ? null : (new Guards())->join($node, $exits, $this->statements->expressions->recording->graph));
87        $result->absorb($body, true);
88
89        return [$result, $back];
90    }

106            }
107            if ($node instanceof Stmt\Do_) {
108                $condition = $transfers->condition($node, $path, $iterable);
109                if ($transfers->truth($node) !== true) {

116        }
117
